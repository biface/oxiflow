1//! # Module `mesh`
2//!
3//! Spatial mesh abstraction — INV-1 invariant ([DA-007](https://github.com/biface/oxiflow/issues/7), [DA-019](https://github.com/biface/oxiflow/issues/56), issue [#31](https://github.com/biface/oxiflow/issues/31)).
4//!
5//! ## Core principle (INV-1)
6//!
7//! No public engine API exposes `dx`, `nx` or raw indices as first-class spatial
8//! parameters. All spatial references go through the `Mesh` trait. This guarantees
9//! forward compatibility with FEM unstructured meshes (v0.11.0 (J11)) — zero breaking change
10//! on existing code when `unstructured/` is added.
11//!
12//! ## Module hierarchy ([DA-019](https://github.com/biface/oxiflow/issues/56))
13//!
14//! ```text
15//! src/mesh/
16//! ├── mod.rs              — Mesh trait + public re-exports
17//! ├── structured/
18//! │   └── mod.rs          — UniformGrid1D (J1); nD generalisation at v0.11.5 (DD-051)
19//! └── unstructured/       — RESERVED v0.11.0 (J11) (UnstructuredMesh2D, TetrahedralMesh3D)
20//! ```
21//!
22//! ## Implementations
23//!
24//! | Type | Family | Description | Milestone |
25//! |---|---|---|---|
26//! | [`UniformGrid1D`] | structured | 1D uniform grid, FD/FV | J1 — v0.1.0 |
27//! | `UnstructuredMesh2D` | unstructured | 2D triangular mesh, FEM | J11 — v0.11.0 |
28//! | `TetrahedralMesh3D` | unstructured | 3D tetrahedral mesh, FEM | J11 — v0.11.0 |
29//!
30//! ## Boundary regions ([DA-071](https://github.com/biface/oxiflow/issues/203))
31//!
32//! The mesh, not the boundary conditions, knows where its boundary is.
33//! [`BoundaryRegion`] names a region (today a [`BoundaryRegion::Face`] of a
34//! structured grid), [`Mesh::boundary_nodes`] lists its nodes and
35//! [`Mesh::boundary_neighbours`] pairs each with its first interior
36//! neighbour. Both methods have a default for 1D meshes whose nodes are
37//! ordered along the axis.
38

52// ── Boundary regions (DA-071) ─────────────────────────────────────────────────
53
54/// A region of the boundary of a mesh ([DA-071](https://github.com/biface/oxiflow/issues/203)).
55///
56/// Boundary conditions designate the nodes they act on through a region,
57/// which the mesh resolves with [`Mesh::boundary_nodes`] and
58/// [`Mesh::boundary_neighbours`], rather than through node indices (INV-1).
59///
60/// `#[non_exhaustive]`: the boundaries of unstructured meshes, identified by
61/// node tags (Gmsh physical groups, [DD-050](https://github.com/biface/oxiflow/issues/151)), will be a further variant.
62///
63/// # Examples
64///
65/// ```rust
66/// use oxiflow::mesh::{BoundaryRegion, FaceSide, Mesh, UniformGrid1D};
67///
68/// let grid = UniformGrid1D::new(5, 0.0, 1.0)?;
69/// let outlet = BoundaryRegion::Face { axis: 0, side: FaceSide::Upper };
70/// assert_eq!(grid.boundary_nodes(&outlet)?, vec![4]);
71///
72/// let neighbours = grid.boundary_neighbours(&outlet)?;
73/// assert_eq!((neighbours[0].node, neighbours[0].inner), (4, 3));
74/// assert_eq!(neighbours[0].distance, 0.25);
75/// # Ok::<(), oxiflow::context::OxiflowError>(())
76/// ```
77#[non_exhaustive]
78#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
79#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
80pub enum BoundaryRegion {
81    /// Face of a structured grid normal to `axis` (0 for x, 1 for y, 2 for
82    /// z), on the given `side`. On a 1D grid, the two ends.
83    Face {
84        /// Axis the face is normal to.
85        axis: usize,
86        /// Which of the two faces normal to `axis`.
87        side: FaceSide,
88    },
89}
90
91/// Which of the two faces normal to an axis ([DA-071](https://github.com/biface/oxiflow/issues/203)).
92#[non_exhaustive]
93#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
94#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
95pub enum FaceSide {
96    /// The face where the coordinate along the axis is smallest.
97    Lower,
98    /// The face where the coordinate along the axis is largest.
99    Upper,
100}
101
102/// A node of a boundary region with its first neighbour along the inward
103/// normal, as returned by [`Mesh::boundary_neighbours`] ([DA-071](https://github.com/biface/oxiflow/issues/203)).
104///
105/// `#[non_exhaustive]` so that further fields (a normal, a weight) can be
106/// added; a mesh implemented outside the crate builds it with
107/// [`BoundaryNode::new`].
108#[non_exhaustive]
109#[derive(Debug, Clone, Copy, PartialEq)]
110pub struct BoundaryNode {
111    /// Index of the boundary node.
112    pub node: usize,
113    /// Index of its first neighbour along the inward normal.
114    pub inner: usize,
115    /// Distance between the two nodes, positive.
116    pub distance: f64,
117}
118
119impl BoundaryNode {
120    /// A boundary node, its inner neighbour and their distance.
121    pub fn new(node: usize, inner: usize, distance: f64) -> Self {
122        Self {
123            node,
124            inner,
125            distance,
126        }
127    }
128}

130/// Checks that `region` is a face the default boundary methods resolve: a
131/// face normal to axis 0 of a 1D mesh with at least `min_nodes` nodes.
132/// Returns the node count and the face side.
133fn default_face(
134    mesh: &(impl Mesh + ?Sized),
135    region: &BoundaryRegion,
136    min_nodes: usize,
137    context: &'static str,
138) -> Result<(usize, FaceSide), OxiflowError> {
139    let refuse = |message: String| OxiflowError::PreconditionFailed { context, message };
140    let BoundaryRegion::Face { axis, side } = region;
141    let dimension = mesh.spatial_dimension();
142    if dimension != 1 || *axis != 0 {
143        return Err(refuse(format!(
144            "{region:?} is not defined by this mesh: the default implementation covers \
145             axis 0 of 1D meshes only, got spatial_dimension = {dimension}"
146        )));
147    }
148    let n = mesh.n_dof();
149    if n < min_nodes {
150        return Err(refuse(format!(
151            "{region:?} needs at least {min_nodes} node(s), the mesh has {n}"
152        )));
153    }
154    Ok((n, *side))
155}

157// ── Mesh trait ────────────────────────────────────────────────────────────────
158
159/// Abstract spatial mesh — INV-1 invariant.
160///
161/// All spatial information consumed by the engine passes through this trait.
162/// No implementation detail (`dx`, `nx`, node table) leaks into the public API.
163///
164/// # Object safety
165///
166/// This trait is object-safe: it can be used as `Box<dyn Mesh>` and `&dyn Mesh`.
167/// Required for INV-4 (plugin-safe API, v0.30.0 / J30, [DA-017](https://github.com/biface/oxiflow/issues/17)).
168///
169/// # Implementing `Mesh`
170///
171/// A minimal implementation for a 1D uniform grid:
172///
173/// ```rust
174/// use oxiflow::mesh::Mesh;
175///
176/// struct MyGrid { n: usize, dx: f64 }
177///
178/// impl Mesh for MyGrid {
179///     fn n_dof(&self) -> usize { self.n }
180///     fn coordinates(&self, i: usize) -> &[f64] {
181///         // In practice, return a slice into a pre-computed node table.
182///         // This example uses a workaround for illustration only.
183///         std::slice::from_ref(Box::leak(Box::new(i as f64 * self.dx)))
184///     }
185///     fn spatial_dimension(&self) -> usize { 1 }
186///     fn characteristic_length(&self) -> f64 { self.dx }
187/// }
188/// ```
189///
190/// # INV-1 compliance
191///
192/// Implementations must **not** expose `dx`, `nx` or raw indices in any public
193/// method beyond those defined here. Spatial parameters are internal details.
194pub trait Mesh: Send + Sync {
195    /// Total number of degrees of freedom (nodes) in the mesh.
196    ///
197    /// For a 1D uniform grid: `n_dof() == n_points`.
198    /// For a 2D FEM mesh: `n_dof()` is the number of mesh nodes.
199    fn n_dof(&self) -> usize;
200
201    /// Coordinates of node `i` as a slice of length `spatial_dimension()`.
202    ///
203    /// Returns `&[f64]` — zero allocation. Callers must not store the returned
204    /// reference beyond the lifetime of the mesh. Implementors typically return
205    /// a slice into a pre-computed node coordinate table.
206    ///
207    /// # Panics
208    ///
209    /// Implementations may panic if `i >= n_dof()`. Use `n_dof()` to guard
210    /// iteration bounds.
211    fn coordinates(&self, i: usize) -> &[f64];
212
213    /// Number of spatial dimensions of this mesh.
214    ///
215    /// - `1` for 1D grids (chromatography column, 1D heat transfer)
216    /// - `2` for 2D meshes (surface flow, 2D diffusion)
217    /// - `3` for 3D meshes (volumetric FEM)
218    fn spatial_dimension(&self) -> usize;
219
220    /// A representative spatial length scale of the mesh.
221    ///
222    /// Used by integrators and operators for stability estimates (CFL condition,
223    /// Péclet number). For a uniform 1D grid: `dx`. For unstructured meshes:
224    /// the minimum element diameter or average edge length.
225    fn characteristic_length(&self) -> f64;
226
227    /// Nodes of the boundary `region` ([DA-071](https://github.com/biface/oxiflow/issues/203)).
228    ///
229    /// The default implementation covers the two faces of a 1D mesh, normal
230    /// to axis 0, assuming nodes ordered along the axis: `Lower` is node `0`,
231    /// `Upper` node `n_dof() − 1`. Meshes of higher dimension, or whose
232    /// nodes are not in that order, override it.
233    ///
234    /// # Errors
235    ///
236    /// `OxiflowError::PreconditionFailed` for a region the mesh does not
237    /// define (with the default: another axis, a mesh that is not 1D) or a
238    /// mesh without nodes.
239    fn boundary_nodes(&self, region: &BoundaryRegion) -> Result<Vec<usize>, OxiflowError> {
240        let (n, side) = default_face(self, region, 1, "Mesh::boundary_nodes")?;
241        Ok(match side {
242            FaceSide::Lower => vec![0],
243            FaceSide::Upper => vec![n - 1],
244        })
245    }
246
247    /// Nodes of the boundary `region`, each with its first neighbour along
248    /// the inward normal and their distance ([DA-071](https://github.com/biface/oxiflow/issues/203)) — what a
249    /// condition written on a structured grid needs, such as a one-sided
250    /// difference across the boundary.
251    ///
252    /// The default implementation covers the two faces of a 1D mesh, normal
253    /// to axis 0, assuming nodes ordered along the axis: `Lower` is node `0`
254    /// with neighbour `1`, `Upper` node `n_dof() − 1` with neighbour
255    /// `n_dof() − 2`. Only the order of that boundary pair is checked (its
256    /// distance must be positive), not the order of the whole mesh.
257    ///
258    /// Unstructured meshes have no such neighbour: they keep a refusal here,
259    /// a condition on them being imposed weakly instead.
260    ///
261    /// # Errors
262    ///
263    /// `OxiflowError::PreconditionFailed` for a region the mesh does not
264    /// define, a mesh with fewer than two nodes, or a boundary pair whose
265    /// distance is not positive (nodes not ordered along the axis).
266    fn boundary_neighbours(
267        &self,
268        region: &BoundaryRegion,
269    ) -> Result<Vec<BoundaryNode>, OxiflowError> {
270        let context = "Mesh::boundary_neighbours";
271        let (n, side) = default_face(self, region, 2, context)?;
272        let (node, inner) = match side {
273            FaceSide::Lower => (0, 1),
274            FaceSide::Upper => (n - 1, n - 2),
275        };
276        let distance = (self.coordinates(node)[0] - self.coordinates(inner)[0]).abs();
277        let ordered = match side {
278            FaceSide::Lower => self.coordinates(inner)[0] > self.coordinates(node)[0],
279            FaceSide::Upper => self.coordinates(node)[0] > self.coordinates(inner)[0],
280        };
281        if !ordered {
282            return Err(OxiflowError::PreconditionFailed {
283                context,
284                message: format!(
285                    "{region:?}: nodes {node} and {inner} are not ordered along axis 0 \
286                     (x = {} and {})",
287                    self.coordinates(node)[0],
288                    self.coordinates(inner)[0]
289                ),
290            });
291        }
292        Ok(vec![BoundaryNode::new(node, inner, distance)])
293    }
294}

298/// A mesh behind an [`Arc`](std::sync::Arc) is a mesh: one grid, built once,
299/// can then serve a component that holds an `Arc` (a flux scheme, a
300/// calculator) and a [`Domain`](crate::solver::Domain), which owns a
301/// `Box<dyn Mesh>`, without being copied or built twice.
302///
303/// Every method forwards to the shared mesh, including those with a default
304/// implementation, so that a mesh which overrides them keeps its own
305/// answers.
306///
307/// # Examples
308///
309/// ```rust
310/// use std::sync::Arc;
311/// use oxiflow::mesh::{Mesh, UniformGrid1D};
312///
313/// let grid = Arc::new(UniformGrid1D::new(5, 0.0, 1.0)?);
314/// let owned: Box<dyn Mesh> = Box::new(Arc::clone(&grid));
315/// assert_eq!(owned.n_dof(), grid.n_dof());
316/// assert_eq!(Arc::strong_count(&grid), 2);
317/// # Ok::<(), oxiflow::context::OxiflowError>(())
318/// ```
319impl<M: Mesh + ?Sized> Mesh for std::sync::Arc<M> {
320    fn n_dof(&self) -> usize {
321        (**self).n_dof()
322    }
323
324    fn coordinates(&self, i: usize) -> &[f64] {
325        (**self).coordinates(i)
326    }
327
328    fn spatial_dimension(&self) -> usize {
329        (**self).spatial_dimension()
330    }
331
332    fn characteristic_length(&self) -> f64 {
333        (**self).characteristic_length()
334    }
335
336    fn boundary_nodes(&self, region: &BoundaryRegion) -> Result<Vec<usize>, OxiflowError> {
337        (**self).boundary_nodes(region)
338    }
339
340    fn boundary_neighbours(
341        &self,
342        region: &BoundaryRegion,
343    ) -> Result<Vec<BoundaryNode>, OxiflowError> {
344        (**self).boundary_neighbours(region)
345    }
346}

348// ── Uniform spacing ───────────────────────────────────────────────────────────
349
350/// Relative tolerance of [`uniform_spacing`] on each node gap.
351///
352/// `UniformGrid1D` builds its nodes as `x_start + i·dx`, so a gap differs from
353/// `dx` by about `ε·|x| / dx` (ε the `f64` machine epsilon): below `1e-6` as
354/// long as `|x| / dx < 1e9`. A spacing defect of `1e-6` relative is also far
355/// below the truncation error of the stencils that rely on this check.
356pub(crate) const SPACING_TOLERANCE: f64 = 1e-6;
357
358/// Node spacing of a 1D mesh whose nodes are equally spaced in index order.
359///
360/// Returns `Some(characteristic_length())` when the mesh is one-dimensional,
361/// has at least two nodes, and every gap `x[i+1] − x[i]` lies within
362/// [`SPACING_TOLERANCE`] (relative) of `characteristic_length()`. Returns
363/// `None` otherwise, including when `characteristic_length()` is not a
364/// positive finite number.
365///
366/// The gaps are compared with `characteristic_length()`, not with their mean:
367/// callers use the returned value as the stencil step, so a mesh with uniform
368/// nodes but a different declared length must be rejected as well. A positive
369/// reference also rules out nodes in decreasing order.
370///
371/// Cost is `O(n_dof)`; callers on a hot path resolve it once and store the
372/// result, which is sound as long as a mesh does not change after
373/// construction.
374pub(crate) fn uniform_spacing(mesh: &dyn Mesh) -> Option<f64> {
375    let n = mesh.n_dof();
376    let step = mesh.characteristic_length();
377    if mesh.spatial_dimension() != 1 || n < 2 || !step.is_finite() || step <= 0.0 {
378        return None;
379    }
380    let bound = SPACING_TOLERANCE * step;
381    let uniform = (0..n - 1).all(|i| {
382        let gap = mesh.coordinates(i + 1)[0] - mesh.coordinates(i)[0];
383        (gap - step).abs() <= bound
384    });
385    uniform.then_some(step)
386}
