1//! # Module `solver::parallel`
2//!
3//! Shared Rayon-dispatch-threshold mechanism (DD-014, DD-048) — a small,
4//! reusable building block for every site in the crate that decides, per
5//! element, whether to run a computation sequentially or through Rayon.
6//! Extracted after the same field/builder/guard triplet was written
7//! twice, verbatim, on `FDGradientCalculator` and `FDLaplacianCalculator`
8//! (DD-014, #51) — see DD-048 for the full rationale (architecture
9//! principles P1-P5) behind this module's shape.
10//!
11//! ## What this is not
12//!
13//! Not a trait imposed on `ContextCalculator`/`DiscreteOperator`/
14//! `FluxDivergenceOperator` — DD-048 explicitly rejects that shape,
15//! mirroring the precedent already set twice (DD-012 amendment 2, DD-039
16//! option A): forcing an orthogonal capability onto every implementer of
17//! a shared trait, when only some of them benefit from it. Each site that
18//! wants Rayon dispatch holds a [`ParallelThreshold`] field and reads it
19//! itself — a plain data-holding type, not a behavioural contract.
20//!
21//! Entirely gated behind the `parallel` feature, at the module
22//! declaration itself (`solver::mod`) — mirrors `solver::sparse`/
23//! `solver::gpu`'s own gating, not merely the items inside it.
24
25use std::sync::atomic::{AtomicUsize, Ordering};
26
27/// Per-instance Rayon dispatch threshold.
28///
29/// Stores `0` internally to mean "no override" — `0` is otherwise
30/// forbidden by [`Self::set`], so it is free to reuse as a sentinel
31/// without an `Option` (which would block mutation through `&self`
32/// without extra indirection). [`Self::get`] resolves the sentinel to a
33/// caller-supplied default rather than materializing that default at
34/// construction — one authoritative default, consulted on demand, rather
35/// than copied into every instance (DD-048; closer to chrom-rs's single
36/// process-wide default than to a per-instance constant).
37///
38/// Deliberately per-instance, not a shared `static` (chrom-rs's own
39/// design) — `MultiDomainOrchestrator` (DD-031) can run several domains,
40/// each with its own solver, in the same process; a single global would
41/// force them to share one setting, with a "last writer wins" race
42/// between domains adjusting it independently. Reconfigurable at runtime
43/// through `&self` via [`Self::set`] — no rebuild required, and no
44/// cross-instance leakage to guard against (unlike chrom-rs's
45/// `ThresholdGuard`, needed only because its threshold is global).
46#[derive(Debug, Default)]
47pub(crate) struct ParallelThreshold(AtomicUsize);
48
49impl ParallelThreshold {
50    /// Creates a threshold with no override — [`Self::get`] reports the
51    /// caller-supplied default until [`Self::set`] is called.
52    pub(crate) fn unset() -> Self {
53        Self(AtomicUsize::new(0))
54    }
55
56    /// Overrides the threshold.
57    ///
58    /// # Panics
59    ///
60    /// Panics when `threshold == 0` — reserved as the "unset" sentinel
61    /// (see the type's own docs), and not a value any real workload
62    /// should want anyway: it would force parallel dispatch on every
63    /// non-empty input, never the intended behaviour (mirrors chrom-rs's
64    /// own `set_parallel_threshold` guard). No other value is validated
65    /// — DD-048 deliberately leaves "how low is too low" to the caller's
66    /// own measurement (see [`calibrate_parallel_threshold`]), consistent
67    /// with `sparse_threshold`/`jacobian_bandwidth` (DD-043), which
68    /// validate nothing at all: there is no threshold floor that is
69    /// universally sensible across workloads (chrom-rs's 999 is nearly a
70    /// total loss for the FD stencils DD-014 calibrated 49999 against).
71    pub(crate) fn set(&self, threshold: usize) {
72        assert!(threshold > 0, "parallel threshold must be at least 1");
73        self.0.store(threshold, Ordering::Relaxed);
74    }
75
76    /// Returns the configured override, or `default` if none was set.
77    pub(crate) fn get(&self, default: usize) -> usize {
78        match self.0.load(Ordering::Relaxed) {
79            0 => default,
80            configured => configured,
81        }
82    }
83}
84
85/// Empirically measures the sequential/parallel crossover point for a
86/// given per-element workload, on the machine actually running the code
87/// — rather than assuming a number calibrated for a different workload
88/// or a different machine (DD-048).
89///
90/// Tries each of `candidate_sizes`, in order, timing a sequential loop
91/// and a Rayon `into_par_iter` dispatch of `work` over `0..n` elements;
92/// returns the first `n` where the parallel run was faster. Returns the
93/// largest candidate if none crossed over (a conservative "parallel
94/// never won at any size tried" outcome, not a panic), or `0` if
95/// `candidate_sizes` is empty.
96///
97/// `work` should be representative of the real per-element cost it will
98/// be used to threshold for — measuring a cheap placeholder closure and
99/// applying the result to a heavier one (or vice versa) defeats the
100/// purpose. Feeds #53 (the Criterion benchmark suite) rather than
101/// duplicating its own crossover measurement.
102///
103/// # Examples
104///
105/// ```rust
106/// use oxiflow::solver::parallel::calibrate_parallel_threshold;
107///
108/// let threshold = calibrate_parallel_threshold(
109///     |i| {
110///         let _ = (i as f64).sqrt();
111///     },
112///     &[100, 1_000, 10_000, 100_000],
113/// );
114/// assert!(threshold > 0);
115/// ```
116pub fn calibrate_parallel_threshold<F>(work: F, candidate_sizes: &[usize]) -> usize
117where
118    F: Fn(usize) + Sync,
119{
120    use rayon::prelude::*;
121    use std::time::Instant;
122
123    for &n in candidate_sizes {
124        let seq_start = Instant::now();
125        for i in 0..n {
126            work(i);
127        }
128        let seq_elapsed = seq_start.elapsed();
129
130        let par_start = Instant::now();
131        (0..n).into_par_iter().for_each(&work);
132        let par_elapsed = par_start.elapsed();
133
134        if par_elapsed < seq_elapsed {
135            return n;
136        }
137    }
138
139    candidate_sizes.iter().copied().max().unwrap_or(0)
140}

144#[cfg(test)]
145mod tests {
146    use super::*;
147
148    // ── ParallelThreshold ──────────────────────────────────────────────────────
149
150    #[test]
151    fn unset_reports_the_supplied_default() {
152        let t = ParallelThreshold::unset();
153        assert_eq!(t.get(12345), 12345);
154    }
155
156    #[test]
157    fn set_overrides_the_default() {
158        let t = ParallelThreshold::unset();
159        t.set(42);
160        assert_eq!(t.get(12345), 42);
161    }
162
163    #[test]
164    #[should_panic(expected = "parallel threshold must be at least 1")]
165    fn set_rejects_zero() {
166        ParallelThreshold::unset().set(0);
167    }
168
169    #[test]
170    fn set_is_visible_through_shared_reference() {
171        let t = ParallelThreshold::unset();
172        let t_ref = &t;
173        t_ref.set(7);
174        assert_eq!(t.get(0), 7);
175    }
176
177    // ── calibrate_parallel_threshold ───────────────────────────────────────────
178    //
179    // No assertion depends on *which* candidate wins — that's a timing
180    // outcome, and asserting a specific crossover would make this test
181    // flaky under CI load. Only the contract (returns a candidate, or 0
182    // on empty input) is checked.
183
184    #[test]
185    fn calibrate_returns_one_of_the_candidates() {
186        let candidates = [10, 100, 1_000];
187        let threshold = calibrate_parallel_threshold(
188            |i| {
189                std::hint::black_box(i);
190            },
191            &candidates,
192        );
193        assert!(candidates.contains(&threshold));
194    }
195
196    #[test]
197    fn calibrate_on_empty_candidates_returns_zero() {
198        let threshold = calibrate_parallel_threshold(
199            |i: usize| {
200                std::hint::black_box(i);
201            },
202            &[],
203        );
204        assert_eq!(threshold, 0);
205    }
206}
