1//! # Module `solver::gpu`
2//!
3//! GPU architecture layer (DD-026, issue #73; DD-045, issue #127).
4//!
5//! ## Scope
6//!
7//! This module owns exactly two things, both HOW-side (never surfacing into
8//! `PhysicalModel`/`Domain`/`Scenario`, per DD-045):
9//!
10//! 1. **Adapter/device acquisition** ([`GpuContext::new`]) — blocking, via
11//!    `pollster`, with an automatic `HighPerformance` adapter preference and
12//!    an explicit [`OxiflowError::GpuUnavailable`] on failure (never a silent
13//!    CPU fallback — the caller decides).
14//! 2. **CPU↔GPU boundary conversion** ([`to_gpu_bytes`]) — reinterprets the
15//!    contiguous `f64` payloads already inside [`ContextValue`] (DD-026
16//!    INV-GPU-1) as raw bytes via `bytemuck::cast_slice`, without requiring
17//!    `ContextValue` itself to implement `bytemuck::Pod` (its enum
18//!    discriminant never can). Only the numeric payload is cast, not the
19//!    enum wrapper.
20//!
21//! ## What this module does *not* do
22//!
23//! No compute shader (WGSL), no buffer upload/dispatch, no `wgpu::Device`
24//! usage beyond acquisition. Actual GPU-accelerated computation is future
25//! work, not part of #74/DD-045 — this module is the boundary layer those
26//! future kernels will sit behind.
27//!
28//! ## Dispatch model (DD-045)
29//!
30//! No configurable runtime threshold (contrast with `parallel`/DD-014): the
31//! `gpu` feature flag is itself the opt-in. Once compiled with `--features
32//! gpu`, [`GpuContext::new`] is always the path taken — there is no
33//! CPU/GPU switch left to flip at runtime by this module.
34

40// ── GpuContext ──────────────────────────────────────────────────────────────
41
42/// Acquired GPU adapter, device and queue — the entry point of the `gpu`
43/// feature.
44///
45/// Construction is blocking ([`GpuContext::new`]), consistent with
46/// `Solver::solve()` already being a single blocking call (DD-045): no
47/// async runtime is introduced anywhere else in the crate.
48pub struct GpuContext {
49    /// Kept for adapter introspection (name, backend, limits) — not read by
50    /// this skeleton yet, reserved for the future dispatch layer.
51    #[allow(dead_code)]
52    adapter: wgpu::Adapter,
53    /// Logical GPU device — will own future compute pipelines/shaders.
54    pub device: wgpu::Device,
55    /// Command queue — will submit future compute dispatches.
56    pub queue: wgpu::Queue,
57}
58
59impl GpuContext {
60    /// Acquires a GPU adapter and device, blocking (DD-045).
61    ///
62    /// Requests a `HighPerformance` adapter (compute workload, not
63    /// interactive rendering — no `compatible_surface`). Returns
64    /// [`OxiflowError::GpuUnavailable`] if no compatible adapter or device
65    /// is found — never a silent CPU fallback; the caller chooses its own
66    /// CPU solver in that case.
67    ///
68    /// # Errors
69    ///
70    /// Returns [`OxiflowError::GpuUnavailable`] if adapter or device
71    /// acquisition fails.
72    pub fn new() -> Result<Self, OxiflowError> {
73        // `wgpu 26.0.1` API: `InstanceDescriptor` still implements `Default`
74        // at this version (removed in a later release, v29+) — explicit
75        // backends restrict to the set decided in DD-045 (vulkan/metal/dx12,
76        // no gles/webgpu), matching the `features` selected in `Cargo.toml`.
77        let instance = wgpu::Instance::new(&wgpu::InstanceDescriptor {
78            backends: wgpu::Backends::VULKAN | wgpu::Backends::METAL | wgpu::Backends::DX12,
79            ..Default::default()
80        });
81
82        let adapter = pollster::block_on(instance.request_adapter(&wgpu::RequestAdapterOptions {
83            power_preference: wgpu::PowerPreference::HighPerformance,
84            force_fallback_adapter: false,
85            compatible_surface: None,
86        }))
87        .map_err(|_| OxiflowError::GpuUnavailable)?;
88
89        let (device, queue) = pollster::block_on(adapter.request_device(&wgpu::DeviceDescriptor {
90            label: Some("oxiflow-gpu-device"),
91            required_features: wgpu::Features::empty(),
92            required_limits: wgpu::Limits::default(),
93            memory_hints: wgpu::MemoryHints::default(),
94            trace: wgpu::Trace::Off,
95        }))
96        .map_err(|_| OxiflowError::GpuUnavailable)?;
97
98        Ok(Self {
99            adapter,
100            device,
101            queue,
102        })
103    }
104}

106// ── CPU↔GPU boundary conversion ──────────────────────────────────────────────
107
108/// Reinterprets a numeric [`ContextValue`] payload as raw bytes, ready for
109/// GPU buffer upload (DD-026 INV-GPU-5).
110///
111/// Only the contiguous `f64` payload is cast (`bytemuck::cast_slice`) — never
112/// `ContextValue` itself, whose enum discriminant is not `Pod`-compatible.
113/// `Boolean` is not a numeric payload and has no meaningful GPU-upload
114/// representation here; it returns [`OxiflowError::TypeMismatch`].
115///
116/// # Errors
117///
118/// Returns [`OxiflowError::TypeMismatch`] for [`ContextValue::Boolean`].
119///
120/// # Examples
121///
122/// ```rust
123/// # #[cfg(feature = "gpu")]
124/// # {
125/// use oxiflow::context::value::ContextValue;
126/// use oxiflow::solver::gpu::to_gpu_bytes;
127/// use nalgebra::DVector;
128///
129/// let field = ContextValue::ScalarField(DVector::from_vec(vec![1.0, 2.0, 3.0]));
130/// let bytes = to_gpu_bytes(&field).unwrap();
131/// assert_eq!(bytes.len(), 3 * std::mem::size_of::<f64>());
132/// # }
133/// ```
134pub fn to_gpu_bytes(value: &ContextValue) -> Result<&[u8], OxiflowError> {
135    match value {
136        ContextValue::Scalar(s) => Ok(bytemuck::bytes_of(s)),
137        ContextValue::Vector(v) | ContextValue::ScalarField(v) => Ok(cast_slice(v.as_slice())),
138        ContextValue::Matrix(m) | ContextValue::VectorField(m) => Ok(cast_slice(m.as_slice())),
139        ContextValue::Boolean(_) => Err(OxiflowError::TypeMismatch {
140            expected: "numeric ContextValue (Scalar/Vector/Matrix/ScalarField/VectorField)",
141            actual: "Boolean",
142        }),
143        // `#[non_exhaustive]` on ContextValue: reserved J7 variants
144        // (Tensor4/TensorField) are not yet defined in the enum itself, so
145        // this arm is unreachable today, not a gap.
146        #[allow(unreachable_patterns)]
147        _ => Err(OxiflowError::TypeMismatch {
148            expected: "numeric ContextValue (Scalar/Vector/Matrix/ScalarField/VectorField)",
149            actual: value.variant_name(),
150        }),
151    }
152}

154#[cfg(test)]
155mod tests {
156    use super::*;
157    use nalgebra::{DMatrix, DVector};
158
159    #[test]
160    fn to_gpu_bytes_scalar_field() {
161        let field = ContextValue::ScalarField(DVector::from_vec(vec![1.0, 2.0, 3.0]));
162        let bytes = to_gpu_bytes(&field).unwrap();
163        assert_eq!(bytes.len(), 3 * std::mem::size_of::<f64>());
164    }
165
166    #[test]
167    fn to_gpu_bytes_vector_field() {
168        let field = ContextValue::VectorField(DMatrix::from_element(2, 3, 1.5));
169        let bytes = to_gpu_bytes(&field).unwrap();
170        assert_eq!(bytes.len(), 6 * std::mem::size_of::<f64>());
171    }
172
173    #[test]
174    fn to_gpu_bytes_scalar() {
175        let value = ContextValue::Scalar(42.0);
176        let bytes = to_gpu_bytes(&value).unwrap();
177        assert_eq!(bytes.len(), std::mem::size_of::<f64>());
178    }
179
180    #[test]
181    fn to_gpu_bytes_rejects_boolean() {
182        let value = ContextValue::Boolean(true);
183        let err = to_gpu_bytes(&value).unwrap_err();
184        assert!(matches!(err, OxiflowError::TypeMismatch { .. }));
185    }
186
187    /// No GPU hardware in CI (documented limitation, DD-045 consequences) —
188    /// `#[ignore]`, run manually on a machine with a real adapter.
189    #[test]
190    #[ignore = "requires real GPU hardware, not available in CI"]
191    fn gpu_context_acquires_adapter() {
192        GpuContext::new().unwrap();
193    }
