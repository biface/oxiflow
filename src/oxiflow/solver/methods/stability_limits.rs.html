1//! # Module `solver::methods::stability_limits`
2//!
3//! Step limits that stability puts on the explicit integrators
4//! ([DA-073](https://github.com/biface/oxiflow/issues/229)).
5//!
6//! ## Advective limit
7//!
8//! An explicit integrator advancing an advective term `∇·F(u)` is stable
9//! only if information does not travel more than one cell per step:
10//!
11//! $$C = \frac{s \cdot \Delta t}{\Delta x} \leq 1$$
12//!
13//! where `s` is the largest absolute wave speed the model declares through
14//! [`PhysicalModel::max_wave_speed`](crate::model::traits::PhysicalModel::max_wave_speed),
15//! `Δx` the characteristic length of the mesh
16//! ([`Mesh::characteristic_length`](crate::mesh::Mesh::characteristic_length))
17//! and `C` the Courant number. The limit is therefore
18//! `dt_max = Δx / s`. It is a *necessary* condition, not a sufficient one:
19//! forward Euler with a centred flux is unstable at any step in pure
20//! advection, and RK4 with an upwind flux would tolerate a Courant number of
21//! about 1.4. The limit of 1 holds for every explicit integrator of the
22//! crate; a configurable limit, or one per scheme–integrator pair, would
23//! come with a model that needs it.
24//!
25//! The limit belongs to the integrator, not to the spatial scheme: the
26//! scheme sees neither the integrator nor its step control. Forward Euler
27//! and RK4, with a fixed step, refuse a step above the limit
28//! ([`require_within_advective_limit`]); DoPri45 shortens its step to it.
29//! The implicit integrators (Backward Euler, Crank–Nicolson, BDF2) have no
30//! advective limit and do not check.
31//!
32//! The speed is evaluated on the state at the start of each step, in the
33//! context of the first stage, so the calculator chain runs once
34//! (`evaluate_derivative_and_step_limit` in [`super`]).
35//!
36//! ## Out of scope
37//!
38//! The diffusive (Fourier) condition `D·Δt/Δx² ≤ 1/2` is not checked. Per-axis
39//! speeds and lengths come with the nD structured grid
40//! ([#152](https://github.com/biface/oxiflow/issues/152)).
41
42use crate::context::error::OxiflowError;
43use crate::solver::scenario::Domain;
44
45/// Advective step limit `dt_max = Δx / s` of `domain` for the wave speed
46/// `speed` its model declared, or `None` if the model declared no speed or
47/// a zero speed (nothing is transported, no limit).
48///
49/// # Errors
50///
51/// `OxiflowError::PreconditionFailed` (context
52/// `"PhysicalModel::max_wave_speed"`) if `speed` is NaN, infinite or
53/// negative: such a speed cannot bound a step, and passing it on would let
54/// a non-finite state through unchecked. The message names the model.
55pub(crate) fn advective_step_limit(
56    domain: &Domain,
57    speed: Option<f64>,
58) -> Result<Option<f64>, OxiflowError> {
59    let Some(speed) = speed else {
60        return Ok(None);
61    };
62    if !speed.is_finite() || speed < 0.0 {
63        return Err(OxiflowError::PreconditionFailed {
64            context: "PhysicalModel::max_wave_speed",
65            message: format!(
66                "model '{}' declared the wave speed {speed}; a wave speed must be \
67                 finite and non-negative",
68                domain.model.name()
69            ),
70        });
71    }
72    if speed == 0.0 {
73        return Ok(None);
74    }
75    Ok(Some(domain.mesh.characteristic_length() / speed))
76}
77
78/// Refuses a fixed step `dt` above the advective step limit `limit`, for
79/// the explicit integrator named `context`. No limit, no check.
80///
81/// # Errors
82///
83/// `OxiflowError::PreconditionFailed` with the given `context` if
84/// `dt > limit`; the message gives the Courant number `dt / limit`, the
85/// limit and the step.
86pub(crate) fn require_within_advective_limit(
87    context: &'static str,
88    dt: f64,
89    limit: Option<f64>,
90) -> Result<(), OxiflowError> {
91    match limit {
92        Some(dt_max) if dt > dt_max => Err(OxiflowError::PreconditionFailed {
93            context,
94            message: format!(
95                "advective stability limit exceeded: Courant number {:.6} > 1 \
96                 (dt = {dt}, dt_max = {dt_max}); an explicit integrator needs \
97                 dt <= dt_max, an implicit one has no such limit",
98                dt / dt_max
99            ),
100        }),
101        _ => Ok(()),
102    }
103}

107#[cfg(test)]
108mod tests {
109    use super::*;
110    use crate::mesh::UniformGrid1D;
111    use crate::model::DeclaredWaveSpeed;
112
113    /// 11 nodes on (0, 1): `Δx = 0.1`.
114    fn domain(speed: f64) -> Domain {
115        Domain::new(
116            "d",
117            Box::new(DeclaredWaveSpeed { speed }),
118            Box::new(UniformGrid1D::new(11, 0.0, 1.0).unwrap()),
119        )
120    }
121
122    #[test]
123    fn limit_is_the_characteristic_length_over_the_speed() {
124        let limit = advective_step_limit(&domain(4.0), Some(4.0))
125            .unwrap()
126            .unwrap();
127        assert!((limit - 0.025).abs() < 1e-15);
128    }
129
130    #[test]
131    fn no_speed_or_zero_speed_gives_no_limit() {
132        assert_eq!(advective_step_limit(&domain(0.0), None).unwrap(), None);
133        assert_eq!(advective_step_limit(&domain(0.0), Some(0.0)).unwrap(), None);
134    }
135
136    #[test]
137    fn invalid_speeds_are_refused_with_the_model_name() {
138        for speed in [f64::NAN, f64::INFINITY, f64::NEG_INFINITY, -1.0] {
139            let err = advective_step_limit(&domain(speed), Some(speed)).unwrap_err();
140            match err {
141                OxiflowError::PreconditionFailed { context, message } => {
142                    assert_eq!(context, "PhysicalModel::max_wave_speed");
143                    assert!(message.contains("declared_wave_speed"), "{message}");
144                }
145                other => panic!("unexpected error {other:?}"),
146            }
147        }
148    }
149
150    #[test]
151    fn a_step_above_the_limit_is_refused_with_its_courant_number() {
152        let err = require_within_advective_limit("Solver::step", 0.2, Some(0.1)).unwrap_err();
153        match err {
154            OxiflowError::PreconditionFailed { context, message } => {
155                assert_eq!(context, "Solver::step");
156                assert!(message.contains("Courant number 2.000000"), "{message}");
157                assert!(message.contains("dt_max = 0.1"), "{message}");
158            }
159            other => panic!("unexpected error {other:?}"),
160        }
161    }
162
163    #[test]
164    fn a_step_at_or_below_the_limit_or_without_limit_passes() {
165        assert!(require_within_advective_limit("Solver::step", 0.1, Some(0.1)).is_ok());
166        assert!(require_within_advective_limit("Solver::step", 0.05, Some(0.1)).is_ok());
167        assert!(require_within_advective_limit("Solver::step", 1e9, None).is_ok());
168    }
169}
