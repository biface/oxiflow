1//! # Module `solver::methods::backward_euler`
2//!
3//! Backward Euler integrator — implicit, 1st order (issue #43).
4//!
5//! ## Algorithm
6//!
7//! $$u^{n+1} = u^n + \Delta t \cdot f(u^{n+1}, t^{n+1})$$
8//!
9//! A thin wrapper around the shared generalised theta method
10//! ([`super::implicit::theta_method_step`], θ=1) — see that module's docs
11//! for the frozen-Jacobian v1 scope and the path to a future iterated
12//! Newton solver (DD-033).
13//!
14//! ## Stability
15//!
16//! Unconditionally A-stable for `f` affine in `u` — no CFL-style
17//! restriction on `dt`, unlike the explicit methods. This is the whole
18//! point: stiff problems (`λΔt ≫ 1`) that would blow up under
19//! `ForwardEulerSolver` remain bounded here.
20//!
21//! ## Scope at J4a
22//!
23//! - Single-domain scenarios only — same restriction as the explicit
24//!   solvers; see #40 for the dedicated multi-domain path.
25//! - `StepControl::Fixed { dt }` only.
26//! - See [`super::implicit`] for the boundary-condition interaction
27//!   caveat — not yet covered by a dedicated test.
28

42/// Backward Euler solver — implicit, 1st order.
43///
44/// # Examples
45///
46/// ```rust,ignore
47/// use oxiflow::solver::methods::backward_euler::BackwardEulerSolver;
48///
49/// let solver = BackwardEulerSolver::new();
50/// // Sparse backend (v0.6.0, DD-013 second phase, DD-043) — dense path
51/// // stays untouched unless all three are configured explicitly:
52/// // let solver = BackwardEulerSolver::new()
53/// //     .with_jacobian_bandwidth(scheme.stencil_radius())
54/// //     .with_sparse_solver(Box::new(FaerSparseSolver))
55/// //     .with_sparse_threshold(100); // default
56/// ```
57pub struct BackwardEulerSolver {
58    linear_solver: Box<dyn LinearSolver>,
59    /// Sparse backend (DD-043) — `None` by default: the dense path
60    /// ([`theta_method_step`]) is unconditionally used unless this is
61    /// explicitly configured via [`Self::with_sparse_solver`].
62    #[cfg(feature = "sparse")]
63    sparse_solver: Option<Box<dyn SparseLinearSolver>>,
64    /// System size above which the sparse path is used, once a
65    /// `sparse_solver` and a `jacobian_bandwidth` are both configured.
66    /// Ignored (no effect) if `sparse_solver` is `None`.
67    #[cfg(feature = "sparse")]
68    sparse_threshold: usize,
69    /// Half-bandwidth of the Jacobian — a HOW-side assertion supplied by
70    /// the caller (e.g. `scheme.stencil_radius()`, DD-039), never read
71    /// from `Domain`/`PhysicalModel` (DD-043, amendment 1). `None` by
72    /// default: no assumption about locality, dense path only.
73    #[cfg(feature = "sparse")]
74    jacobian_bandwidth: Option<usize>,
75}
76
77impl Default for BackwardEulerSolver {
78    fn default() -> Self {
79        Self {
80            linear_solver: Box::new(NalgebraDenseSolver),
81            #[cfg(feature = "sparse")]
82            sparse_solver: None,
83            #[cfg(feature = "sparse")]
84            sparse_threshold: 100,
85            #[cfg(feature = "sparse")]
86            jacobian_bandwidth: None,
87        }
88    }
89}

91impl BackwardEulerSolver {
92    /// Creates a solver using the default `nalgebra` dense backend.
93    pub fn new() -> Self {
94        Self::default()
95    }
96
97    /// Substitutes the linear solver backend (DD-013).
98    pub fn with_linear_solver(mut self, linear_solver: Box<dyn LinearSolver>) -> Self {
99        self.linear_solver = linear_solver;
100        self
101    }
102
103    /// Configures the sparse backend (DD-043). Has no effect on its own —
104    /// the sparse path also requires [`Self::with_jacobian_bandwidth`] and
105    /// a system larger than [`Self::with_sparse_threshold`] (default 100)
106    /// at solve time; otherwise the dense path is used unchanged.
107    #[cfg(feature = "sparse")]
108    pub fn with_sparse_solver(mut self, sparse_solver: Box<dyn SparseLinearSolver>) -> Self {
109        self.sparse_solver = Some(sparse_solver);
110        self
111    }
112
113    /// System size above which the sparse path is used (default 100) —
114    /// see [`Self::with_sparse_solver`].
115    #[cfg(feature = "sparse")]
116    pub fn with_sparse_threshold(mut self, sparse_threshold: usize) -> Self {
117        self.sparse_threshold = sparse_threshold;
118        self
119    }
120
121    /// Declares the Jacobian's half-bandwidth (e.g.
122    /// `scheme.stencil_radius()`, DD-039) — see [`Self::with_sparse_solver`].
123    #[cfg(feature = "sparse")]
124    pub fn with_jacobian_bandwidth(mut self, jacobian_bandwidth: usize) -> Self {
125        self.jacobian_bandwidth = Some(jacobian_bandwidth);
126        self
127    }
128}

132        &self,
133        scenario: &Scenario,
134        config: &SolverConfiguration,
135    ) -> Result<SimulationResult, OxiflowError> {

140#[cfg(feature = "sparse")]
141impl SteppableSolver for BackwardEulerSolver {
142    fn step(
143        &self,
144        domain: &Domain,
145        chain: &[&dyn ContextCalculator],
146        state: &mut ContextValue,
147        _history: &[ContextValue],
148        t: f64,
149        dt: f64,
150    ) -> Result<ContextValue, OxiflowError> {
151        // history_depth() defaults to 0 -- Backward Euler is a one-step
152        // method, `_history` is always empty here and intentionally unused.
153        //
154        // Dispatch is gated on `sparse_solver` alone: with no solver
155        // configured there is nothing to hand `theta_method_step_adaptive`,
156        // so the dense path is taken directly rather than going through it
157        // with a `None` bandwidth (equivalent result, one fewer branch to
158        // reason about). The `n > sparse_threshold && bandwidth.is_some()`
159        // condition is `theta_method_step_adaptive`'s own responsibility.
160        match &self.sparse_solver {
161            Some(sparse_solver) => theta_method_step_adaptive(
162                domain,
163                chain,
164                state,
165                t,
166                dt,
167                1.0,
168                self.linear_solver.as_ref(),
169                sparse_solver.as_ref(),
170                self.sparse_threshold,
171                self.jacobian_bandwidth,
172            ),
173            None => theta_method_step(
174                domain,
175                chain,
176                state,
177                t,
178                dt,
179                1.0,
180                self.linear_solver.as_ref(),
181            ),
182        }
183    }
184}

204            dt,

206            self.linear_solver.as_ref(),
207        )
208    }

285    /// Delegates to `NalgebraDenseSolver` but counts calls.
286    #[derive(Debug)]
287    struct CountingLinearSolver {
288        calls: Arc<AtomicUsize>,
289    }
290
291    impl LinearSolver for CountingLinearSolver {
292        fn solve(&self, a: &DMatrix<f64>, b: &DVector<f64>) -> Result<DVector<f64>, OxiflowError> {
293            self.calls.fetch_add(1, Ordering::SeqCst);
294            NalgebraDenseSolver.solve(a, b)
295        }
296    }

377    #[test]
378    fn step_matches_one_iteration_of_solve() {
379        let scenario = Scenario::single(Box::new(ExponentialDecay { lambda: 0.7 }), make_mesh(3));
380        let config = make_config(0.1, 0.1);
381
382        let solver = BackwardEulerSolver::new();
383        let via_solve = solver.solve(&scenario, &config).unwrap();
384        let final_via_solve = via_solve.states.last().unwrap().as_scalar_field().unwrap();
385
386        let domain = scenario.single_domain().unwrap();
387        let requirements = scenario.context_requirements();
388        let chain =
389            crate::solver::chain::build_calculator_chain(&requirements, &config.calculators)
390                .unwrap();
391        let mut u = domain.model.initial_state(domain.mesh.as_ref());
392        let next = solver.step(domain, &chain, &mut u, &[], 0.0, 0.1).unwrap();
393        let final_via_step = next.as_scalar_field().unwrap();
394
395        assert_eq!(final_via_solve.len(), final_via_step.len());
396        for i in 0..final_via_solve.len() {
397            assert!((final_via_solve[i] - final_via_step[i]).abs() < 1e-15);
398        }
399    }
