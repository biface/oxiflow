1//! # Module `solver::methods::newton`
2//!
3//! Iterated Newton solver — shared, integrator-agnostic core (DD-044,
4//! issue #109, activating DD-033's Option A).
5//!
6//! ## Why this exists
7//!
8//! [`super::implicit::theta_method_step`] and [`super::bdf2::bdf2_step`]
9//! (through v0.6.0) each performed exactly **one** Newton-style
10//! correction with a frozen Jacobian — exact when the underlying
11//! `compute_physics` is affine in `u`, a first-order approximation
12//! otherwise. This module provides the genuinely iterated alternative
13//! DD-033 always anticipated, shared by both call sites rather than
14//! duplicated: theta-method and BDF2 (this sprint's two concrete
15//! consumers) differ only in how they express their residual as an
16//! affine transform of `u` plus an implicit term — see
17//! [the residual convention](#residual-convention) below.
18//!
19//! ## Residual convention
20//!
21//! Both consumers' implicit equations reduce to the same shape:
22//!
23//! $$g(u) = \alpha (u - u^{*}) - c \cdot f(u)$$
24//!
25//! - **theta-method** (Backward Euler `θ=1`, Crank-Nicolson `θ=0.5`):
26//!   $\alpha = 1$, $c = \theta \Delta t$, $u^{*} = u^n + \Delta t(1-\theta)
27//!   f(u^n)$.
28//! - **BDF2**: $\alpha = 3/2$, $c = \Delta t$, $u^{*} = \frac{4u^n -
29//!   u^{n-1}}{3}$.
30//!
31//! `u^{*}` folds in every term that doesn't depend on the unknown `u` —
32//! callers compute it once, outside the loop, from already-known past
33//! states. [`residual`] and [`linear_correction`] are the two building
34//! blocks operating on this shape; [`solve`] is the generic loop wiring
35//! them together with a [`NewtonConvergence`] criterion and a
36//! [`JacobianStrategy`].
37//!
38//! ## Dense/sparse agnosticism
39//!
40//! This module knows nothing about `Domain`, `ContextCalculator`, or
41//! `faer` — [`solve`] is parameterised over caller-supplied closures for
42//! evaluating $f(u)$ and its Jacobian, and over `&dyn LinearSolver` for
43//! the correction step. The existing sparse/banded dispatch
44//! (DD-043, `theta_method_step_adaptive`) is therefore untouched by this
45//! module: composing the two is [`super::implicit`]'s concern (chantier
46//! 2), not this one's.
47//!
48//! ## Wiring status
49//!
50//! [`solve`] is called from
51//! [`super::implicit::theta_method_step_newton`] (DD-044, #111) and
52//! [`super::bdf2::bdf2_step_newton`] (DD-044, #112) — used respectively by
53//! [`BackwardEulerSolver`](super::backward_euler::BackwardEulerSolver)/
54//! [`CrankNicolsonSolver`](super::crank_nicolson::CrankNicolsonSolver) and
55//! [`BDF2Solver`](super::bdf2::BDF2Solver). [`super::bdf2`]'s
56//! startup/bootstrap step is the one exception, unconditionally using
57//! [`super::implicit::theta_method_step`]'s non-configurable path — see
58//! that module's docs.

67/// Convergence criterion for the iterated Newton loop ([`solve`]).
68///
69/// Every variant carries the same `tol_abs`/`tol_rel` pair: convergence
70/// is tested against `tol_abs + tol_rel * ||u||`, evaluated on the
71/// iterate *after* applying the current correction (so that, e.g.,
72/// `max_iterations = 1` with an exactly affine `f` succeeds immediately —
73/// the single correction is already exact up to floating-point rounding).
74///
75/// Default: [`ResidualOnly`](Self::ResidualOnly) with `tol_abs = 1e-8`,
76/// `tol_rel = 1e-6` — starting points, not fixed constants; tune via the
77/// solver builders once this lands (#111/#112).
78///
79/// Note: this default is *not* what
80/// [`theta_method_step`](super::implicit::theta_method_step) or the
81/// implicit solvers' own zero-config constructors use internally — those
82/// need a looser, explicitly named tolerance to guarantee the pre-DD-044
83/// single-correction contract on very stiff affine problems (see
84/// [`super::implicit`]'s module docs for why). This `Default` impl is a
85/// general starting point for callers configuring the loop themselves,
86/// not a guarantee that it tolerates every finite-difference precision
87/// floor.
88#[non_exhaustive]
89#[derive(Debug, Clone, Copy, PartialEq)]
90#[cfg_attr(feature = "serde", derive(serde::Deserialize))]
91pub enum NewtonConvergence {
92    /// Converged once `||g(u)|| < tol_abs + tol_rel * ||u||`.
93    ///
94    /// The residual is available at zero extra cost — `f(u)` is already
95    /// evaluated to build the next correction — so this is the cheapest
96    /// criterion, not merely the default for convenience.
97    ResidualOnly { tol_abs: f64, tol_rel: f64 },
98    /// Converged once **either** the residual **or** the correction norm
99    /// (`||δu||`) satisfies the tolerance.
100    CombinedOr { tol_abs: f64, tol_rel: f64 },
101    /// Converged only once **both** the residual **and** the correction
102    /// norm satisfy the tolerance — the strictest of the three.
103    CombinedAnd { tol_abs: f64, tol_rel: f64 },
104}
105
106impl Default for NewtonConvergence {
107    fn default() -> Self {
108        NewtonConvergence::ResidualOnly {
109            tol_abs: 1e-8,
110            tol_rel: 1e-6,
111        }
112    }
113}
114
115impl NewtonConvergence {
116    /// Extracts the common `(tol_abs, tol_rel)` pair, regardless of
117    /// variant.
118    fn tolerances(&self) -> (f64, f64) {
119        match *self {
120            NewtonConvergence::ResidualOnly { tol_abs, tol_rel }
121            | NewtonConvergence::CombinedOr { tol_abs, tol_rel }
122            | NewtonConvergence::CombinedAnd { tol_abs, tol_rel } => (tol_abs, tol_rel),
123        }
124    }
125}

127/// Jacobian refresh strategy for the iterated Newton loop ([`solve`]).
128///
129/// Default: [`ModifiedFrozen`](Self::ModifiedFrozen) — matches the
130/// pre-DD-044 behaviour (Jacobian evaluated once, at the initial guess,
131/// never refreshed) as a special case of the generic loop, rather than a
132/// separate code path.
133#[non_exhaustive]
134#[derive(Debug, Clone, Copy, PartialEq, Default)]
135#[cfg_attr(feature = "serde", derive(serde::Deserialize))]
136pub enum JacobianStrategy {
137    /// Evaluate the Jacobian once, at the initial guess, and reuse it for
138    /// every correction (modified Newton).
139    #[default]
140    ModifiedFrozen,
141    /// Re-evaluate the Jacobian at every iterate (full Newton) — most
142    /// robust on strongly nonlinear problems, at the cost of one
143    /// [`finite_difference_jacobian`](super::implicit::finite_difference_jacobian)
144    /// (or its banded equivalent) per iteration.
145    FullNewton,
146    /// Reuse the frozen Jacobian while convergence is proceeding
147    /// acceptably; re-evaluate it once the residual stops shrinking fast
148    /// enough. Refreshed whenever
149    /// `residual_norm_k / residual_norm_{k-1} > stagnation_ratio` — an
150    /// explicit extension point for J8 (v0.8.0) cost/robustness tuning,
151    /// not yet auto-tuned.
152    ModifiedWithStagnationCheck { stagnation_ratio: f64 },
153}

157/// Evaluates the residual $g(u) = \alpha (u - u^{*}) - c \cdot f(u)$ — see
158/// the [module docs](self#residual-convention) for what `alpha`/`coeff`/
159/// `u_star` mean for each of this loop's two consumers.
160///
161/// `f_u` is `f(u)`, already evaluated by the caller (it is also needed
162/// to decide, e.g., whether to refresh the Jacobian — evaluating it here
163/// again would be redundant).
164pub(crate) fn residual(
165    u: &DVector<f64>,
166    u_star: &DVector<f64>,
167    alpha: f64,
168    coeff: f64,
169    f_u: &DVector<f64>,
170) -> DVector<f64> {
171    let delta: DVector<f64> = u - u_star;
172    delta * alpha - f_u.clone() * coeff
173}
174
175/// Solves the linearised correction $[\alpha I - c J_f] \delta u = -g(u)$
176/// for $\delta u$, given an already-assembled Jacobian $J_f =
177/// \partial f/\partial u$.
178///
179/// Resolution only — `jacobian` is supplied by the caller (typically
180/// [`finite_difference_jacobian`](super::implicit::finite_difference_jacobian),
181/// unchanged) rather than computed here, so this function stays
182/// dense/sparse-agnostic in principle; only the dense path
183/// (`&dyn LinearSolver`) is wired at this increment (#110/#111) —
184/// banded/sparse composition remains a known gap, see
185/// [`super::implicit`]'s module docs.
186pub(crate) fn linear_correction(
187    jacobian: &DMatrix<f64>,
188    alpha: f64,
189    coeff: f64,
190    residual: &DVector<f64>,
191    linear_solver: &dyn LinearSolver,
192) -> Result<DVector<f64>, OxiflowError> {
193    let n = jacobian.nrows();
194    let identity = DMatrix::<f64>::identity(n, n);
195    let system_matrix = identity * alpha - jacobian.clone() * coeff;
196    let rhs: DVector<f64> = -residual.clone();
197    linear_solver.solve(&system_matrix, &rhs)
198}

202/// Parameters governing the iterated Newton loop ([`solve`]), gathered
203/// into one struct so the loop's own argument list stays manageable
204/// alongside the residual-shape parameters (`u_star`/`alpha`/`coeff`)
205/// and the two evaluation closures.
206#[derive(Debug, Clone, Copy)]
207pub(crate) struct NewtonParams {
208    pub convergence: NewtonConvergence,
209    pub jacobian_strategy: JacobianStrategy,
210    pub max_iterations: usize,
211}
212
213/// Runs the iterated Newton loop for $g(u) = \alpha(u - u^{*}) - c f(u) =
214/// 0$, starting from `u0`.
215///
216/// `f_eval` evaluates $f(u)$ for a candidate iterate (in production,
217/// this wraps [`evaluate_derivative`](super::evaluate_derivative) —
218/// including its boundary-condition re-application, see
219/// [`super::implicit`]'s known-limitation note); `jac_eval` evaluates
220/// $\partial f/\partial u$ at a candidate iterate (wraps
221/// [`finite_difference_jacobian`](super::implicit::finite_difference_jacobian)
222/// or its banded equivalent). Both are called at most `max_iterations + 1`
223/// times each.
224///
225/// Convergence is checked **after** applying each correction, uniformly
226/// regardless of `max_iterations` — there is no special case for
227/// `max_iterations == 1`. This is deliberate: whether `f` is affine in
228/// `u` is not tracked or asserted anywhere (doing so would require the
229/// HOW pole to know something about the WHAT pole's `PhysicalModel`,
230/// which DD-044 does not introduce) — it is instead *inferred for free*
231/// from the residual itself. For an affine `f`, a single correction is
232/// exact, so the post-application residual already sits at
233/// floating-point precision and passes any reasonable tolerance; for a
234/// genuinely nonlinear `f`, it generally does not, and
235/// `max_iterations == 1` then correctly reports
236/// [`OxiflowError::NewtonNotConverged`] instead of silently returning a
237/// first-order-accurate value.
238///
239/// Consequence for callers migrating from the pre-DD-044
240/// `theta_method_step`/`bdf2_step` (which never checked convergence, and
241/// so could never fail this way): calling this loop with the historical
242/// defaults (`max_iterations = 1`, `ModifiedFrozen`) reproduces the exact
243/// same numeric result on every existing (affine) regression case, but
244/// — unlike before — now surfaces an explicit error rather than a silent
245/// first-order approximation on a genuinely nonlinear model. Widen
246/// `max_iterations` (or the tolerance) for such models rather than
247/// relying on the single-correction default.
248///
249/// # Errors
250///
251/// Returns [`OxiflowError::NewtonNotConverged`] if `max_iterations` is
252/// exhausted without satisfying `params.convergence`. Propagates any
253/// error from `f_eval`, `jac_eval`, or the linear solve unchanged.
254#[allow(clippy::too_many_arguments)]
255pub(crate) fn solve(
256    u0: DVector<f64>,
257    u_star: &DVector<f64>,
258    alpha: f64,
259    coeff: f64,
260    mut f_eval: impl FnMut(&DVector<f64>) -> Result<DVector<f64>, OxiflowError>,
261    mut jac_eval: impl FnMut(&DVector<f64>) -> Result<DMatrix<f64>, OxiflowError>,
262    linear_solver: &dyn LinearSolver,
263    params: &NewtonParams,
264) -> Result<DVector<f64>, OxiflowError> {
265    let (tol_abs, tol_rel) = params.convergence.tolerances();
266
267    let mut u = u0;
268    let mut jacobian = jac_eval(&u)?;
269    let mut f_u = f_eval(&u)?;
270    let mut prev_residual_norm: Option<f64> = None;
271    let mut last_residual_norm = f64::INFINITY;
272
273    for _ in 0..params.max_iterations {
274        let res = residual(&u, u_star, alpha, coeff, &f_u);
275        let correction = linear_correction(&jacobian, alpha, coeff, &res, linear_solver)?;
276        let correction_norm = correction.norm();
277        u += correction;
278
279        f_u = f_eval(&u)?;
280        let res_new = residual(&u, u_star, alpha, coeff, &f_u);
281        let res_new_norm = res_new.norm();
282        last_residual_norm = res_new_norm;
283
284        let tol = tol_abs + tol_rel * u.norm();
285        let residual_ok = res_new_norm < tol;
286        let correction_ok = correction_norm < tol;
287        let converged = match params.convergence {
288            NewtonConvergence::ResidualOnly { .. } => residual_ok,
289            NewtonConvergence::CombinedOr { .. } => residual_ok || correction_ok,
290            NewtonConvergence::CombinedAnd { .. } => residual_ok && correction_ok,
291        };
292        if converged {
293            return Ok(u);
294        }
295
296        match params.jacobian_strategy {
297            JacobianStrategy::ModifiedFrozen => {}
298            JacobianStrategy::FullNewton => {
299                jacobian = jac_eval(&u)?;
300            }
301            JacobianStrategy::ModifiedWithStagnationCheck { stagnation_ratio } => {
302                let stagnating = prev_residual_norm
303                    .map(|prev| prev > 0.0 && res_new_norm / prev > stagnation_ratio)
304                    .unwrap_or(false);
305                if stagnating {
306                    jacobian = jac_eval(&u)?;
307                }
308            }
309        }
310        prev_residual_norm = Some(res_new_norm);
311    }
312
313    Err(OxiflowError::NewtonNotConverged {
314        iterations: params.max_iterations,
315        residual: last_residual_norm,
316    })
317}

326    // Two synthetic, domain-free test problems — this module is
327    // integrator-agnostic, so its own tests exercise `solve` directly

331    /// f(u) = -lambda*u — affine, so a single correction is exact
332    /// regardless of `max_iterations`/`jacobian_strategy`.
333    fn affine_decay(
334        lambda: f64,
335    ) -> (
336        impl FnMut(&DVector<f64>) -> Result<DVector<f64>, OxiflowError>,
337        impl FnMut(&DVector<f64>) -> Result<DMatrix<f64>, OxiflowError>,
338    ) {
339        let f = move |u: &DVector<f64>| Ok(u.map(|v| -lambda * v));
340        let jac =
341            move |u: &DVector<f64>| Ok(DMatrix::<f64>::identity(u.len(), u.len()) * (-lambda));
342        (f, jac)
343    }
344
345    /// f(u) = -u^3 — genuinely nonlinear; one frozen-Jacobian correction
346    /// is only a first-order approximation, iterated Newton is needed
347    /// for tight convergence.
348    fn cubic_decay() -> (
349        impl FnMut(&DVector<f64>) -> Result<DVector<f64>, OxiflowError>,
350        impl FnMut(&DVector<f64>) -> Result<DMatrix<f64>, OxiflowError>,
351    ) {
352        let f = move |u: &DVector<f64>| Ok(u.map(|v| -v * v * v));
353        let jac =
354            move |u: &DVector<f64>| Ok(DMatrix::<f64>::from_diagonal(&u.map(|v| -3.0 * v * v)));
355        (f, jac)
356    }

358    #[test]
359    fn residual_matches_hand_derivation() {
360        let u = DVector::from_vec(vec![2.0, 3.0]);
361        let u_star = DVector::from_vec(vec![1.0, 1.0]);
362        let f_u = DVector::from_vec(vec![0.5, 0.5]);
363        // g(u) = 1.5*(u - u_star) - 0.1*f(u)
364        let res = residual(&u, &u_star, 1.5, 0.1, &f_u);
365        assert!((res[0] - (1.5 * 1.0 - 0.1 * 0.5)).abs() < 1e-12);
366        assert!((res[1] - (1.5 * 2.0 - 0.1 * 0.5)).abs() < 1e-12);
367    }

369    #[test]
370    fn linear_correction_solves_the_linearised_system() {
371        // alpha*I - coeff*J with J = -2*I, alpha=1, coeff=0.1 -> 1.2*I.
372        let jacobian = DMatrix::<f64>::identity(2, 2) * -2.0;
373        let res = DVector::from_vec(vec![1.2, 2.4]);
374        let delta = linear_correction(&jacobian, 1.0, 0.1, &res, &NalgebraDenseSolver).unwrap();
375        assert!((delta[0] - (-1.0)).abs() < 1e-9);
376        assert!((delta[1] - (-2.0)).abs() < 1e-9);
377    }

379    #[test]
380    fn single_iteration_is_exact_for_affine_problem() {
381        // Backward-Euler-style step: alpha=1, coeff=theta*dt=0.1,
382        // u_star = u_n (theta=1 so the (1-theta) term vanishes).
383        let lambda = 3.0;
384        let (f, jac) = affine_decay(lambda);
385        let u0 = DVector::from_element(3, 1.0);
386        let u_star = u0.clone();
387        let params = NewtonParams {
388            convergence: NewtonConvergence::default(),
389            jacobian_strategy: JacobianStrategy::ModifiedFrozen,
390            max_iterations: 1,
391        };
392
393        let u_next = solve(u0, &u_star, 1.0, 0.1, f, jac, &NalgebraDenseSolver, &params).unwrap();
394
395        let expected = 1.0 / (1.0 + lambda * 0.1);
396        for v in u_next.iter() {
397            assert!((v - expected).abs() < 1e-9, "got {v}, expected {expected}");
398        }
399    }

401    #[test]
402    fn full_newton_converges_on_nonaffine_problem() {
403        let (f, jac) = cubic_decay();
404        let u0 = DVector::from_element(2, 1.0);
405        let u_star = u0.clone();
406        let params = NewtonParams {
407            convergence: NewtonConvergence::default(),
408            jacobian_strategy: JacobianStrategy::FullNewton,
409            max_iterations: 20,
410        };
411
412        let u_next = solve(u0, &u_star, 1.0, 0.1, f, jac, &NalgebraDenseSolver, &params).unwrap();
413
414        // Cross-check against an independently solved scalar equation:
415        // u + 0.1*u^3 = 1.0 (Backward-Euler-style implicit cubic decay).
416        let mut u_scalar = 1.0_f64;
417        for _ in 0..50 {
418            let g = u_scalar + 0.1 * u_scalar.powi(3) - 1.0;
419            let dg = 1.0 + 0.3 * u_scalar.powi(2);
420            u_scalar -= g / dg;
421        }
422        for v in u_next.iter() {
423            assert!((v - u_scalar).abs() < 1e-5, "got {v}, expected {u_scalar}");
424        }
