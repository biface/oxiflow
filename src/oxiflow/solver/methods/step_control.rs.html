1//! # Module `solver::methods::step_control`
2//!
3//! Adaptive step-size control — shared machinery for variable-step
4//! integrators ([DD-036](https://github.com/biface/oxiflow/issues/89)).
5//!
6//! ## Why this is separate from `dopri45.rs`
7//!
8//! [`StepSizeController`] knows nothing about Runge-Kutta, Butcher
9//! tableaux, or any specific integrator — it takes a single abstract
10//! error norm in and produces an accept/reject decision and a new `dt`
11//! out. [`DoPri45Solver`](super::dopri45::DoPri45Solver) is the first
12//! consumer (the error norm comes from the embedded RK4/5 difference),
13//! but a future adaptive implicit solver (deferred, see [DD-033](https://github.com/biface/oxiflow/issues/86)'s note on
14//! iterated Newton) would plug in the same controller with a different
15//! error source: the usual truncation-error norm when Newton converges,
16//! or a forced rejection when it doesn't — a guard applied *before*
17//! calling the controller, not a change to the controller itself.
18//!
19//! ## Error norm convention
20//!
21//! Per-component error is scaled against `atol + rtol * |u|` and combined
22//! as an RMS norm — the standard convention (Hairer, Nørsett & Wanner,
23//! *Solving Ordinary Differential Equations I*, §II.4). A norm `<= 1.0`
24//! means the step satisfies the requested tolerance.
25//!
26//! ## Controller
27//!
28//! A PI (proportional-integral) controller (Gustafsson): the new `dt`
29//! depends on both the current error and the previous one, which damps
30//! oscillation in `dt` better than plain proportional control. Falls back
31//! to proportional-only on the very first call (no previous error yet).
32
33use nalgebra::DVector;
34
35use crate::context::error::OxiflowError;
36use crate::context::value::ContextValue;

38/// Adaptive step-size controller — error-source-agnostic (see [module
39/// docs](self)).
40#[derive(Debug, Clone)]
41#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
42pub struct StepSizeController {
43    rtol: f64,
44    atol: f64,
45    dt_min: f64,
46    dt_max: f64,
47    /// Safety factor applied to every suggested `dt` — standard practice
48    /// to bias slightly toward acceptance margin rather than the
49    /// theoretical optimum, which tends to under-predict in practice.
50    safety: f64,
51    /// Order of the embedded error estimator (DoPri45: 4 — the local
52    /// error order of the 4th-order solution used only for error
53    /// estimation; the 5th-order result is what is actually propagated,
54    /// "local extrapolation").
55    order: f64,
56    /// Most recent error norm seen (accepted or rejected attempt) — `None`
57    /// before the first call, in which case `next_dt` falls back to plain
58    /// proportional control.
59    prev_error_norm: Option<f64>,
60}
61
62impl StepSizeController {
63    /// Creates a controller with the standard safety factor (`0.9`).
64    pub fn new(rtol: f64, atol: f64, dt_min: f64, dt_max: f64, order: f64) -> Self {
65        Self {
66            rtol,
67            atol,
68            dt_min,
69            dt_max,
70            safety: 0.9,
71            order,
72            prev_error_norm: None,
73        }
74    }
75
76    /// Combines per-component `error` against `reference` (typically the
77    /// higher-order solution) into a single RMS norm — see [module
78    /// docs](self) for the convention.
79    ///
80    /// `error.len()` and `reference.len()` are assumed equal; mismatched
81    /// lengths are zipped to the shorter one (callers are responsible for
82    /// passing matching vectors — this is an internal helper, not a
83    /// public-API boundary that needs to defend against misuse).
84    pub fn error_norm(&self, error: &DVector<f64>, reference: &DVector<f64>) -> f64 {
85        self.rms_norm(error.as_slice(), reference.as_slice())
86    }
87
88    /// [`Self::error_norm`] for an error and a reference given as states:
89    /// `ScalarField` or `VectorField`, every value of the state weighing
90    /// the same in the RMS ([#192](https://github.com/biface/oxiflow/issues/192)). Reads the values in place, without
91    /// copying them.
92    ///
93    /// # Errors
94    ///
95    /// - `OxiflowError::TypeMismatch` if `error` and `reference` are not
96    ///   the same variant, or neither is `ScalarField`/`VectorField`.
97    /// - `OxiflowError::InvalidDomain` if their shapes differ.
98    pub(crate) fn state_error_norm(
99        &self,
100        error: &ContextValue,
101        reference: &ContextValue,
102    ) -> Result<f64, OxiflowError> {
103        let (e, u) = match (error, reference) {
104            (ContextValue::ScalarField(e), ContextValue::ScalarField(u)) if e.len() == u.len() => {
105                (e.as_slice(), u.as_slice())
106            }
107            (ContextValue::VectorField(e), ContextValue::VectorField(u))
108                if e.shape() == u.shape() =>
109            {
110                (e.as_slice(), u.as_slice())
111            }
112            (ContextValue::ScalarField(e), ContextValue::ScalarField(u)) => {
113                return Err(OxiflowError::InvalidDomain(format!(
114                    "error length {} != reference length {}",
115                    e.len(),
116                    u.len()
117                )))
118            }
119            (ContextValue::VectorField(e), ContextValue::VectorField(u)) => {
120                return Err(OxiflowError::InvalidDomain(format!(
121                    "error shape {:?} != reference shape {:?}",
122                    e.shape(),
123                    u.shape()
124                )))
125            }
126            (e, u) => {
127                return Err(OxiflowError::TypeMismatch {
128                    expected: e.variant_name(),
129                    actual: u.variant_name(),
130                })
131            }
132        };
133        Ok(self.rms_norm(e, u))
134    }
135
136    /// Weighted RMS of `error` against `reference` — see [module
137    /// docs](self).
138    fn rms_norm(&self, error: &[f64], reference: &[f64]) -> f64 {
139        let n = error.len().max(1);
140        let sum_sq: f64 = error
141            .iter()
142            .zip(reference.iter())
143            .map(|(e, u)| {
144                let scale = self.atol + self.rtol * u.abs();
145                (e / scale).powi(2)
146            })
147            .sum();
148        (sum_sq / n as f64).sqrt()
149    }
150
151    /// `true` if `error_norm` satisfies the tolerance — the step should
152    /// be accepted.
153    pub fn accept(&self, error_norm: f64) -> bool {
154        error_norm <= 1.0
155    }
156
157    /// Computes the next `dt` to try, given the `dt` just attempted and
158    /// the error norm it produced — regardless of whether that attempt is
159    /// accepted or rejected (the caller decides that via
160    /// [`accept`](Self::accept) separately).
161    ///
162    /// Updates the internal PI memory on every call, not only on accepted
163    /// steps: the most recent error measurement is informative even from
164    /// a rejected attempt.
165    ///
166    /// The growth/shrink factor is clamped to `[0.2, 5.0]` (standard
167    /// practice — an unclamped controller can oscillate wildly on rough
168    /// problems), and the resulting `dt` is clamped to `[dt_min, dt_max]`.
169    pub fn next_dt(&mut self, current_dt: f64, error_norm: f64) -> f64 {
170        // Floor to avoid division by zero when the error happens to be
171        // exactly representable as 0.0 (e.g. a perfectly linear problem
172        // hitting machine-exact agreement between the two RK orders).
173        let error_norm = error_norm.max(1e-12);
174
175        let beta1 = 0.7 / self.order;
176        let beta2 = 0.4 / self.order;
177
178        let factor = match self.prev_error_norm {
179            Some(prev) => {
180                let prev = prev.max(1e-12);
181                self.safety * error_norm.powf(-beta1) * prev.powf(beta2)
182            }
183            None => self.safety * error_norm.powf(-1.0 / self.order),
184        };
185
186        let factor = factor.clamp(0.2, 5.0);
187        self.prev_error_norm = Some(error_norm);
188
189        (current_dt * factor).clamp(self.dt_min, self.dt_max)
190    }
191
192    /// The configured minimum step — exposed so callers can detect
193    /// "stuck at dt_min" without re-deriving it.
194    pub fn dt_min(&self) -> f64 {
195        self.dt_min
196    }
197}

209    #[test]
210    fn error_norm_zero_when_error_is_zero() {
211        let controller = make_controller();
212        let error = DVector::from_vec(vec![0.0, 0.0, 0.0]);
213        let reference = DVector::from_vec(vec![1.0, 2.0, 3.0]);
214        assert_eq!(controller.error_norm(&error, &reference), 0.0);
215    }

217    #[test]
218    fn error_norm_scales_with_tolerance() {
219        let controller = StepSizeController::new(0.0, 1.0, 1e-8, 1.0, 4.0);
220        // rtol=0, atol=1 -> scale is always 1 -> norm = RMS(error) directly.
221        let error = DVector::from_vec(vec![1.0, 1.0]);
222        let reference = DVector::from_vec(vec![100.0, 100.0]); // irrelevant when rtol=0
223        assert!((controller.error_norm(&error, &reference) - 1.0).abs() < 1e-12);
224    }

228        let controller = make_controller();

230        assert!(controller.accept(0.5));
231        assert!(!controller.accept(1.0001));
232    }

280    #[test]
281    fn pi_memory_affects_second_call() {
282        // Same error norm twice in a row -- the second call's factor
283        // differs from the first because the PI term now has a previous
284        // error to compare against (None -> Some transition).
285        let mut controller = make_controller();
286        let dt1 = controller.next_dt(0.1, 0.5);
287        let dt2 = controller.next_dt(dt1, 0.5);
288        // Not asserting a specific direction -- just that the PI memory
289        // actually changes the computation path (first call uses the
290        // P-only fallback, second uses the full PI formula).
291        assert!(controller.prev_error_norm.is_some());
292        let _ = dt2; // exercised for the side effect on prev_error_norm
293    }
