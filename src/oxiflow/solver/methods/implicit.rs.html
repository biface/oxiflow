1//! # Module `solver::methods::implicit`
2//!
3//! Shared machinery for implicit (theta-method) integrators — DD-033.
4//!
5//! [`BackwardEulerSolver`](super::backward_euler::BackwardEulerSolver) (θ=1)
6//! and [`CrankNicolsonSolver`](super::crank_nicolson::CrankNicolsonSolver)
7//! (θ=0.5) are thin wrappers around [`theta_method_step_newton`] (DD-044,
8//! #111), which itself relies on [`finite_difference_jacobian`] and the
9//! shared [`super::newton`] loop. [`theta_method_step`] keeps the
10//! original, non-configurable entry point for direct callers — see below.
11//!
12//! ## Why a single shared function for two methods
13//!
14//! The generalised theta method is:
15//!
16//! $$u^{n+1} = u^n + \Delta t\left[(1-\theta) f(u^n) + \theta f(u^{n+1})\right]$$
17//!
18//! Linearising the residual $g(u) = u - u^n - \Delta t[(1-\theta)f(u^n) +
19//! \theta f(u)]$ at $u = u^n$ gives $g(u^n) = -\Delta t \cdot f(u^n)$ for
20//! **any** $\theta$ — the right-hand side of the linear correction doesn't
21//! depend on θ at all. Only the system matrix does
22//! ($I - \theta \Delta t J_f$). One function, one parameter.
23//!
24//! ## v1 scope — frozen Jacobian, single correction (DD-033) by default
25//!
26//! [`theta_method_step`] performs exactly **one** Newton-style correction,
27//! with the Jacobian frozen at $u^n$ — this is exact when `compute_physics`
28//! is affine in `u` and a first-order approximation otherwise, and stays
29//! the behaviour of [`theta_method_step`] itself (used directly by
30//! callers that don't need more) and of
31//! [`BackwardEulerSolver`](super::backward_euler::BackwardEulerSolver)/
32//! [`CrankNicolsonSolver`](super::crank_nicolson::CrankNicolsonSolver)
33//! when left unconfigured.
34//!
35//! [`theta_method_step_newton`] (DD-044, #111) is the genuinely iterated
36//! alternative DD-033 anticipated: it routes through the shared
37//! [`super::newton`] loop with caller-supplied convergence/Jacobian-
38//! strategy/iteration-budget configuration.
39//! `BackwardEulerSolver`/`CrankNicolsonSolver` call it directly (with
40//! `with_newton_convergence`/`with_jacobian_strategy`/
41//! `with_max_newton_iterations` builders); [`theta_method_step`] itself
42//! stays at its original signature by delegating to it with
43//! `k_max = 1`/[`JacobianStrategy::ModifiedFrozen`](super::newton::JacobianStrategy::ModifiedFrozen)
44//! baked in, reproducing its historical output exactly (see
45//! [`super::newton`]'s own docs for why no special-casing is needed for
46//! this to hold on affine problems).
47//!
48//! ## Known gap — sparse path not yet Newton-aware
49//!
50//! [`theta_method_step_adaptive`] (the sparse/banded dispatch, DD-043)
51//! still performs a single frozen-Jacobian correction unconditionally —
52//! it is not wired through [`super::newton`] at this increment. Composing
53//! the two was verified, not implemented, per #111's stated scope:
54//! configuring both a sparse backend *and* a non-default Newton budget on
55//! the same solver silently keeps the sparse path single-shot. Tracked as
56//! a follow-up, mirroring the existing BDF2/sparse gap.
57//!
58//! ## Boundary conditions × the perturbed Jacobian
59//!
60//! [`finite_difference_jacobian`] perturbs each component of the state and
61//! re-evaluates [`evaluate_derivative`], which re-applies boundary
62//! conditions on every call. A Dirichlet-constrained node will have its
63//! perturbation overwritten before `compute_physics` sees it — physically
64//! correct (a BC-constrained node isn't a free unknown): the constrained
65//! node's own *column* in the assembled Jacobian is exactly zero, while
66//! its *row* still reflects genuine physics-based coupling to
67//! unconstrained neighbours. Verified by
68//! `tests::dirichlet_constrained_column_is_exactly_zero_row_keeps_real_coupling`
69//! (#113) — previously documented as a known untested limitation (DD-033,
70//! v0.4.0).

84/// Finite-difference step size for [`finite_difference_jacobian`].
85///
86/// Not scaled by state magnitude (v1 simplification) — fine for the O(1)
87/// stiff linear test problems this targets; revisit if used on
88/// significantly different magnitude scales.
89const FD_EPSILON: f64 = 1e-7;
90
91/// Estimates $\partial f/\partial u$ at `state`, `t`, by forward
92/// differences.
93///
94/// For `f` affine in `u` (the case these implicit methods are validated
95/// against), forward differences are exact regardless of step size — no
96/// truncation error from the linear approximation itself.
97///
98/// See [boundary conditions × the perturbed Jacobian](self#boundary-conditions--the-perturbed-jacobian).
99pub(crate) fn finite_difference_jacobian(
100    domain: &Domain,
101    chain: &[&dyn ContextCalculator],
102    state: &ContextValue,
103    t: f64,
104    dt: f64,
105) -> Result<DMatrix<f64>, OxiflowError> {
106    let base_field = state.as_scalar_field()?.clone();
107    let n = base_field.len();
108
109    let mut base_state = state.clone();
110    let f0 = evaluate_derivative(domain, chain, &mut base_state, t, dt)?;
111    let f0_field = f0.as_scalar_field()?.clone();
112
113    let mut jacobian = DMatrix::<f64>::zeros(n, n);
114
115    for j in 0..n {
116        let mut perturbed_field = base_field.clone();
117        perturbed_field[j] += FD_EPSILON;
118        let mut perturbed_state = ContextValue::ScalarField(perturbed_field);
119
120        let f_j = evaluate_derivative(domain, chain, &mut perturbed_state, t, dt)?;
121        let f_j_field = f_j.as_scalar_field()?;
122
123        for i in 0..n {
124            jacobian[(i, j)] = (f_j_field[i] - f0_field[i]) / FD_EPSILON;
125        }
126    }
127
128    Ok(jacobian)
129}

131/// Convergence criterion guaranteeing [`theta_method_step`]'s
132/// single-correction contract succeeds on the stiffest affine problems
133/// this crate currently tests (`λ = 1e4`-scale exponential decay).
134///
135/// Deliberately distinct from [`NewtonConvergence::default`] (`tol_abs =
136/// 1e-8`): on such problems, [`finite_difference_jacobian`]'s fixed step
137/// size suffers catastrophic-cancellation error whose absolute magnitude
138/// scales roughly as `λ · ε_machine / FD_EPSILON` — empirically up to
139/// ~1e-6 for the stiffest cases tested. The single correction itself is
140/// still mathematically exact for affine `f`; this is a
141/// finite-difference precision floor, not a nonlinearity signal, and
142/// [`NewtonConvergence::default`] remains the right general-purpose
143/// starting point for callers configuring the loop themselves — it just
144/// isn't loose enough to guarantee success on this specific,
145/// already-tested edge of the stiffness range. Used here and by
146/// [`BackwardEulerSolver::default`](super::backward_euler::BackwardEulerSolver)/
147/// [`CrankNicolsonSolver::default`](super::crank_nicolson::CrankNicolsonSolver)
148/// — not by `NewtonConvergence`'s own `Default` impl — precisely so that
149/// zero-config solver construction keeps reproducing history exactly,
150/// without changing what a fresh, explicitly-configured
151/// `NewtonConvergence::default()` means elsewhere.
152pub(crate) fn stiff_jacobian_convergence() -> NewtonConvergence {
153    NewtonConvergence::ResidualOnly {
154        tol_abs: 1e-5,
155        tol_rel: 1e-6,
156    }
157}

159/// Performs one step of the generalised theta method, with the Jacobian
160/// frozen at `state` (one Newton-style correction, not iterated — see
161/// [module docs](self)).
162///
163/// `theta = 1.0` is Backward Euler; `theta = 0.5` is Crank-Nicolson.
164///
165/// `state` is mutated in-place by boundary condition application — same
166/// contract as the explicit solvers (see
167/// [`crate::solver::methods::evaluate_derivative`]): callers should not
168/// assume `state` is left unchanged after this call.
169///
170/// Delegates to [`theta_method_step_newton`] with the pre-DD-044 defaults
171/// (`k_max = 1`, [`JacobianStrategy::ModifiedFrozen`],
172/// [`stiff_jacobian_convergence`]) baked in — kept at its original
173/// signature (DD-044, #110) for direct callers that don't need the
174/// configurable path.
175pub(crate) fn theta_method_step(
176    domain: &Domain,
177    chain: &[&dyn ContextCalculator],
178    state: &mut ContextValue,
179    t: f64,
180    dt: f64,
181    theta: f64,
182    linear_solver: &dyn LinearSolver,
183) -> Result<ContextValue, OxiflowError> {
184    theta_method_step_newton(
185        domain,
186        chain,
187        state,
188        t,
189        dt,
190        theta,
191        linear_solver,
192        stiff_jacobian_convergence(),
193        JacobianStrategy::default(),
194        1,
195    )
196}

198/// Same contract as [`theta_method_step`], routed through the generic
199/// iterated Newton loop ([`newton::solve`]) with caller-supplied
200/// convergence/Jacobian-strategy/iteration-budget configuration (DD-044,
201/// #111) — [`BackwardEulerSolver`](super::backward_euler::BackwardEulerSolver)
202/// and [`CrankNicolsonSolver`](super::crank_nicolson::CrankNicolsonSolver)
203/// call this directly, passing their own configured fields; their
204/// defaults reproduce [`theta_method_step`]'s call above exactly.
205///
206/// Follows [`newton`]'s residual convention with $\alpha = 1$, $c = \theta
207/// \Delta t$, $u^{*} = u^n + \Delta t(1-\theta)f(u^n)$ — see that
208/// module's docs. `f(u^n)` is evaluated once, at time `t` (the known
209/// term); every subsequent candidate `f`/Jacobian evaluation inside the
210/// Newton loop is at time `t + dt` (the implicit/target time), matching
211/// [`theta_method_step`]'s original convention.
212#[allow(clippy::too_many_arguments)]
213pub(crate) fn theta_method_step_newton(
214    domain: &Domain,
215    chain: &[&dyn ContextCalculator],
216    state: &mut ContextValue,
217    t: f64,
218    dt: f64,
219    theta: f64,
220    linear_solver: &dyn LinearSolver,
221    newton_convergence: NewtonConvergence,
222    jacobian_strategy: JacobianStrategy,
223    max_iterations: usize,
224) -> Result<ContextValue, OxiflowError> {
225    // f(u^n, t) -- BCs applied in-place to `state` itself, consistent with
226    // the explicit solvers' contract (same as `theta_method_step`).
227    let f_n = evaluate_derivative(domain, chain, state, t, dt)?;
228    let u_n_field = state.as_scalar_field()?.clone();
229    let f_n_field = f_n.as_scalar_field()?.clone();
230
231    let u0 = u_n_field.clone();
232    let u_star: DVector<f64> = u_n_field + f_n_field * (dt * (1.0 - theta));
233    let coeff = theta * dt;
234    let t_target = t + dt;
235
236    let u_next = newton::solve(
237        u0,
238        &u_star,
239        1.0,
240        coeff,
241        move |candidate: &DVector<f64>| {
242            let mut candidate_state = ContextValue::ScalarField(candidate.clone());
243            let f = evaluate_derivative(domain, chain, &mut candidate_state, t_target, dt)?;
244            Ok(f.as_scalar_field()?.clone())
245        },
246        move |candidate: &DVector<f64>| {
247            let candidate_state = ContextValue::ScalarField(candidate.clone());
248            finite_difference_jacobian(domain, chain, &candidate_state, t_target, dt)
249        },
250        linear_solver,
251        &newton::NewtonParams {
252            convergence: newton_convergence,
253            jacobian_strategy,
254            max_iterations,
255        },
256    )?;
257
258    Ok(ContextValue::ScalarField(u_next))
259}

261// ── Sparse / banded path (DD-013 second phase, DD-043) ─────────────────────────
262//
263// `#[cfg(feature = "sparse")]` only. Bandwidth is always an explicit
264// parameter here, supplied by the caller (the solver's own HOW-side
265// configuration, e.g. `BackwardEulerSolver::with_jacobian_bandwidth`) — it
266// is never read from `Domain`/`PhysicalModel`. See DD-043, amendment 1, for
267// why `PhysicalModel::jacobian_bandwidth` was rejected: bandwidth is a
268// property of the numerical scheme (HOW), not of the physical model (WHAT).
269
270/// A single Jacobian entry: `(row, col, value)` — one nonzero of `∂f/∂u`
271/// within the band, as produced by [`banded_jacobian_entries`]. Not yet a
272/// `faer` triplet: callers convert only when they actually need the
273/// sparse matrix type (see [`banded_finite_difference_jacobian`] and
274/// [`theta_method_step_adaptive`]), rather than paying that conversion
275/// cost inside the assembly loop itself.
276#[cfg(feature = "sparse")]
277type SparseEntry = (usize, usize, f64);
278
279/// Return type of [`banded_jacobian_entries`]: the system size `n`, and
280/// the raw list of [`SparseEntry`] within the declared band. Each
281/// `(row, col)` pair appears at most once — every column belongs to
282/// exactly one CPR color, so no two colors can ever produce the same
283/// entry.
284#[cfg(feature = "sparse")]
285type BandedJacobianResult = Result<(usize, Vec<SparseEntry>), OxiflowError>;
286
287/// Assembles the nonzero entries of $\partial f/\partial u$ within a band of
288/// half-width `bandwidth` around the diagonal, via Curtis-Powell-Reid (CPR)
289/// graph coloring: for a genuinely banded Jacobian, columns more than
290/// `2*bandwidth + 1` apart never share a nonzero row, so all columns of the
291/// same color can be perturbed **simultaneously** in a single
292/// [`evaluate_derivative`] call — `2*bandwidth + 1` evaluations total,
293/// instead of [`finite_difference_jacobian`]'s `n`.
294///
295/// Returns `(n, triplets)` — raw `(row, col, value)` entries, not yet a
296/// `faer` sparse matrix, so callers computing the *system* matrix
297/// (`I - theta*dt*J`, see [`theta_method_step_adaptive`]) can fold the
298/// identity in before building the sparse type once, rather than building
299/// the raw Jacobian in sparse form only to rebuild it again.
300#[cfg(feature = "sparse")]
301pub(crate) fn banded_jacobian_entries(
302    domain: &Domain,
303    chain: &[&dyn ContextCalculator],
304    state: &ContextValue,
305    t: f64,
306    dt: f64,
307    bandwidth: usize,
308) -> BandedJacobianResult {
309    let base_field = state.as_scalar_field()?.clone();
310    let n = base_field.len();
311
312    let mut base_state = state.clone();
313    let f0 = evaluate_derivative(domain, chain, &mut base_state, t, dt)?;
314    let f0_field = f0.as_scalar_field()?.clone();
315
316    let n_colors = 2 * bandwidth + 1;
317    let mut triplets = Vec::new();
318
319    for color in 0..n_colors {
320        let columns: Vec<usize> = (color..n).step_by(n_colors).collect();
321        if columns.is_empty() {
322            continue;
323        }
324
325        let mut perturbed_field = base_field.clone();
326        for &j in &columns {
327            perturbed_field[j] += FD_EPSILON;
328        }
329        let mut perturbed_state = ContextValue::ScalarField(perturbed_field);
330
331        let f_c = evaluate_derivative(domain, chain, &mut perturbed_state, t, dt)?;
332        let f_c_field = f_c.as_scalar_field()?;
333
334        // Columns of the same color are spaced `n_colors > 2*bandwidth`
335        // apart, so their bands ([j-bandwidth, j+bandwidth]) never
336        // overlap — each perturbed row's response is attributable to
337        // exactly one column, no disambiguation needed.
338        for &j in &columns {
339            let lo = j.saturating_sub(bandwidth);
340            let hi = (j + bandwidth + 1).min(n);
341            for i in lo..hi {
342                let deriv = (f_c_field[i] - f0_field[i]) / FD_EPSILON;
343                triplets.push((i, j, deriv));
344            }
345        }
346    }
347
348    Ok((n, triplets))
349}

351/// [`banded_jacobian_entries`], assembled into a `faer` sparse matrix.
352///
353/// Kept separate from [`banded_jacobian_entries`] for dense/sparse parity
354/// testing against [`finite_difference_jacobian`] — see the test module.
355// TODO(v0.6.0): wire in once BackwardEulerSolver/CrankNicolsonSolver
356// select the banded path in practice (DD-043) — currently only exercised
357// by the dense/sparse parity test.
358#[allow(dead_code)]
359#[cfg(feature = "sparse")]
360pub(crate) fn banded_finite_difference_jacobian(
361    domain: &Domain,
362    chain: &[&dyn ContextCalculator],
363    state: &ContextValue,
364    t: f64,
365    dt: f64,
366    bandwidth: usize,
367) -> Result<faer::sparse::SparseColMat<usize, f64>, OxiflowError> {
368    let (n, triplets) = banded_jacobian_entries(domain, chain, state, t, dt, bandwidth)?;
369
370    let faer_triplets: Vec<faer::sparse::Triplet<usize, usize, f64>> = triplets
371        .into_iter()
372        .map(|(i, j, v)| faer::sparse::Triplet::new(i, j, v))
373        .collect();
374
375    faer::sparse::SparseColMat::try_new_from_triplets(n, n, &faer_triplets).map_err(|e| {
376        OxiflowError::PreconditionFailed {
377            context: "banded_finite_difference_jacobian",
378            message: format!("failed to build sparse matrix from triplets: {e:?}"),
379        }
380    })
381}

383/// Same contract as [`theta_method_step`], with an additional sparse path:
384/// when `jacobian_bandwidth` is `Some(k)` **and** `n > sparse_threshold`,
385/// assembles `I - theta*dt*J` directly in sparse (banded) form via
386/// [`banded_jacobian_entries`] and solves it with `sparse_solver`. Falls
387/// back to [`theta_method_step`] unchanged in every other case — `None`
388/// bandwidth, or a small enough system that the dense path is cheaper
389/// regardless of bandwidth.
390///
391/// `jacobian_bandwidth`/`sparse_threshold`/`sparse_solver` are supplied by
392/// the caller's own HOW-side configuration (DD-043, amendment 1) — never
393/// read from `Domain` or `PhysicalModel`.
394#[cfg(feature = "sparse")]
395#[allow(clippy::too_many_arguments)]
396pub(crate) fn theta_method_step_adaptive(
397    domain: &Domain,
398    chain: &[&dyn ContextCalculator],
399    state: &mut ContextValue,
400    t: f64,
401    dt: f64,
402    theta: f64,
403    dense_solver: &dyn LinearSolver,
404    sparse_solver: &dyn SparseLinearSolver,
405    sparse_threshold: usize,
406    jacobian_bandwidth: Option<usize>,
407) -> Result<ContextValue, OxiflowError> {
408    let n = state.as_scalar_field()?.len();
409
410    match jacobian_bandwidth {
411        Some(k) if n > sparse_threshold => {
412            let f_n = evaluate_derivative(domain, chain, state, t, dt)?;
413            let u_n_field = state.as_scalar_field()?.clone();
414            let f_n_field = f_n.as_scalar_field()?.clone();
415
416            // Jacobian frozen at the (now BC-corrected) u^n, evaluated at
417            // t + dt — same convention as theta_method_step.
418            let (n_check, raw_triplets) =
419                banded_jacobian_entries(domain, chain, state, t + dt, dt, k)?;
420            debug_assert_eq!(n_check, n);
421
422            // I - theta*dt*J, folded into the triplets directly rather than
423            // building J's sparse matrix first and combining afterward —
424            // banded_jacobian_entries always includes the diagonal (j is
425            // always within its own band), so every row gets its "+1.0".
426            let coeff = -theta * dt;
427            let system_triplets: Vec<faer::sparse::Triplet<usize, usize, f64>> = raw_triplets
428                .into_iter()
429                .map(|(i, j, deriv)| {
430                    let mut value = coeff * deriv;
431                    if i == j {
432                        value += 1.0;
433                    }
434                    faer::sparse::Triplet::new(i, j, value)
435                })
436                .collect();
437
438            let system_matrix =
439                faer::sparse::SparseColMat::try_new_from_triplets(n, n, &system_triplets).map_err(
440                    |e| OxiflowError::PreconditionFailed {
441                        context: "theta_method_step_adaptive",
442                        message: format!("failed to build sparse system matrix: {e:?}"),
443                    },
444                )?;
445
446            let rhs = f_n_field * dt;
447            let delta_u = sparse_solver.solve(&system_matrix, &rhs)?;
448
449            let u_next: DVector<f64> = u_n_field + delta_u;
450            Ok(ContextValue::ScalarField(u_next))
451        }
452        _ => theta_method_step(domain, chain, state, t, dt, theta, dense_solver),
453    }
454}

620    /// Discrete tridiagonal diffusion with a spike initial condition —
621    /// genuine spatial coupling to neighbours, needed to distinguish
622    /// "legitimate physics coupling" from "spurious coupling introduced by
623    /// BC re-application" in the test below. A local copy rather than
624    /// reusing `sparse_tests::TridiagonalDiffusion`: that fixture only
625    /// exists under `feature = "sparse"`, and this test needs no sparse
626    /// machinery.
627    #[derive(Debug)]
628    struct TridiagonalDiffusion {
629        diffusion: f64,
630    }
631
632    impl RequiresContext for TridiagonalDiffusion {
633        fn required_variables(&self) -> Vec<ContextVariable> {
634            vec![]
635        }
636    }
637
638    impl PhysicalModel for TridiagonalDiffusion {
639        fn compute_physics(
640            &self,
641            state: &ContextValue,
642            _ctx: &ComputeContext,
643        ) -> Result<ContextValue, OxiflowError> {
644            let u = state.as_scalar_field()?;
645            let n = u.len();
646            let d = self.diffusion;
647            let out = DVector::from_fn(n, |i, _| {
648                let left = if i > 0 { u[i - 1] } else { 0.0 };
649                let right = if i + 1 < n { u[i + 1] } else { 0.0 };
650                d * (left - 2.0 * u[i] + right)
651            });
652            Ok(ContextValue::ScalarField(out))
653        }
654
655        fn initial_state(&self, mesh: &dyn Mesh) -> ContextValue {
656            let n = mesh.n_dof();
657            ContextValue::ScalarField(DVector::from_fn(
658                n,
659                |i, _| if i == n / 2 { 1.0 } else { 0.0 },
660            ))
661        }
662
663        fn name(&self) -> &str {
664            "tridiagonal_diffusion_dirichlet_test"
665        }
666    }

700    #[test]
701    fn dirichlet_constrained_column_is_exactly_zero_row_keeps_real_coupling() {
702        // DD-033's known limitation, closed by #113: `finite_difference_jacobian`
703        // perturbs each state component and re-evaluates `evaluate_derivative`,
704        // which re-applies boundary conditions on every call. A
705        // Dirichlet-constrained node's perturbation is therefore overwritten
706        // before `compute_physics` ever sees it -- physically correct (a
707        // BC-constrained node isn't a free unknown), verified here rather
708        // than merely documented.
709        let n = 5;
710        let constrained = n / 2; // the model's initial spike location
711        let diffusion = 0.7;
712
713        let domain = Domain::new(
714            "dirichlet_test",
715            Box::new(TridiagonalDiffusion { diffusion }),
716            make_mesh(n),
717        )
718        .with_boundary_conditions(vec![Box::new(FixedDirichlet {
719            node_index: constrained,
720            value: 0.0,
721        })]);
722        let scenario = Scenario::multi(vec![domain]).unwrap();
723        let domain = scenario.single_domain().unwrap();
724        let requirements = scenario.context_requirements();
725        let chain = build_calculator_chain(&requirements, &[]).unwrap();
726
727        let state = domain.model.initial_state(domain.mesh.as_ref());
728        let jac = finite_difference_jacobian(domain, &chain, &state, 0.0, 0.01).unwrap();
729
730        // Column: perturbing the constrained node itself must leave every
731        // row's output exactly unchanged -- the perturbation never
732        // survives BC re-application, so both the base and perturbed
733        // evaluations see bit-identical states, giving f_j - f0 = 0
734        // exactly, not merely small.
735        for i in 0..n {
736            assert_eq!(
737                jac[(i, constrained)],
738                0.0,
739                "column {constrained} (the constrained node) must be exactly \
740                 zero at row {i} -- got {}",
741                jac[(i, constrained)]
742            );
743        }
744
745        // Row: the constrained node's own output still genuinely depends
746        // on its unconstrained neighbours through the model's real spatial
747        // coupling -- legitimate physics, not spurious coupling, and must
748        // not be zeroed out by the fix the column check above verifies.
749        let left = constrained - 1;
750        let right = constrained + 1;
751        assert!(
752            (jac[(constrained, left)] - diffusion).abs() < 1e-6,
753            "expected left-neighbour coupling ~{diffusion}, got {}",
754            jac[(constrained, left)]
755        );
756        assert!(
757            (jac[(constrained, right)] - diffusion).abs() < 1e-6,
758            "expected right-neighbour coupling ~{diffusion}, got {}",
759            jac[(constrained, right)]
760        );
761    }
