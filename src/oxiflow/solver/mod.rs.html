1//! # Module `solver`
2//!
3//! Numerical solving orchestration — WHAT/HOW separation (issue [#32](https://github.com/biface/oxiflow/issues/32)).
4//!
5//! ## Responsibilities
6//!
7//! | Type | Role |
8//! |---|---|
9//! | [`scenario::Scenario`] | Declares the problem (WHAT) |
10//! | [`config::SolverConfiguration`] | Configures solving (HOW) |
11//! | [`Solver`] | Orchestrates execution |
12//!
13//! ## Contractual execution order
14//!
15//! `Solver::solve()` implementations must follow this order at each derivative
16//! evaluation ([DD-070](https://github.com/biface/oxiflow/issues/200)):
17//!
18//! 1. **Calculators required by the boundary conditions** — populate
19//!    `ComputeContext` with the variables the conditions require (with their
20//!    dependencies), in chain order
21//! 2. **Boundary conditions** — apply to state using `ctx` (J2)
22//! 3. **Other calculators** — complete `ComputeContext` from the constrained
23//!    state, in chain order
24//! 4. **`compute_physics`** — compute `du/dt` from state + context, then set
25//!    it to zero at the nodes the conditions declare
26//! 5. **Integrate** — advance state by `dt`
27//!
28//! The boundary conditions are also applied to the initial state and to the
29//! state at the end of each step, before it is checked and saved.
30//!
31//! This order is a contract, not a convention. Deviating from it produces
32//! silently incorrect results.

58// ── SimulationResult ──────────────────────────────────────────────────────────

60/// Result of a completed simulation.
61///
62/// `states` and `times` have the same length. The save frequency is controlled
63/// by `SolverConfiguration::time.save_every`.
64///
65/// # Examples
66///
67/// ```rust
68/// use oxiflow::mesh::UniformGrid1D;
69/// use oxiflow::solver::methods::ForwardEulerSolver;
70/// use oxiflow::solver::{IntegratorKind, Scenario, SimulationResult, Solver,
71///                       SolverConfiguration, StepControl, TimeConfiguration};
72/// # use nalgebra::DVector;
73/// # use oxiflow::context::{ComputeContext, ContextValue, ContextVariable, OxiflowError};
74/// # use oxiflow::mesh::Mesh;
75/// # use oxiflow::model::{PhysicalModel, RequiresContext};
76/// # struct Decay;
77/// # impl RequiresContext for Decay {
78/// #     fn required_variables(&self) -> Vec<ContextVariable> { vec![] }
79/// # }
80/// # impl PhysicalModel for Decay {
81/// #     fn compute_physics(
82/// #         &self,
83/// #         state: &ContextValue,
84/// #         _ctx: &ComputeContext,
85/// #     ) -> Result<ContextValue, OxiflowError> {
86/// #         Ok(ContextValue::ScalarField(-state.as_scalar_field()?.clone()))
87/// #     }
88/// #     fn initial_state(&self, mesh: &dyn Mesh) -> ContextValue {
89/// #         ContextValue::ScalarField(DVector::from_element(mesh.n_dof(), 1.0))
90/// #     }
91/// #     fn name(&self) -> &str { "decay" }
92/// # }
93/// // `SimulationResult` is `#[non_exhaustive]`: it is obtained from a solver,
94/// // not built by hand.
95/// let scenario = Scenario::single(Box::new(Decay), Box::new(UniformGrid1D::new(10, 0.0, 1.0)?));
96/// let config = SolverConfiguration::new(
97///     TimeConfiguration::new(1.0, StepControl::Fixed { dt: 0.1 }),
98///     IntegratorKind::Euler,
99/// );
100/// let result: SimulationResult = ForwardEulerSolver.solve(&scenario, &config)?;
101/// assert_eq!(result.states.len(), result.times.len());
102/// assert_eq!(result.times[0], 0.0);
103/// # Ok::<(), OxiflowError>(())
104/// ```
105#[non_exhaustive]
106#[derive(Debug, Clone)]
107#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
108pub struct SimulationResult {
109    /// Saved field states at each recorded time.
110    pub states: Vec<crate::context::value::ContextValue>,
111    /// Simulation times corresponding to each saved state.
112    pub times: Vec<f64>,
113    /// Total number of time steps taken (may be larger than `states.len()`
114    /// if `save_every > 1`).
115    pub n_steps: usize,
116    /// Solver metadata: timing, rejected steps, convergence info.
117    ///
118    /// Keys follow the convention `"solver.<key>"` (e.g. `"solver.rejected_steps"`).
119    /// Filled by `DoPri45Solver` (`solver.accepted_steps`,
120    /// `solver.rejected_steps`, `solver.cfl_limited_steps`); empty for the
121    /// fixed-step integrators.
122    pub metadata: std::collections::HashMap<String, f64>,
123}
124
125impl SimulationResult {
126    /// Returns the number of saved states.
127    pub fn len(&self) -> usize {
128        self.states.len()
129    }
130
131    /// Returns `true` if no states were saved.
132    pub fn is_empty(&self) -> bool {
133        self.states.is_empty()
134    }
135
136    /// Returns the final simulation time.
137    pub fn t_final(&self) -> Option<f64> {
138        self.times.last().copied()
139    }

141    /// Writes the **final** saved state to an XML VTK unstructured grid
142    /// (`.vtu`), for visualisation in ParaView/VisIt ([DT-027](https://github.com/biface/oxiflow/issues/75), issue [#78](https://github.com/biface/oxiflow/issues/78)).
143    ///
144    /// Node coordinates come from `mesh`; nodes are connected by `Line`
145    /// cells in `Mesh` node order (`n_dof() - 1` segments), matching the
146    /// only mesh currently implemented (`UniformGrid1D`). Only `mesh.
147    /// spatial_dimension() == 1` is supported for this first increment —
148    /// coordinates are padded with zeros to the 3 components VTK requires.
149    /// 2D/3D meshes are planned for v0.11.0.
150    ///
151    /// The state is written as point data, one scalar `Float64` array per
152    /// component:
153    ///
154    /// | Final state | Point-data arrays |
155    /// |---|---|
156    /// | `ScalarField` (n values) | `field` |
157    /// | `VectorField` (n × d) | `field_0`, `field_1`, …, `field_{d-1}` (column `c` → `field_c`) |
158    ///
159    /// As a state, a `VectorField` holds d unknowns per node (several
160    /// species, several physical fields), with d unrelated to the mesh
161    /// dimension: a 1D two-species model has d = 2. One array per component
162    /// gives each its own colour map and range in ParaView, where a single
163    /// `NumberOfComponents = d` array would be shown as a geometric vector
164    /// (magnitude by default). The first array is declared as the
165    /// active scalars (`<PointData Scalars="…">`).
166    ///
167    /// The file is UTF-8 XML; data arrays use VTK's inline `binary` format
168    /// (little-endian bytes, base64-encoded, `UInt64` size header), so every
169    /// `f64` is stored bit for bit — NaN and infinities included, which lets
170    /// a diverged state be inspected in ParaView. It is written by the crate
171    /// itself, with no third-party VTK dependency ([#191](https://github.com/biface/oxiflow/issues/191)).
172    ///
173    /// Exports the last entry of `states` only — not the full time series.
174    /// A `.pvd` time-series collection is future work, not part of this
175    /// increment (see [#78](https://github.com/biface/oxiflow/issues/78)'s acceptance criteria, scoped to a single file).
176    ///
177    /// Requires the `vtk` feature.
178    ///
179    /// # Errors
180    ///
181    /// Returns [`OxiflowError::Persistence`] if: `states` is empty; `mesh`
182    /// has `spatial_dimension() != 1` or zero degrees of freedom; the last
183    /// state is neither a [`crate::context::value::ContextValue::ScalarField`]
184    /// nor a [`crate::context::value::ContextValue::VectorField`]; a
185    /// `ScalarField` length or a `VectorField` row count doesn't match
186    /// `mesh.n_dof()`; a `VectorField` has no column; or the file cannot be
187    /// created or written.
188    #[cfg(feature = "vtk")]
189    pub fn write_vtk(
190        &self,
191        mesh: &dyn crate::mesh::Mesh,
192        path: &std::path::Path,
193    ) -> Result<(), OxiflowError> {
194        use crate::context::value::ContextValue;
195        use std::io::Write;
196
197        if mesh.spatial_dimension() != 1 {
198            return Err(OxiflowError::Persistence(format!(
199                "write_vtk currently only supports 1D meshes (Line cells), \
200                 got spatial_dimension() = {}",
201                mesh.spatial_dimension()
202            )));
203        }
204
205        let n = mesh.n_dof();
206        if n == 0 {
207            return Err(OxiflowError::Persistence(
208                "write_vtk: mesh has zero degrees of freedom".to_string(),
209            ));
210        }
211
212        let last_state = self.states.last().ok_or_else(|| {
213            OxiflowError::Persistence("write_vtk: SimulationResult has no saved states".to_string())
214        })?;
215        // Un tableau `(nom, valeurs)` par composante de l'état.
216        let fields: Vec<(String, &[f64])> = match last_state {
217            ContextValue::ScalarField(v) => {
218                if v.len() != n {
219                    return Err(OxiflowError::Persistence(format!(
220                        "write_vtk: state has {} values, mesh has {n} degrees of freedom",
221                        v.len()
222                    )));
223                }
224                vec![("field".to_string(), v.as_slice())]
225            }
226            ContextValue::VectorField(m) => {
227                if m.nrows() != n {
228                    return Err(OxiflowError::Persistence(format!(
229                        "write_vtk: VectorField state has {} rows, mesh has {n} degrees of freedom",
230                        m.nrows()
231                    )));
232                }
233                if m.ncols() == 0 {
234                    return Err(OxiflowError::Persistence(
235                        "write_vtk: VectorField state has no component".to_string(),
236                    ));
237                }
238                // `DMatrix` est stocké par colonnes (column-major) : la
239                // composante `c` occupe la tranche contiguë `[c·n, (c+1)·n)`.
240                m.as_slice()
241                    .chunks_exact(n)
242                    .enumerate()
243                    .map(|(c, column)| (format!("field_{c}"), column))
244                    .collect()
245            }
246            other => {
247                return Err(OxiflowError::Persistence(format!(
248                    "write_vtk only supports ScalarField and VectorField states, got {other:?}"
249                )));
250            }
251        };
252        let x: Vec<f64> = (0..n).map(|i| mesh.coordinates(i)[0]).collect();
253
254        let io_err =
255            |e: std::io::Error| OxiflowError::Persistence(format!("VTK export failed: {e}"));
256        let file = std::fs::File::create(path).map_err(io_err)?;
257        let mut w = std::io::BufWriter::new(file);
258        write_vtu_1d(&mut w, &x, &fields)
259            .and_then(|()| w.flush())
260            .map_err(io_err)
261    }
262}

264/// Écrit une grille VTK XML `UnstructuredGrid` 1D, tableaux au format
265/// `binary` en ligne — corps de [`SimulationResult::write_vtk`], qui a déjà
266/// validé les entrées (`x.len() >= 1`, au moins un tableau, chacun de
267/// longueur `x.len()`).
268///
269/// `fields` : tableaux scalaires de `PointData`, écrits dans l'ordre donné ;
270/// le premier est déclaré comme scalaires actifs (`Scalars`).
271///
272/// Nœuds reliés par des cellules `Line` (type VTK 3) dans l'ordre de `x` ;
273/// un seul `Vertex` (type 1) si `x.len() == 1`. Coordonnées complétées par
274/// des zéros (VTK stocke toujours 3 composantes).
275#[cfg(feature = "vtk")]
276fn write_vtu_1d<W: std::io::Write, S: AsRef<str>>(
277    w: &mut W,
278    x: &[f64],
279    fields: &[(S, &[f64])],
280) -> std::io::Result<()> {
281    let n = x.len();
282    let (n_cells, nodes_per_cell, cell_type) = if n >= 2 { (n - 1, 2, 3u8) } else { (1, 1, 1u8) };
283
284    let points: Vec<u8> = x
285        .iter()
286        .flat_map(|&xi| [xi, 0.0, 0.0])
287        .flat_map(f64::to_le_bytes)
288        .collect();
289    let connectivity: Vec<u8> = if n >= 2 {
290        (0..n as i64 - 1)
291            .flat_map(|i| [i, i + 1])
292            .flat_map(i64::to_le_bytes)
293            .collect()
294    } else {
295        0i64.to_le_bytes().to_vec()
296    };
297    let offsets: Vec<u8> = (1..=n_cells as i64)
298        .flat_map(|c| (c * nodes_per_cell).to_le_bytes())
299        .collect();
300
301    writeln!(w, r#"<?xml version="1.0" encoding="UTF-8"?>"#)?;
302    writeln!(
303        w,
304        r#"<VTKFile type="UnstructuredGrid" version="1.0" byte_order="LittleEndian" header_type="UInt64">"#
305    )?;
306    writeln!(w, "  <UnstructuredGrid>")?;
307    writeln!(
308        w,
309        r#"    <Piece NumberOfPoints="{n}" NumberOfCells="{n_cells}">"#
310    )?;
311    let active = fields.first().map_or("", |(name, _)| name.as_ref());
312    writeln!(w, r#"      <PointData Scalars="{active}">"#)?;
313    for (name, values) in fields {
314        let bytes: Vec<u8> = values.iter().flat_map(|v| v.to_le_bytes()).collect();
315        write_data_array(w, "Float64", name.as_ref(), 1, &bytes)?;
316    }
317    writeln!(w, "      </PointData>")?;
318    writeln!(w, "      <Points>")?;
319    write_data_array(w, "Float64", "Points", 3, &points)?;
320    writeln!(w, "      </Points>")?;
321    writeln!(w, "      <Cells>")?;
322    write_data_array(w, "Int64", "connectivity", 1, &connectivity)?;
323    write_data_array(w, "Int64", "offsets", 1, &offsets)?;
324    write_data_array(w, "UInt8", "types", 1, &vec![cell_type; n_cells])?;
325    writeln!(w, "      </Cells>")?;
326    writeln!(w, "    </Piece>")?;
327    writeln!(w, "  </UnstructuredGrid>")?;
328    writeln!(w, "</VTKFile>")
329}

331/// Écrit un `DataArray` au format `binary` en ligne : en-tête `UInt64`
332/// (taille des données en octets, petit-boutiste) suivi des données, encodés
333/// ensemble en base64 — la forme non compressée que VTK lit, et celle que
334/// produisait `vtkio`.
335#[cfg(feature = "vtk")]
336fn write_data_array<W: std::io::Write>(
337    w: &mut W,
338    scalar_type: &str,
339    name: &str,
340    components: usize,
341    bytes: &[u8],
342) -> std::io::Result<()> {
343    let mut payload = Vec::with_capacity(8 + bytes.len());
344    payload.extend_from_slice(&(bytes.len() as u64).to_le_bytes());
345    payload.extend_from_slice(bytes);
346    writeln!(
347        w,
348        r#"        <DataArray type="{scalar_type}" Name="{name}" NumberOfComponents="{components}" format="binary">"#
349    )?;
350    writeln!(w, "          {}", base64_encode(&payload))?;
351    writeln!(w, "        </DataArray>")
352}
353
354/// Encodage base64 standard (RFC 4648 §4, alphabet `A–Z a–z 0–9 + /`, avec
355/// remplissage `=`), seul encodage requis par le format `binary` de VTK.
356#[cfg(feature = "vtk")]
357fn base64_encode(bytes: &[u8]) -> String {
358    const ALPHABET: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
359    let mut out = String::with_capacity(bytes.len().div_ceil(3) * 4);
360    for chunk in bytes.chunks(3) {
361        let b = [
362            chunk[0],
363            *chunk.get(1).unwrap_or(&0),
364            *chunk.get(2).unwrap_or(&0),
365        ];
366        let triple = u32::from(b[0]) << 16 | u32::from(b[1]) << 8 | u32::from(b[2]);
367        for k in 0..4 {
368            if k <= chunk.len() {
369                out.push(ALPHABET[(triple >> (18 - 6 * k) & 0x3f) as usize] as char);
370            } else {
371                out.push('=');
372            }
373        }
374    }
375    out
376}

378// ── Solver trait ──────────────────────────────────────────────────────────────

380/// Orchestrates the time integration loop.
381///
382/// Implementations receive a `Scenario` (WHAT) and a `SolverConfiguration`
383/// (HOW) and execute the contractual loop until `t_end`.
384///
385/// `Solver` implementations must:
386/// - Verify `scenario.n_domains() == 1`
387/// - Build the calculator chain via `chain::build_calculator_chain()`
388/// - Follow the contractual execution order
389///
390/// `Solver` is deliberately kept single-domain ([DA-031](https://github.com/biface/oxiflow/issues/82)) — coupled
391/// multi-domain scenarios go through [`orchestrator::MultiDomainOrchestrator`]
392/// instead, which drives one [`methods::SteppableSolver`] per domain and
393/// invokes `CouplingOperator`s between steps. This keeps each domain free
394/// to use a different integrator, and leaves `Solver`/`SimulationResult`
395/// unchanged for the well-tested single-domain path.
396///
397/// # Object safety
398///
399/// This trait is object-safe to support INV-4 (plugin-safe API, v0.30.0 / J30, [DA-017](https://github.com/biface/oxiflow/issues/17)).
400pub trait Solver: Send + Sync {
401    /// Runs the simulation and returns the collected states.
402    fn solve(
403        &self,
404        scenario: &Scenario,
405        config: &SolverConfiguration,
406    ) -> Result<SimulationResult, OxiflowError>;
407
408    /// Called automatically right before a divergence-class error is
409    /// returned, with a [`SimulationSnapshot`] capturing the state at the
410    /// point of failure ([DT-025](https://github.com/biface/oxiflow/issues/66) Option B, issue [#71](https://github.com/biface/oxiflow/issues/71)). Two distinct
411    /// failure modes are routed through this hook:
412    ///
413    /// - [`OxiflowError::SolverDivergence`] — a non-finite value appeared
414    ///   in the field itself (checked after every step by
415    ///   [`methods`]'s internal finiteness check).
416    /// - [`OxiflowError::NewtonNotConverged`] — the iterated Newton
417    ///   solver ([`methods::newton`]) exhausted its iteration budget
418    ///   without satisfying its convergence criterion ([DD-044](https://github.com/biface/oxiflow/issues/109), v0.7.0).
419    ///   This is a failure of the *nonlinear correction* to converge, not
420    ///   necessarily a non-finite state — the last iterate may still be
421    ///   finite, just not accepted as converged.
422    ///
423    /// Both cases reuse the same [`SimulationSnapshot`] checkpoint
424    /// infrastructure; implementations that override this hook to
425    /// persist or log should not assume the snapshot's `error` field
426    /// always describes a non-finite value.
427    ///
428    /// Default implementation is a no-op — override to persist (e.g. via
429    /// `snapshot::write_snapshot`, `serde` feature) or log. There is no `checkpoint()`
430    /// counterpart on this trait; see the [`snapshot`] module docs for why
431    /// explicit checkpoints are constructed directly by caller code instead.
432    fn on_divergence(&self, snapshot: &snapshot::SimulationSnapshot) {
433        let _ = snapshot;
434    }
435}

579    /// Relit un `.vtu` écrit par `write_vtk` : renvoie `(NumberOfPoints,
580    /// NumberOfCells)` de la `Piece` et, pour chaque `DataArray`, ses valeurs
581    /// (converties en `f64`) indexées par l'attribut `Name`. Vérifie au passage
582    /// le format `binary`, l'en-tête de taille `UInt64` et la longueur des
583    /// données selon le type déclaré. Échoue si le XML est mal formé.
584    #[cfg(feature = "vtk")]
585    fn parse_vtu(text: &str) -> ((usize, usize), std::collections::HashMap<String, Vec<f64>>) {
586        use quick_xml::events::Event;
587        use quick_xml::Reader;
588
589        let attr = |e: &quick_xml::events::BytesStart, key: &str| -> String {
590            let a = e
591                .try_get_attribute(key)
592                .unwrap()
593                .expect("missing attribute");
594            String::from_utf8(a.value.into_owned()).unwrap()
595        };
596
597        let mut reader = Reader::from_str(text);
598        let mut piece = (0, 0);
599        let mut arrays = std::collections::HashMap::new();
600        let mut current: Option<(String, String)> = None; // (Name, type)
601        loop {
602            match reader.read_event().expect("malformed XML") {
603                Event::Start(e) if e.name().as_ref() == b"Piece" => {
604                    piece = (
605                        attr(&e, "NumberOfPoints").parse().unwrap(),
606                        attr(&e, "NumberOfCells").parse().unwrap(),
607                    );
608                }
609                Event::Start(e) if e.name().as_ref() == b"DataArray" => {
610                    assert_eq!(attr(&e, "format"), "binary");
611                    current = Some((attr(&e, "Name"), attr(&e, "type")));
612                }
613                Event::Text(t) => {
614                    if let Some((name, scalar_type)) = current.take() {
615                        let raw = base64_decode(t.decode().unwrap().trim());
616                        let (header, data) = raw.split_at(8);
617                        let size = u64::from_le_bytes(header.try_into().unwrap());
618                        assert_eq!(size as usize, data.len(), "size header of {name}");
619                        let values: Vec<f64> = match scalar_type.as_str() {
620                            "Float64" => data
621                                .chunks_exact(8)
622                                .map(|c| f64::from_le_bytes(c.try_into().unwrap()))
623                                .collect(),
624                            "Int64" => data
625                                .chunks_exact(8)
626                                .map(|c| i64::from_le_bytes(c.try_into().unwrap()) as f64)
627                                .collect(),
628                            "UInt8" => data.iter().map(|&b| f64::from(b)).collect(),
629                            other => panic!("unexpected DataArray type {other}"),
630                        };
631                        arrays.insert(name, values);
632                    }
633                }
634                Event::Eof => break,
635                _ => {}
636            }
637        }
638        (piece, arrays)
639    }

641    /// Décodeur base64 indépendant de `base64_encode`, pour que les tests ne
642    /// valident pas l'encodeur par lui-même.
643    #[cfg(feature = "vtk")]
644    fn base64_decode(text: &str) -> Vec<u8> {
645        let sextet = |c: u8| -> u32 {
646            match c {
647                b'A'..=b'Z' => u32::from(c - b'A'),
648                b'a'..=b'z' => u32::from(c - b'a') + 26,
649                b'0'..=b'9' => u32::from(c - b'0') + 52,
650                b'+' => 62,
651                b'/' => 63,
652                _ => panic!("invalid base64 character {:?}", c as char),
653            }
654        };
655        let bytes = text.as_bytes();
656        assert_eq!(bytes.len() % 4, 0, "base64 length must be a multiple of 4");
657        let mut out = Vec::with_capacity(bytes.len() / 4 * 3);
658        for quad in bytes.chunks(4) {
659            let pad = quad.iter().rev().take_while(|&&c| c == b'=').count();
660            let n = quad[..4 - pad]
661                .iter()
662                .fold(0u32, |acc, &c| acc << 6 | sextet(c))
663                << (6 * pad);
664            out.extend_from_slice(&n.to_be_bytes()[1..4 - pad]);
665        }
666        out
667    }

671    fn base64_encode_matches_rfc4648_vectors() {
672        // RFC 4648 §10.
673        for (input, expected) in [
674            ("", ""),
675            ("f", "Zg=="),
676            ("fo", "Zm8="),
677            ("foo", "Zm9v"),
678            ("foob", "Zm9vYg=="),

680            ("foobar", "Zm9vYmFy"),
681        ] {
682            assert_eq!(super::base64_encode(input.as_bytes()), expected);
683            assert_eq!(base64_decode(expected), input.as_bytes());
684        }
685        let all: Vec<u8> = (0..=255).collect();
686        assert_eq!(base64_decode(&super::base64_encode(&all)), all);
