1//! # Module `solver::snapshot`
2//!
3//! Serialisable snapshot of a simulation's physical state, for post-mortem
4//! divergence analysis and normal pause/resume (DD-025 Option B, issue #71 ;
5//! generalised by DD-029, issue #77).
6//!
7//! ## Two-phase restore contract (DD-025)
8//!
9//! `SimulationSnapshot` captures only the **physical state** (`ContextValue`,
10//! the temporal configuration in effect, and the integration method used) —
11//! never `Scenario`/`Domain`, which hold `Box<dyn Trait>` fields and are not
12//! serialisable. Restoring a run is a two-phase contract: oxiflow supplies
13//! the physical state via this type; the caller reconstructs
14//! `Domain`/`Scenario`/`SolverConfiguration` themselves and re-injects the
15//! state as the initial condition.
16//!
17//! ## Two triggers, one type (DD-029)
18//!
19//! - **Automatic** — [`Solver::on_divergence`](super::Solver::on_divergence)
20//!   is called with a snapshot right before a [`OxiflowError::SolverDivergence`]
21//!   is returned. Default implementation is a no-op; override to persist.
22//! - **Explicit** — normal pause/resume checkpointing. There is no
23//!   `Solver::checkpoint()` method: every field of `SimulationSnapshot` is
24//!   already public, and the live physical state (`ContextValue`, `t`,
25//!   `n_steps`) only exists in the caller's own loop when driving
26//!   [`SteppableSolver::step`](super::methods::SteppableSolver::step)
27//!   directly — `Solver::solve()` is a single blocking call with no
28//!   externally reachable state to hand a trait method. Callers construct
29//!   `SimulationSnapshot` directly instead (see example below).
30//!
31//! # Examples
32//!
33//! ```rust
34//! use oxiflow::solver::snapshot::SimulationSnapshot;
35//! use oxiflow::solver::config::{IntegratorKind, StepControl, TimeConfiguration};
36//! use oxiflow::context::value::ContextValue;
37//! use nalgebra::DVector;
38//!
39//! // Explicit checkpoint, constructed directly by caller code driving its
40//! // own `SteppableSolver::step()` loop — no trait method involved.
41//! let snapshot = SimulationSnapshot {
42//!     time_config: TimeConfiguration::new(600.0, StepControl::Fixed { dt: 0.1 }),
43//!     integrator:  IntegratorKind::BackwardEuler,
44//!     state:       ContextValue::ScalarField(DVector::from_element(10, 0.0)),
45//!     t:           12.3,
46//!     n_steps:     123,
47//!     error:       None,
48//!     partial:     None,
49//! };
50//! assert_eq!(snapshot.t, 12.3);
51//! ```
52

59// ── SimulationSnapshot ──────────────────────────────────────────────────────────
60
61/// Serialisable snapshot of a simulation's physical state.
62///
63/// See the [module docs](self) for the two-phase restore contract and the
64/// two triggers (`on_divergence` automatic, explicit checkpoint constructed
65/// directly).
66///
67/// Deliberately **not** `#[non_exhaustive]`: the whole point of not having a
68/// `Solver::checkpoint()` method is that callers construct this struct
69/// directly via its public fields (see module docs) — `#[non_exhaustive]`
70/// would block exactly that from outside this crate.
71#[derive(Debug, Clone)]
72#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
73pub struct SimulationSnapshot {
74    /// Serialisable temporal configuration in effect when the snapshot was
75    /// taken.
76    pub time_config: TimeConfiguration,
77    /// Integration method in effect when the snapshot was taken.
78    pub integrator: IntegratorKind,
79    /// Physical state at snapshot time.
80    pub state: ContextValue,
81    /// Simulation time at snapshot.
82    pub t: f64,
83    /// Number of time steps taken up to the snapshot.
84    pub n_steps: usize,
85    /// Populated on failure — error message for post-mortem analysis.
86    /// `None` for a normal explicit checkpoint.
87    pub error: Option<String>,
88    /// Partial result up to the snapshot point, if available.
89    pub partial: Option<SimulationResult>,
90}

92// ── File format ──────────────────────────────────────────────────────────────
93
94/// On-disk format for [`write_snapshot`] / [`read_snapshot`].
95#[non_exhaustive]
96#[derive(Debug, Clone, Copy, PartialEq, Eq)]
97pub enum SnapshotFormat {
98    /// Human-readable JSON (`serde_json`) — intended for debugging.
99    Json,
100    /// Native binary format (`bincode` 1.x) — fidelity and speed, the
101    /// recommended default for `on_divergence`/checkpoint round-trips.
102    Bincode,
103}
104
105/// Writes `snapshot` to `path` in the given [`SnapshotFormat`].
106///
107/// Requires the `serde` feature.
108///
109/// # Errors
110///
111/// Returns [`OxiflowError::Persistence`] if the file cannot be created or
112/// written, or if serialisation fails.
113#[cfg(feature = "serde")]
114pub fn write_snapshot(
115    snapshot: &SimulationSnapshot,
116    path: &std::path::Path,
117    format: SnapshotFormat,
118) -> Result<(), OxiflowError> {
119    match format {
120        SnapshotFormat::Json => {
121            let file = std::fs::File::create(path)
122                .map_err(|e| OxiflowError::Persistence(format!("cannot create {path:?}: {e}")))?;
123            serde_json::to_writer_pretty(file, snapshot)
124                .map_err(|e| OxiflowError::Persistence(format!("JSON serialisation failed: {e}")))
125        }
126        SnapshotFormat::Bincode => {
127            let bytes = bincode::serialize(snapshot).map_err(|e| {
128                OxiflowError::Persistence(format!("bincode serialisation failed: {e}"))
129            })?;
130            std::fs::write(path, bytes)
131                .map_err(|e| OxiflowError::Persistence(format!("cannot write {path:?}: {e}")))
132        }
133    }
134}

136/// Reads a [`SimulationSnapshot`] previously written by [`write_snapshot`].
137///
138/// Requires the `serde` feature.
139///
140/// # Errors
141///
142/// Returns [`OxiflowError::Persistence`] if the file cannot be read or if
143/// deserialisation fails.
144#[cfg(feature = "serde")]
145pub fn read_snapshot(
146    path: &std::path::Path,
147    format: SnapshotFormat,
148) -> Result<SimulationSnapshot, OxiflowError> {
149    match format {
150        SnapshotFormat::Json => {
151            let file = std::fs::File::open(path)
152                .map_err(|e| OxiflowError::Persistence(format!("cannot open {path:?}: {e}")))?;
153            serde_json::from_reader(file)
154                .map_err(|e| OxiflowError::Persistence(format!("JSON deserialisation failed: {e}")))
155        }
156        SnapshotFormat::Bincode => {
157            let bytes = std::fs::read(path)
158                .map_err(|e| OxiflowError::Persistence(format!("cannot read {path:?}: {e}")))?;
159            bincode::deserialize(&bytes).map_err(|e| {
160                OxiflowError::Persistence(format!("bincode deserialisation failed: {e}"))
161            })
162        }
163    }
164}

168#[cfg(test)]
169mod tests {
170    use super::*;
171    use crate::solver::config::StepControl;
172    use nalgebra::DVector;
173
174    fn sample_snapshot() -> SimulationSnapshot {
175        SimulationSnapshot {
176            time_config: TimeConfiguration::new(10.0, StepControl::Fixed { dt: 0.1 }),
177            integrator: IntegratorKind::BackwardEuler,
178            state: ContextValue::ScalarField(DVector::from_element(5, 1.5)),
179            t: 3.4,
180            n_steps: 34,
181            error: Some("non-finite value detected in state vector".to_string()),
182            partial: None,
183        }
184    }
185
186    #[test]
187    fn snapshot_fields_accessible() {
188        let s = sample_snapshot();
189        assert_eq!(s.t, 3.4);
190        assert_eq!(s.n_steps, 34);
191        assert!(s.error.is_some());
192    }
193
194    #[cfg(feature = "serde")]
195    #[test]
196    fn json_round_trip() {
197        let dir = tempdir();
198        let path = dir.join("snapshot.json");
199        let original = sample_snapshot();
200
201        write_snapshot(&original, &path, SnapshotFormat::Json).unwrap();
202        let restored = read_snapshot(&path, SnapshotFormat::Json).unwrap();
203
204        assert_eq!(restored.t, original.t);
205        assert_eq!(restored.n_steps, original.n_steps);
206        assert_eq!(restored.error, original.error);
207        std::fs::remove_file(&path).ok();
208    }
209
210    #[cfg(feature = "serde")]
211    #[test]
212    fn bincode_round_trip() {
213        let dir = tempdir();
214        let path = dir.join("snapshot.bin");
215        let original = sample_snapshot();
216
217        write_snapshot(&original, &path, SnapshotFormat::Bincode).unwrap();
218        let restored = read_snapshot(&path, SnapshotFormat::Bincode).unwrap();
219
220        assert_eq!(restored.t, original.t);
221        assert_eq!(restored.n_steps, original.n_steps);
222        assert_eq!(restored.error, original.error);
223        std::fs::remove_file(&path).ok();
224    }
225
226    #[cfg(feature = "serde")]
227    #[test]
228    fn read_snapshot_missing_file_is_persistence_error() {
229        let path = std::path::PathBuf::from("/nonexistent/path/snapshot.json");
230        let err = read_snapshot(&path, SnapshotFormat::Json).unwrap_err();
231        assert!(matches!(err, OxiflowError::Persistence(_)));
232    }
233
234    /// Returns the system temp dir — no external crate needed for this.
235    #[cfg(feature = "serde")]
236    fn tempdir() -> std::path::PathBuf {
237        std::env::temp_dir()
238    }
239}
