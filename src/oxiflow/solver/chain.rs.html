1//! # Module `solver::chain`
2//!
3//! Calculator chain — validation and ordered execution (issues [#33](https://github.com/biface/oxiflow/issues/33), [#36](https://github.com/biface/oxiflow/issues/36), [DD-009](https://github.com/biface/oxiflow/issues/9)).
4//!
5//! ## Responsibility
6//!
7//! `build_calculator_chain` verifies that every context variable required by the
8//! scenario has a corresponding calculator in `SolverConfiguration`, then returns
9//! the calculators in a correct execution order.
10//!
11//! ## Ordering strategy ([DD-009](https://github.com/biface/oxiflow/issues/9))
12//!
13//! A hybrid two-path algorithm is selected at chain-build time:
14//!
15//! **Priority path (fast path)** — no calculator declares a non-empty `depends_on()`.
16//! Calculators are sorted by ascending `priority()` (stable sort, O(n log n)).
17//! A total order by priority cannot contain a cycle. Priority ranges:
18//! - **0–49**: values that read nothing but the state or the time: the
19//!   system variables (`Time`, `TimeStep`, injected directly by the solver,
20//!   or provided by `TimeCalculator`/`TimeStepCalculator` at 0) and the
21//!   finite-difference derivatives of the state (`FDGradientCalculator`,
22//!   `FDLaplacianCalculator`, at 10)
23//! - **50–99**: external data providers (`ExternalTabulated`,
24//!   `CoefficientField`)
25//! - **100+**: derived quantities (default; `TrapezoidalIntegral`)
26//!
27//! **Topological path (Kahn)** — at least one calculator declares a non-empty
28//! `depends_on()`. A directed acyclic graph is built from the declared dependencies:
29//! if calculator C declares `depends_on: [X]`, an edge is drawn from every calculator
30//! providing X to C. Within each topological tier, calculators are ordered by
31//! ascending `priority()`. A cycle returns
32//! [`OxiflowError::CircularDependency`].
33//!
34//! ## Built-in variables
35//!
36//! `Time` and `TimeStep` are always available in `ComputeContext` — the solver
37//! injects them directly via `ComputeContext::new(t, dt)` before running the
38//! chain. Built-in variables declared in `depends_on()` are ignored when building
39//! the dependency graph.
40

45/// Built-in variables provided directly by the solver — no calculator needed.
46///
47/// `Time` and `TimeStep` are injected via `ComputeContext::new(t, dt)` before
48/// the calculator chain runs. Declaring them as requirements is valid; checking
49/// them against the calculator list would be a false negative. Declaring them in
50/// `depends_on()` is valid and ignored when building the dependency graph.
51const BUILTIN_VARIABLES: &[ContextVariable] = &[ContextVariable::Time, ContextVariable::TimeStep];
52
53/// Validates requirements against provided calculators and returns an
54/// execution-ordered chain.
55///
56/// # Validation
57///
58/// Every variable in `requirements` must be either:
59/// - a built-in variable (`Time`, `TimeStep`), or
60/// - covered by at least one calculator via `provides()`.
61///
62/// A variable covered by multiple calculators is accepted — the last one
63/// to execute wins (consistent with `ComputeContext::insert` overwrite).
64///
65/// # Ordering ([DD-009](https://github.com/biface/oxiflow/issues/9))
66///
67/// If no calculator declares a non-empty `depends_on()`, calculators are sorted
68/// by ascending `priority()` (stable sort — registration order preserved within
69/// equal priorities).
70///
71/// If at least one calculator declares a non-empty `depends_on()`, Kahn's
72/// algorithm is applied on the full dependency graph. Within each topological
73/// tier, calculators are sorted by ascending `priority()`. Built-in variables
74/// in `depends_on()` declarations are ignored when building the graph.
75///
76/// # Errors
77///
78/// - [`OxiflowError::MissingCalculator`] — first uncovered non-builtin requirement.
79/// - [`OxiflowError::CircularDependency`] — cycle detected in the dependency graph
80///   (Kahn path only).
81///
82/// # Examples
83///
84/// ```rust
85/// use oxiflow::solver::chain::build_calculator_chain;
86/// use oxiflow::context::variable::ContextVariable;
87/// use oxiflow::context::value::ContextValue;
88/// use oxiflow::context::compute::ComputeContext;
89/// use oxiflow::context::error::OxiflowError;
90/// use oxiflow::context::calculator::ContextCalculator;
91/// use oxiflow::model::traits::RequiresContext;
92///
93/// #[derive(Debug)]
94/// struct TimeCalc;
95/// impl RequiresContext for TimeCalc {
96///     fn required_variables(&self) -> Vec<ContextVariable> { vec![] }
97///     fn priority(&self) -> u32 { 0 }
98/// }
99/// impl ContextCalculator for TimeCalc {
100///     fn provides(&self) -> ContextVariable { ContextVariable::Time }
101///     fn compute(&self, _: &ContextValue, ctx: &ComputeContext)
102///         -> Result<ContextValue, OxiflowError>
103///     { Ok(ContextValue::Scalar(ctx.time())) }
104/// }
105///
106/// let requirements = vec![ContextVariable::Time];
107/// let calculators: Vec<Box<dyn ContextCalculator>> = vec![Box::new(TimeCalc)];
108/// let chain = build_calculator_chain(&requirements, &calculators).unwrap();
109/// assert_eq!(chain.len(), 1);
110/// ```
111pub fn build_calculator_chain<'a>(
112    requirements: &[ContextVariable],
113    calculators: &'a [Box<dyn ContextCalculator>],
114) -> Result<Vec<&'a dyn ContextCalculator>, OxiflowError> {
115    // Validate: every non-builtin requirement must have a provider.
116    for req in requirements {
117        if is_builtin(req) {
118            continue;
119        }
120        let covered = calculators.iter().any(|c| &c.provides() == req);
121        if !covered {
122            return Err(OxiflowError::MissingCalculator(req.clone()));
123        }
124    }
125
126    // Select ordering path based on whether any calculator declares dependencies.
127    let has_deps = calculators.iter().any(|c| !c.depends_on().is_empty());
128    if has_deps {
129        build_kahn_chain(calculators)
130    } else {
131        build_priority_chain(calculators)
132    }
133}
134
135/// Fast path — no `depends_on()` declared. Stable sort by ascending priority.
136fn build_priority_chain(
137    calculators: &[Box<dyn ContextCalculator>],
138) -> Result<Vec<&dyn ContextCalculator>, OxiflowError> {
139    let mut chain: Vec<&dyn ContextCalculator> = calculators.iter().map(|c| c.as_ref()).collect();
140    chain.sort_by_key(|c| c.priority());
141    Ok(chain)
142}
143
144/// Kahn path — topological sort with priority as tiebreaker within each tier.
145///
146/// Edges: for each calculator C declaring `depends_on: [X]`, draw an edge from
147/// every calculator providing X to C. Built-in variables are excluded from the
148/// graph. A cycle returns `CircularDependency` naming the first `depends_on`
149/// variable of the first blocked calculator found.
150fn build_kahn_chain(
151    calculators: &[Box<dyn ContextCalculator>],
152) -> Result<Vec<&dyn ContextCalculator>, OxiflowError> {
153    let n = calculators.len();
154
155    // Build adjacency list and in-degree table.
156    // successors[j] = indices of calculators that must run after j.
157    let mut successors: Vec<Vec<usize>> = vec![vec![]; n];
158    let mut in_degree: Vec<usize> = vec![0; n];
159
160    for (i, calc) in calculators.iter().enumerate() {
161        for dep_var in calc.depends_on() {
162            if is_builtin(&dep_var) {
163                continue;
164            }
165            for (j, provider) in calculators.iter().enumerate() {
166                if provider.provides() == dep_var {
167                    successors[j].push(i);
168                    in_degree[i] += 1;
169                }
170            }
171        }
172    }
173
174    // Initial queue: calculators with no unresolved predecessors, by priority.
175    let mut queue: Vec<usize> = (0..n).filter(|&i| in_degree[i] == 0).collect();
176    queue.sort_by_key(|&i| calculators[i].priority());
177
178    let mut result: Vec<&dyn ContextCalculator> = Vec::with_capacity(n);
179
180    while !queue.is_empty() {
181        // Emit the first element (lowest priority in current tier).
182        let i = queue.remove(0);
183        result.push(calculators[i].as_ref());
184
185        // Decrement in-degree of successors; collect newly unblocked nodes.
186        let mut newly_free: Vec<usize> = Vec::new();
187        for &j in &successors[i] {
188            in_degree[j] -= 1;
189            if in_degree[j] == 0 {
190                newly_free.push(j);
191            }
192        }
193
194        // Merge newly unblocked nodes into the queue, maintaining priority order.
195        queue.extend(newly_free);
196        queue.sort_by_key(|&i| calculators[i].priority());
197    }
198
199    // If not all calculators were emitted, a cycle exists.
200    if result.len() < n {
201        let blocked = (0..n).find(|&i| in_degree[i] > 0).unwrap();
202        let var = calculators[blocked]
203            .depends_on()
204            .into_iter()
205            .find(|v| !is_builtin(v))
206            .unwrap_or_else(|| calculators[blocked].provides());
207        return Err(OxiflowError::CircularDependency(var));
208    }
209
210    Ok(result)
211}
212
213fn is_builtin(var: &ContextVariable) -> bool {
214    BUILTIN_VARIABLES.contains(var)
215}

405        assert!(middle.contains(&var("Y")));

426        // A and B both provide X; C depends on X.
427        // Both A and B must appear before C.
428        let calcs = vec![

434        let c_pos = chain.iter().position(|c| c.provides() == var("Z")).unwrap();
435        let a_pos = chain

442            .unwrap();
443        assert!(a_pos < c_pos);
444        assert!(b_pos < c_pos);
445    }

485        // Calculators without depends_on coexist with one that has deps.
486        // Independent ones (no deps) run by priority; the dependent one after its provider.
487        let calcs = vec![

493        // gamma (priority 50) and alpha (priority 200) have no deps on each other.
494        // beta depends on alpha → alpha must precede beta.

496            .iter()
497            .position(|c| c.provides() == var("alpha"))
498            .unwrap();
499        let beta_pos = chain
500            .iter()
501            .position(|c| c.provides() == var("beta"))
502            .unwrap();
503        assert!(alpha_pos < beta_pos);
504        // gamma has no deps → should appear first (priority 50 < 200).
505        assert_eq!(chain[0].provides(), var("gamma"));
