1//! # Module `model::discretized`
2//!
3//! `DiscretizedModel<Op>` — wraps a [`FluxDivergenceOperator`] scheme (FV,
4//! WENO) as an ordinary [`PhysicalModel`], returning only the `-∇·F(u, ∇u)`
5//! contribution of the canonical form (DD-038, #93).
6//!
7//! ## Why
8//!
9//! FD (gradient/Laplacian, DD-012) is consumed transparently through
10//! [`crate::context::ContextCalculator`] — a model declares a required
11//! context variable and never knows which scheme produced it. FV (#48) and
12//! WENO/limiters (#49) are different: they compute the complete `∇·F` term
13//! directly — exactly what `PhysicalModel::compute_physics` must return.
14//! `DiscretizedModel<Op>` is the insertion point for that case: a spatial
15//! composite, cousin of
16//! [`crate::solver::methods::imex::SplitOperator`] (DD-037, temporal
17//! composition) but for wiring a single spatial scheme into the engine.
18//!
19//! A trait `SourceTerm` mirroring `DiscreteOperator`'s `S(u, x, t)` was
20//! considered and rejected (DD-038, amendment 1) — nothing in the v0.5.0
21//! scope (#48, #49) needs a source term richer than an ordinary
22//! `PhysicalModel` already expresses. When a source term exists, it composes
23//! with `DiscretizedModel<Op>` through
24//! [`crate::model::CompositeModel`] (DD-037) — already delivered, already
25//! tested, no new trait:
26//!
27//! ```rust,ignore
28//! let full_model = CompositeModel::new(
29//!     vec![
30//!         Box::new(DiscretizedModel::new(scheme, mesh, "advection")),
31//!         Box::new(my_source_model), // ordinary PhysicalModel, S(u, x, t)
32//!     ],
33//!     "advection_reaction",
34//! )?;
35//! ```
36//!
37//! `Op` is bound to [`FluxDivergenceOperator`] (DD-039), not
38//! `DiscreteOperator` (DD-038, amendment 2) — an FD scheme cannot be plugged
39//! in here; that is a compile error, not a silent runtime bug.

54/// Negates a [`ContextValue`] element-wise.
55///
56/// `Boolean` does not support negation as a numeric operation.
57fn negate_context_value(value: ContextValue) -> Result<ContextValue, OxiflowError> {
58    use ContextValue::*;
59    match value {
60        Scalar(x) => Ok(Scalar(-x)),
61        Vector(x) => Ok(Vector(-x)),
62        Matrix(x) => Ok(Matrix(-x)),
63        ScalarField(x) => Ok(ScalarField(-x)),
64        VectorField(x) => Ok(VectorField(-x)),
65        other => Err(OxiflowError::TypeMismatch {
66            expected: "negatable ContextValue (Scalar/Vector/Matrix/ScalarField/VectorField)",
67            actual: other.variant_name(),
68        }),
69    }
70}

72/// Computes the flux via `scheme` — not yet registered in any
73/// `ContextCalculator` chain (DD-038: private posture). `instance_id` is
74/// reserved, unused today, to allow multi-instance naming (DD-031, DD-037)
75/// the day DD-027 requires publication into `ComputeContext` (VTK/HDF5
76/// export, v0.6.0). No concrete need justifies publishing it sooner.
77#[allow(dead_code)]
78struct FluxDivergenceCalculator<Op: FluxDivergenceOperator> {
79    scheme: Arc<Op>,
80    mesh: Arc<Op::MeshType>,
81    instance_id: Cow<'static, str>,
82}

84/// Wraps a spatial discretization scheme as an ordinary [`PhysicalModel`] —
85/// the `-∇·F(u, ∇u)` contribution of the canonical form, nothing else.
86/// Cousin of [`crate::solver::methods::imex::SplitOperator`] (DD-037), but
87/// spatial rather than temporal.
88///
89/// See the module documentation for how to compose a source term via
90/// [`crate::model::CompositeModel`].
91pub struct DiscretizedModel<Op: FluxDivergenceOperator> {
92    scheme: Arc<Op>,
93    mesh: Arc<Op::MeshType>,
94    // Not read yet — reserved for DD-027 (VTK/HDF5 export, v0.6.0). See
95    // FluxDivergenceCalculator's doc comment for why publication is deferred.
96    #[allow(dead_code)]
97    flux_calculator: FluxDivergenceCalculator<Op>,
98    name: String,
99}
100
101impl<Op: FluxDivergenceOperator> DiscretizedModel<Op> {
102    /// Wraps `scheme` evaluated on `mesh` as a `PhysicalModel` named `name`.
103    pub fn new(scheme: Arc<Op>, mesh: Arc<Op::MeshType>, name: impl Into<String>) -> Self {
104        let name = name.into();
105        let flux_calculator = FluxDivergenceCalculator {
106            scheme: Arc::clone(&scheme),
107            mesh: Arc::clone(&mesh),
108            instance_id: Cow::Owned(name.clone()),
109        };
110        Self {
111            scheme,
112            mesh,
113            flux_calculator,
114            name,
115        }
116    }
117}

119impl<Op: FluxDivergenceOperator> RequiresContext for DiscretizedModel<Op> {

126    }

130    }

134    }

137impl<Op: FluxDivergenceOperator> PhysicalModel for DiscretizedModel<Op> {
138    fn compute_physics(
139        &self,
140        state: &ContextValue,
141        ctx: &ComputeContext,
142    ) -> Result<ContextValue, OxiflowError> {
143        negate_context_value(self.scheme.apply(state, &self.mesh, ctx)?)
144    }
145
146    fn initial_state(&self, mesh: &dyn Mesh) -> ContextValue {
147        // DiscretizedModel represents the flux term only, not the physical
148        // initial condition — delegates to the mesh for sizing, zero
149        // elsewhere. The real initial condition comes from whichever
150        // PhysicalModel composed alongside it (CompositeModel::initial_state
151        // uses its first operand).
152        ContextValue::ScalarField(DVector::from_element(mesh.n_dof(), 0.0))
153    }
154
155    fn name(&self) -> &str {
156        &self.name
157    }
158
159    fn description(&self) -> Option<&str> {
160        Some("discretized model — wraps a FluxDivergenceOperator as -∇·F(u, ∇u) (DD-038)")
161    }
162}

166#[cfg(test)]
167mod tests {
168    use super::*;
169    use crate::mesh::structured::UniformGrid1D;
170
171    /// Minimal `FluxDivergenceOperator` fixture: returns a constant field,
172    /// declares one required variable to exercise delegation.
173    struct ConstantFlux(f64);
174
175    impl RequiresContext for ConstantFlux {
176        fn required_variables(&self) -> Vec<ContextVariable> {
177            vec![ContextVariable::Time]
178        }
179    }
180
181    impl FluxDivergenceOperator for ConstantFlux {
182        type MeshType = UniformGrid1D;
183
184        fn apply(
185            &self,
186            field: &ContextValue,
187            _mesh: &Self::MeshType,
188            _ctx: &ComputeContext,
189        ) -> Result<ContextValue, OxiflowError> {
190            let n = field.as_scalar_field()?.len();
191            Ok(ContextValue::ScalarField(DVector::from_element(n, self.0)))
192        }
193
194        fn stencil_radius(&self) -> usize {
195            // Constant output, no neighbor access at all.
196            0
197        }
198    }
199
200    fn fixture(value: f64) -> DiscretizedModel<ConstantFlux> {
201        let mesh = Arc::new(UniformGrid1D::new(4, 0.0, 1.0).unwrap());
202        DiscretizedModel::new(Arc::new(ConstantFlux(value)), mesh, "constant_flux_test")
203    }
204
205    #[test]
206    fn compute_physics_negates_scheme_output() {
207        let model = fixture(3.0);
208        let state = ContextValue::ScalarField(DVector::from_element(4, 0.0));
209        let ctx = ComputeContext::new(0.0, 0.01);
210        let result = model.compute_physics(&state, &ctx).unwrap();
211        assert!(result
212            .as_scalar_field()
213            .unwrap()
214            .iter()
215            .all(|&v| (v + 3.0).abs() < 1e-12));
216    }
217
218    #[test]
219    fn required_variables_delegates_to_scheme() {
220        let model = fixture(1.0);
221        assert_eq!(model.required_variables(), vec![ContextVariable::Time]);
222    }
223
224    #[test]
225    fn initial_state_is_zero_sized_from_mesh() {
226        let model = fixture(1.0);
227        let mesh = UniformGrid1D::new(6, 0.0, 1.0).unwrap();
228        let state = model.initial_state(&mesh);
229        let field = state.as_scalar_field().unwrap();
230        assert_eq!(field.len(), 6);
231        assert!(field.iter().all(|&v| v == 0.0));
232    }
233
234    #[test]
235    fn name_returns_identifier() {
236        let model = fixture(1.0);
237        assert_eq!(model.name(), "constant_flux_test");
238    }
239
240    #[test]
241    fn negate_context_value_rejects_boolean() {
242        let err = negate_context_value(ContextValue::Boolean(true)).unwrap_err();
243        assert!(matches!(err, OxiflowError::TypeMismatch { .. }));
244    }
245}
