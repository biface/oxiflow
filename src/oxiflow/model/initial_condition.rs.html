1//! # Module `model::initial_condition`
2//!
3//! `PrescribedInitialState` — any [`PhysicalModel`] started from a given state
4//! instead of its own initial condition.
5//!
6//! ## Why
7//!
8//! A model's [`PhysicalModel::initial_state`] is fixed by the model itself.
9//! Two situations need another starting point without changing the model:
10//!
11//! 1. **Resuming a run** from a
12//!    [`SimulationSnapshot`](crate::solver::snapshot::SimulationSnapshot): the
13//!    restore contract of [DT-025](https://github.com/biface/oxiflow/issues/66)
14//!    leaves it to the caller to rebuild the scenario and re-inject the saved
15//!    state as the initial condition. Wrapping the model is that re-injection;
16//!    [`Scenario::with_t_start`](crate::solver::scenario::Scenario::with_t_start)
17//!    sets the matching start time.
18//! 2. **Starting from a state computed elsewhere**: a developed flow, a
19//!    perturbed equilibrium, the end state of another run.
20//!
21//! The wrapper changes the initial condition and nothing else: every other
22//! [`RequiresContext`] and [`PhysicalModel`] method is delegated, `name()`
23//! included, so the result of a run keeps the wrapped model's name.
24//!
25//! ## What a resume reproduces
26//!
27//! A run resumed at step N from a snapshot and continued for M steps matches
28//! an uninterrupted N + M run under two conditions:
29//!
30//! - **One-step integrators only.** A snapshot carries no history, so a
31//!   multistep integrator (BDF2) restarts on its bootstrap step and the two
32//!   runs part from there.
33//! - **Times agree to rounding, not bit for bit.** The fixed-step loop
34//!   computes each time as `t_start + k·dt`. Resumed from `t_start = t_N`, it
35//!   computes `t_N + k·dt` where the uninterrupted run computes
36//!   `(N + k)·dt`; the two can differ in the last bits. States are identical
37//!   for a model that does not read the time; a model that does sees those
38//!   last-bit differences.
39//!
40//! ## Shape
41//!
42//! [`initial_state`](PhysicalModel::initial_state) cannot report an error, so
43//! the prescribed state is checked against the mesh by
44//! [`Scenario::validate`](crate::solver::scenario::Scenario::validate), which
45//! [`Solver::solve`](crate::solver::Solver::solve) and the orchestrator call
46//! before the first step. A caller that drives
47//! [`SteppableSolver::step`](crate::solver::methods::SteppableSolver::step)
48//! in its own loop calls it itself.
49//!
50//! # Examples
51//!
52//! Resuming a run from a snapshot taken at `t = 0.5`:
53//!
54//! ```rust
55//! use nalgebra::DVector;
56//! use oxiflow::context::{ComputeContext, ContextValue, ContextVariable, OxiflowError};
57//! use oxiflow::mesh::{Mesh, UniformGrid1D};
58//! use oxiflow::model::{PhysicalModel, PrescribedInitialState, RequiresContext};
59//! use oxiflow::solver::config::{IntegratorKind, StepControl, TimeConfiguration};
60//! use oxiflow::solver::methods::ForwardEulerSolver;
61//! use oxiflow::solver::scenario::Scenario;
62//! use oxiflow::solver::snapshot::SimulationSnapshot;
63//! use oxiflow::solver::{Solver, SolverConfiguration};
64//!
65//! struct Decay;
66//! impl RequiresContext for Decay {
67//!     fn required_variables(&self) -> Vec<ContextVariable> { vec![] }
68//! }
69//! impl PhysicalModel for Decay {
70//!     fn compute_physics(&self, s: &ContextValue, _: &ComputeContext)
71//!         -> Result<ContextValue, OxiflowError> {
72//!         Ok(ContextValue::ScalarField(s.as_scalar_field()?.map(|v| -v)))
73//!     }
74//!     fn initial_state(&self, mesh: &dyn Mesh) -> ContextValue {
75//!         ContextValue::ScalarField(DVector::from_element(mesh.n_dof(), 1.0))
76//!     }
77//!     fn name(&self) -> &str { "decay" }
78//! }
79//!
80//! let time = TimeConfiguration::new(0.5, StepControl::Fixed { dt: 0.125 });
81//! let first = Scenario::single(Box::new(Decay), Box::new(UniformGrid1D::new(4, 0.0, 1.0)?));
82//! let run = ForwardEulerSolver.solve(&first, &SolverConfiguration::new(time.clone(), IntegratorKind::Euler))?;
83//!
84//! // Explicit checkpoint at the end of the first run.
85//! let snapshot = SimulationSnapshot {
86//!     time_config: time,
87//!     integrator: IntegratorKind::Euler,
88//!     state: run.states.last().unwrap().clone(),
89//!     t: run.t_final().unwrap(),
90//!     n_steps: run.n_steps,
91//!     error: None,
92//!     partial: None,
93//! };
94//!
95//! // Rebuild the scenario, re-inject the state, restart from its time.
96//! let model = PrescribedInitialState::new(Box::new(Decay), snapshot.state.clone());
97//! let resumed = Scenario::single(Box::new(model), Box::new(UniformGrid1D::new(4, 0.0, 1.0)?))
98//!     .with_t_start(snapshot.t);
99//! let config = SolverConfiguration::new(
100//!     TimeConfiguration::new(1.0, StepControl::Fixed { dt: 0.125 }),
101//!     IntegratorKind::Euler,
102//! );
103//! let rest = ForwardEulerSolver.solve(&resumed, &config)?;
104//! assert_eq!(rest.states[0], snapshot.state);
105//! assert_eq!(rest.t_final(), Some(1.0));
106//! # Ok::<(), OxiflowError>(())
107//! ```

109use crate::context::compute::ComputeContext;
110use crate::context::error::OxiflowError;
111use crate::context::value::ContextValue;
112use crate::context::variable::ContextVariable;
113use crate::mesh::Mesh;
114use crate::model::traits::{PhysicalModel, RequiresContext};
115
116/// A [`PhysicalModel`] whose initial condition is replaced by a given state.
117///
118/// See the [module documentation](self) for the restore contract it serves
119/// and what a resumed run reproduces.
120pub struct PrescribedInitialState {
121    model: Box<dyn PhysicalModel>,
122    state: ContextValue,
123}
124
125impl PrescribedInitialState {
126    /// Wraps `model` so that its initial condition is `state`.
127    ///
128    /// `state` is not checked here: the mesh is not known yet.
129    /// [`Scenario::validate`](crate::solver::scenario::Scenario::validate)
130    /// checks it against the domain's mesh (see the
131    /// [module documentation](self#shape) for who calls it).
132    pub fn new(model: Box<dyn PhysicalModel>, state: ContextValue) -> Self {
133        Self { model, state }
134    }
135
136    /// The wrapped model.
137    pub fn model(&self) -> &dyn PhysicalModel {
138        self.model.as_ref()
139    }
140
141    /// The prescribed initial state.
142    pub fn state(&self) -> &ContextValue {
143        &self.state
144    }
145}
146
147impl RequiresContext for PrescribedInitialState {
148    fn required_variables(&self) -> Vec<ContextVariable> {
149        self.model.required_variables()
150    }
151
152    fn optional_variables(&self) -> Vec<ContextVariable> {
153        self.model.optional_variables()
154    }
155
156    fn depends_on(&self) -> Vec<ContextVariable> {
157        self.model.depends_on()
158    }
159
160    fn priority(&self) -> u32 {
161        self.model.priority()
162    }
163}
164
165impl PhysicalModel for PrescribedInitialState {
166    fn compute_physics(
167        &self,
168        state: &ContextValue,
169        ctx: &ComputeContext,
170    ) -> Result<ContextValue, OxiflowError> {
171        self.model.compute_physics(state, ctx)
172    }
173
174    /// Returns the prescribed state, whatever the mesh: its size is checked
175    /// by `Scenario::validate`.
176    fn initial_state(&self, _mesh: &dyn Mesh) -> ContextValue {
177        self.state.clone()
178    }
179
180    fn name(&self) -> &str {
181        self.model.name()
182    }
183
184    fn description(&self) -> Option<&str> {
185        self.model.description()
186    }
187
188    fn max_wave_speed(
189        &self,
190        state: &ContextValue,
191        ctx: &ComputeContext,
192    ) -> Result<Option<f64>, OxiflowError> {
193        self.model.max_wave_speed(state, ctx)
194    }
195}

199#[cfg(test)]
200mod tests {
201    use super::*;
202    use crate::mesh::structured::UniformGrid1D;
203    use crate::model::FullModel;
204    use nalgebra::{DMatrix, DVector};
205
206    fn prescribed() -> PrescribedInitialState {
207        PrescribedInitialState::new(
208            Box::new(FullModel),
209            ContextValue::ScalarField(DVector::from_vec(vec![3.0, 4.0, 5.0])),
210        )
211    }
212
213    #[test]
214    fn context_requirements_are_those_of_the_wrapped_model() {
215        let model = prescribed();
216        assert_eq!(model.required_variables(), FullModel.required_variables());
217        assert_eq!(model.optional_variables(), FullModel.optional_variables());
218        assert_eq!(model.depends_on(), FullModel.depends_on());
219        assert_eq!(model.priority(), FullModel.priority());
220    }
221
222    #[test]
223    fn name_and_description_are_those_of_the_wrapped_model() {
224        let model = prescribed();
225        assert_eq!(model.name(), FullModel.name());
226        assert_eq!(model.description(), FullModel.description());
227    }
228
229    #[test]
230    fn wave_speed_is_that_of_the_wrapped_model() {
231        let ctx = ComputeContext::new(0.0, 0.01);
232        let state = ContextValue::ScalarField(DVector::from_vec(vec![1.0, 2.0, 3.0]));
233        let declared = PrescribedInitialState::new(
234            Box::new(crate::model::DeclaredWaveSpeed { speed: 3.0 }),
235            state.clone(),
236        );
237        assert_eq!(declared.max_wave_speed(&state, &ctx).unwrap(), Some(3.0));
238        assert_eq!(prescribed().max_wave_speed(&state, &ctx).unwrap(), None);
239    }
240
241    #[test]
242    fn physics_is_that_of_the_wrapped_model() {
243        let ctx = ComputeContext::new(0.0, 0.01);
244        let state = ContextValue::ScalarField(DVector::from_vec(vec![1.0, 2.0, 3.0]));
245        assert_eq!(
246            prescribed().compute_physics(&state, &ctx).unwrap(),
247            FullModel.compute_physics(&state, &ctx).unwrap()
248        );
249    }
250
251    #[test]
252    fn initial_state_is_the_prescribed_state_not_the_model_one() {
253        let mesh = UniformGrid1D::new(3, 0.0, 1.0).unwrap();
254        let model = prescribed();
255        assert_eq!(model.initial_state(&mesh), *model.state());
256        assert_ne!(model.initial_state(&mesh), FullModel.initial_state(&mesh));
257    }
258
259    #[test]
260    fn a_vector_field_state_is_returned_unchanged() {
261        let mesh = UniformGrid1D::new(2, 0.0, 1.0).unwrap();
262        let state = ContextValue::VectorField(DMatrix::from_row_slice(2, 2, &[1., 2., 3., 4.]));
263        let model = PrescribedInitialState::new(Box::new(FullModel), state.clone());
264        assert_eq!(model.initial_state(&mesh), state);
265        assert_eq!(model.model().name(), FullModel.name());
266    }
267}
