1//! # Module `boundary::spec`
2//!
3//! External, serialisable description of boundary conditions — the boundary
4//! part of the WHAT-axis configuration
5//! ([DT-027](https://github.com/biface/oxiflow/issues/75) amendment 1,
6//! [#108](https://github.com/biface/oxiflow/issues/108)).
7//!
8//! [`BoundaryConditionSpec`] names one of the conditions of this module and
9//! its parameters; [`TimeFunctionSpec`] describes a value that depends on
10//! time. Both are consumed by [`DomainSpec`](crate::solver::DomainSpec).
11//!
12//! A function of time held as a Rust closure, as
13//! [`Dirichlet::of_t`](super::Dirichlet::of_t) takes, has no serialisable
14//! form: its type gives its signature, not the function. [`TimeFunctionSpec`]
15//! describes the function instead (a constant, a parametric profile, or a
16//! tabulated series), and the closure is rebuilt from that description when
17//! the domain is built. Closed-form expressions of time are planned by
18//! [DD-072](https://github.com/biface/oxiflow/issues/215).
19

28/// A value that depends on time.
29///
30/// With the `serde` feature:
31///
32/// ```text
33/// value: { Constant: 1.0 }
34/// value: { Profile: { Gaussian: { center: 5.0, width: 1.0, peak_concentration: 2.0 } } }
35/// value: { Tabulated: { points: [[0.0, 0.0], [10.0, 1.0]], interpolation: Linear } }
36/// ```
37#[non_exhaustive]
38#[derive(Debug, Clone, PartialEq)]
39#[cfg_attr(feature = "serde", derive(serde::Deserialize))]
40pub enum TimeFunctionSpec {
41    /// The same value at every time.
42    Constant(f64),
43    /// A parametric profile of time; see [`TemporalInjection`].
44    Profile(TemporalInjection),
45    /// `(t, value)` pairs, interpolated between them and held at the end
46    /// values outside their range, as
47    /// [`ExternalTabulated`](crate::context::calculators::ExternalTabulated)
48    /// does. At least two points, `t` strictly ascending, checked when the
49    /// domain is built.
50    Tabulated {
51        /// `(t, value)` pairs, `t` strictly ascending.
52        points: Vec<(f64, f64)>,
53        /// Interpolation between the points.
54        interpolation: Interpolation,
55    },
56}
57
58/// Which boundary condition to build, and with what parameters.
59///
60/// With the `serde` feature, a condition is written as its variant name
61/// around its parameters:
62///
63/// ```text
64/// boundary_conditions:
65///   - Dirichlet: { region: { Face: { axis: 0, side: Lower } }, value: { Constant: 1.0 } }
66///   - DanckwertsOutlet: {}
67/// ```
68#[non_exhaustive]
69#[derive(Debug, Clone)]
70#[cfg_attr(feature = "serde", derive(serde::Deserialize))]
71pub enum BoundaryConditionSpec {
72    /// A [`Dirichlet`] condition holding `value` on `region`.
73    Dirichlet {
74        /// Boundary region the condition acts on.
75        region: BoundaryRegion,
76        /// Value held on the region.
77        value: TimeFunctionSpec,
78    },
79    /// A [`DanckwertsInlet`], given by its own fields; `region` defaults to
80    /// the lower face of axis 0.
81    DanckwertsInlet(DanckwertsInlet),
82    /// A [`DanckwertsOutlet`], given by its own fields; `region` defaults to
83    /// the upper face of axis 0.
84    DanckwertsOutlet(DanckwertsOutlet),
85    /// An [`InjectionInlet`] following `profile`.
86    InjectionInlet {
87        /// Injection profile in time.
88        profile: TemporalInjection,
89    },
90}
91
92impl BoundaryConditionSpec {
93    /// The condition this spec describes.
94    ///
95    /// # Errors
96    ///
97    /// `OxiflowError::ExternalData` if a tabulated value has fewer than two
98    /// points or its `t` are not strictly ascending.
99    pub(crate) fn into_condition(self) -> Result<Box<dyn BoundaryCondition>, OxiflowError> {
100        Ok(match self {
101            Self::Dirichlet { region, value } => Box::new(dirichlet(region, value)?),
102            Self::DanckwertsInlet(inlet) => Box::new(inlet),
103            Self::DanckwertsOutlet(outlet) => Box::new(outlet),
104            Self::InjectionInlet { profile } => Box::new(InjectionInlet::new(profile)),
105        })
106    }
107}
108
109/// A [`Dirichlet`] condition holding `value` on `region`: a constant keeps
110/// the constant form, any other value becomes a function of time.
111fn dirichlet(region: BoundaryRegion, value: TimeFunctionSpec) -> Result<Dirichlet, OxiflowError> {
112    Ok(match value {
113        TimeFunctionSpec::Constant(value) => Dirichlet::constant(region, value),
114        TimeFunctionSpec::Profile(profile) => Dirichlet::of_t(region, move |t| profile.evaluate(t)),
115        TimeFunctionSpec::Tabulated {
116            points,
117            interpolation,
118        } => {
119            let series =
120                TabulatedSeries::new("TimeFunctionSpec::Tabulated", points, interpolation)?;
121            Dirichlet::of_t(region, move |t| series.value_at(t))
122        }
123    })
124}

131    const LOWER: BoundaryRegion = BoundaryRegion::Face {
132        axis: 0,
133        side: FaceSide::Lower,

136    #[test]
137    fn a_tabulated_value_needs_ordered_points() {
138        let spec = BoundaryConditionSpec::Dirichlet {
139            region: LOWER,
140            value: TimeFunctionSpec::Tabulated {
141                points: vec![(1.0, 0.0), (0.0, 1.0)],
142                interpolation: Interpolation::Linear,
143            },
144        };
145        let Err(OxiflowError::ExternalData(message)) = spec.into_condition() else {
146            panic!("an unordered series must be refused");
147        };
148        assert!(
149            message.starts_with("TimeFunctionSpec::Tabulated"),
150            "{message}"
151        );
152    }
153
154    #[test]
155    fn every_condition_builds() {
156        let specs = [
157            BoundaryConditionSpec::Dirichlet {
158                region: LOWER,
159                value: TimeFunctionSpec::Constant(1.0),
160            },
161            BoundaryConditionSpec::Dirichlet {
162                region: LOWER,
163                value: TimeFunctionSpec::Profile(TemporalInjection::SustainedStep {
164                    start: 0.0,
165                    concentration: 1.0,
166                }),
167            },
168            BoundaryConditionSpec::DanckwertsInlet(DanckwertsInlet::new(
169                1e-3,
170                0.1,
171                crate::context::variable::ContextVariable::Time,
172            )),
173            BoundaryConditionSpec::DanckwertsOutlet(DanckwertsOutlet::new()),
174            BoundaryConditionSpec::InjectionInlet {
175                profile: TemporalInjection::SustainedStep {
176                    start: 0.0,
177                    concentration: 1.0,
178                },
179            },
180        ];
181        for spec in specs {
182            assert!(spec.into_condition().is_ok());
183        }
184    }

186    #[cfg(feature = "serde")]
187    #[test]
188    fn time_functions_read_from_their_description() {
189        let read = |json: &str| serde_json::from_str::<TimeFunctionSpec>(json).unwrap();
190        assert_eq!(
191            read(r#"{ "Constant": 2.0 }"#),
192            TimeFunctionSpec::Constant(2.0)
193        );
194        assert_eq!(
195            read(
196                r#"{ "Profile": { "Gaussian": { "center": 5.0, "width": 1.0, "peak_concentration": 2.0 } } }"#
197            ),
198            TimeFunctionSpec::Profile(TemporalInjection::Gaussian {
199                center: 5.0,
200                width: 1.0,
201                peak_concentration: 2.0
202            })
203        );
204        assert_eq!(
205            read(
206                r#"{ "Tabulated": { "points": [[0.0, 0.0], [10.0, 1.0]], "interpolation": "Linear" } }"#
207            ),
208            TimeFunctionSpec::Tabulated {
209                points: vec![(0.0, 0.0), (10.0, 1.0)],
210                interpolation: Interpolation::Linear
211            }
212        );
213    }

215    #[cfg(feature = "serde")]
216    #[test]
217    fn danckwerts_conditions_read_their_own_fields_and_default_regions() {
218        let read = |json: &str| serde_json::from_str::<BoundaryConditionSpec>(json).unwrap();
219        let BoundaryConditionSpec::DanckwertsInlet(inlet) = read(
220            r#"{ "DanckwertsInlet": { "dispersion": 1e-6, "velocity": 1e-3, "feed_variable": { "External": { "name": "feed" } } } }"#,
221        ) else {
222            panic!("expected an inlet");
223        };
224        assert_eq!(inlet.region, LOWER);
225        assert_eq!(inlet.velocity, 1e-3);
226        let BoundaryConditionSpec::DanckwertsOutlet(outlet) = read(r#"{ "DanckwertsOutlet": {} }"#)
227        else {
228            panic!("expected an outlet");
229        };
230        assert_eq!(
231            outlet.region,
232            BoundaryRegion::Face {
233                axis: 0,
234                side: FaceSide::Upper
235            }
236        );
237    }
