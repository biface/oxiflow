1//! # Module `context::calculator`
2//!
3//! Trait `ContextCalculator` — computes a single context variable (issue [#32](https://github.com/biface/oxiflow/issues/32), [DD-020](https://github.com/biface/oxiflow/issues/57)).
4//!
5//! A calculator is the symmetric counterpart of `RequiresContext`: where a model
6//! *declares* what it needs, a calculator *provides* what it computes.
7//! The solver chains calculators in topological order ([DD-009](https://github.com/biface/oxiflow/issues/9), J2) and feeds
8//! results into `ComputeContext` before each time step.
9//!
10//! ## Placement
11//!
12//! Calculators live in `SolverConfiguration` — they are part of HOW the problem is
13//! solved, not WHAT the problem is. An FD `DiscreteOperator` (INV-2, J5) is an
14//! implementation detail inside spatial calculators, never exposed at configuration level.
15

20use crate::context::value::ContextValue;
21use crate::context::variable::ContextVariable;

24/// Computes one context variable and injects it into `ComputeContext`.
25///
26/// The solver calls calculators in topological order — a calculator that
27/// `depends_on()` variable X is guaranteed to run after the calculator that
28/// `provides()` X. Within a dependency tier, execution order is determined
29/// by `priority()` (inherited from `RequiresContext`).
30///
31/// # Contract
32///
33/// - `provides()` must return a stable `ContextVariable` — the same value
34///   across calls for a given calculator instance.
35/// - `compute()` receives the current field state and a **partially populated**
36///   `ComputeContext` (variables with lower priority already resolved).
37/// - `compute()` must not modify `ctx` directly — the solver inserts the result.
38/// - The solver calls [`compute_shared()`](Self::compute_shared), which defaults
39///   to `compute()`. A calculator that keeps its value between evaluations
40///   overrides it to hand that value over without a copy ([DT-068](https://github.com/biface/oxiflow/issues/195)).
41///
42/// # DiscreteOperator (INV-2, J5)
43///
44/// Spatial calculators (gradient, Laplacian, flux) may internally hold a
45/// `DiscreteOperator` implementation. This is an implementation detail of the
46/// calculator — `DiscreteOperator` is never exposed in `SolverConfiguration`.
47///
48/// # Examples
49///
50/// ```rust
51/// use oxiflow::context::calculator::ContextCalculator;
52/// use oxiflow::context::variable::ContextVariable;
53/// use oxiflow::context::value::ContextValue;
54/// use oxiflow::context::compute::ComputeContext;
55/// use oxiflow::context::error::OxiflowError;
56/// use oxiflow::model::traits::RequiresContext;
57///
58/// #[derive(Debug)]
59/// struct TimeCalculator;
60///
61/// impl RequiresContext for TimeCalculator {
62///     fn required_variables(&self) -> Vec<ContextVariable> { vec![] }
63///     fn priority(&self) -> u32 { 0 }
64/// }
65///
66/// impl ContextCalculator for TimeCalculator {
67///     fn provides(&self) -> ContextVariable { ContextVariable::Time }
68///     fn compute(&self, _state: &ContextValue, ctx: &ComputeContext)
69///         -> Result<ContextValue, OxiflowError>
70///     {
71///         Ok(ContextValue::Scalar(ctx.time()))
72///     }
73///     fn name(&self) -> &str { "time" }
74/// }
75/// ```
76pub trait ContextCalculator: RequiresContext + Send + Sync + std::fmt::Debug {
77    /// The context variable this calculator produces.
78    ///
79    /// Must be stable across calls. The solver uses this to match calculators
80    /// against `RequiresContext::required_variables()` declarations.
81    fn provides(&self) -> ContextVariable;
82
83    /// Computes the value for `provides()` given the current field state and
84    /// the partially populated context.
85    ///
86    /// Variables declared in `depends_on()` are guaranteed to be present in
87    /// `ctx` when this method is called.
88    fn compute(
89        &self,
90        state: &ContextValue,
91        ctx: &ComputeContext,
92    ) -> Result<ContextValue, OxiflowError>;
93
94    /// Computes the value for `provides()` as a shared allocation.
95    ///
96    /// This is the method the solver calls at each derivative evaluation.
97    /// The default wraps the result of [`compute()`](Self::compute) in an
98    /// `Arc`: the value is moved, not copied, at the cost of one `Arc`
99    /// allocation per evaluation.
100    ///
101    /// # When to override
102    ///
103    /// Override it when the calculator keeps a value from one evaluation to
104    /// the next (for instance a coefficient field evaluated once on the mesh).
105    /// Returning a clone of the kept `Arc` hands over the same allocation to
106    /// every evaluation, where `compute()` would have to copy the whole value.
107    ///
108    /// # Contract
109    ///
110    /// An override returns the same value as `compute()` for the same `state`
111    /// and `ctx`. `compute()` stays required, and remains the method to call
112    /// outside the solver.
113    ///
114    /// This method is not a parallel-dispatch capability and carries no
115    /// configuration, so it is consistent with P3 of [DA-048](https://github.com/biface/oxiflow/issues/140): an
116    /// implementer that ignores it is unaffected.
117    ///
118    /// # Examples
119    ///
120    /// ```rust
121    /// use std::sync::Arc;
122    ///
123    /// use nalgebra::DVector;
124    /// use oxiflow::context::{
125    ///     ComputeContext, ContextCalculator, ContextValue, ContextVariable, OxiflowError,
126    /// };
127    /// use oxiflow::model::RequiresContext;
128    ///
129    /// /// Coefficient field evaluated once, then shared with every evaluation.
130    /// #[derive(Debug)]
131    /// struct FixedCoefficient {
132    ///     field: Arc<ContextValue>,
133    /// }
134    ///
135    /// impl RequiresContext for FixedCoefficient {
136    ///     fn required_variables(&self) -> Vec<ContextVariable> { vec![] }
137    /// }
138    ///
139    /// impl ContextCalculator for FixedCoefficient {
140    ///     fn provides(&self) -> ContextVariable {
141    ///         ContextVariable::External { name: "c".into() }
142    ///     }
143    ///     fn compute(&self, _state: &ContextValue, _ctx: &ComputeContext)
144    ///         -> Result<ContextValue, OxiflowError>
145    ///     {
146    ///         Ok(self.field.as_ref().clone())
147    ///     }
148    ///     fn compute_shared(&self, _state: &ContextValue, _ctx: &ComputeContext)
149    ///         -> Result<Arc<ContextValue>, OxiflowError>
150    ///     {
151    ///         Ok(Arc::clone(&self.field))
152    ///     }
153    /// }
154    ///
155    /// let calc = FixedCoefficient {
156    ///     field: Arc::new(ContextValue::ScalarField(DVector::from_element(5, 0.5))),
157    /// };
158    /// let state = ContextValue::ScalarField(DVector::zeros(5));
159    /// let ctx = ComputeContext::new(0.0, 0.01);
160    ///
161    /// let first = calc.compute_shared(&state, &ctx)?;
162    /// let second = calc.compute_shared(&state, &ctx)?;
163    /// assert!(Arc::ptr_eq(&first, &second));
164    /// assert_eq!(*first, calc.compute(&state, &ctx)?);
165    /// # Ok::<(), OxiflowError>(())
166    /// ```
167    fn compute_shared(
168        &self,
169        state: &ContextValue,
170        ctx: &ComputeContext,
171    ) -> Result<Arc<ContextValue>, OxiflowError> {
172        self.compute(state, ctx).map(Arc::new)
173    }
174
175    /// Human-readable name for logging and error messages.
176    fn name(&self) -> &str {
177        "unnamed calculator"
178    }
179}

183/// Calculator that keeps one value and hands over the same allocation at every
184/// call of `compute_shared` — the case [DT-068](https://github.com/biface/oxiflow/issues/195) is about.
185///
186/// Shared by the unit tests of this module and of `solver::methods`.
187#[cfg(test)]
188#[derive(Debug)]
189pub(crate) struct CachedValueCalculator {
190    /// Variable provided.
191    pub(crate) var: ContextVariable,
192    /// Value kept between calls.
193    pub(crate) value: Arc<ContextValue>,
194}
195
196#[cfg(test)]
197impl RequiresContext for CachedValueCalculator {
198    fn required_variables(&self) -> Vec<ContextVariable> {
199        vec![]
200    }
201}
202
203#[cfg(test)]
204impl ContextCalculator for CachedValueCalculator {
205    fn provides(&self) -> ContextVariable {
206        self.var.clone()
207    }
208    fn compute(
209        &self,
210        _state: &ContextValue,
211        _ctx: &ComputeContext,
212    ) -> Result<ContextValue, OxiflowError> {
213        Ok(self.value.as_ref().clone())
214    }
215    fn compute_shared(
216        &self,
217        _state: &ContextValue,
218        _ctx: &ComputeContext,
219    ) -> Result<Arc<ContextValue>, OxiflowError> {
220        Ok(Arc::clone(&self.value))
221    }
222}

408    #[test]
409    fn overridden_compute_shared_hands_over_the_same_allocation() {
410        let calc = CachedValueCalculator {
411            var: ContextVariable::External { name: "c".into() },
412            value: Arc::new(ContextValue::ScalarField(nalgebra::DVector::from_element(
413                8, 1.5,
414            ))),
415        };
416        let ctx = ComputeContext::new(0.0, 0.01);
417        let state = ContextValue::Scalar(0.0);
418        let first = calc.compute_shared(&state, &ctx).unwrap();
419        let second = calc.compute_shared(&state, &ctx).unwrap();
420        assert!(Arc::ptr_eq(&first, &second));
421        assert!(Arc::ptr_eq(&first, &calc.value));
422        // Same value as compute(), as the contract requires.
423        assert_eq!(*first, calc.compute(&state, &ctx).unwrap());
424    }
