1//! # Module `context::value`
2//!
3//! Typed values carried by the compute context ([DD-003](https://github.com/biface/oxiflow/issues/3), issue [#27](https://github.com/biface/oxiflow/issues/27)).
4//!
5//! ## Two orthogonal axes
6//!
7//! `ContextValue` is structured along two independent axes:
8//!
9//! - **Rank** (algebraic): scalar (rank 0), vector (rank 1), matrix (rank 2 tensor).
10//! - **Distribution**: pointwise algebraic object vs. nodal field (one value per mesh node).
11//!
12//! ## Reserved variants (v0.14.0 (J14), [DD-057](https://github.com/biface/oxiflow/issues/159))
13//!
14//! `Tensor4` (rank-4 tensor C_ijkl) and `TensorField` (rank-2 tensor per node) are
15//! deliberately absent. They require `DiscreteOperator` (INV-2, J5) for covariant
16//! transformation semantics and will be added at v0.14.0 (J14) ([DD-057](https://github.com/biface/oxiflow/issues/159)). Tensors of rank > 4 are out of
17//! scope for all target physics; extension for third-party frameworks follows INV-4.
18//!
19//! ## Distinction from `PhysicalData`
20//!
21//! `PhysicalData` is the primary field `u` discretised on the mesh — a dimensionality-
22//! based array container with no tensor semantics. `ContextValue` holds coefficients,
23//! parameters and derived quantities consumed by physical models during time integration.
24

31/// Typed value of a context variable.
32///
33/// # Pointwise algebraic objects
34///
35/// | Variant | Rank | Example |
36/// |---------|------|---------|
37/// | `Scalar` | 0 | time, axial dispersion coefficient D_ax |
38/// | `Boolean` | — | convergence flag, saturation condition |
39/// | `Vector` | 1 | velocity at a point, pointwise gradient |
40/// | `Matrix` | 2 | diffusion tensor D_ij, stress tensor σ_ij |
41///
42/// # Nodal fields
43///
44/// | Variant | Content | Example |
45/// |---------|---------|---------|
46/// | `ScalarField` | `DVector` — one scalar per node | porosity(x), T(x) |
47/// | `VectorField` | `DMatrix` — n_nodes × d | velocity field u(x,y), d species per node |
48///
49/// # Reserved (v0.14.0 (J14), [DD-057](https://github.com/biface/oxiflow/issues/159) — requires INV-2 / DiscreteOperator)
50///
51/// ```text
52/// // Tensor4(Array4<f64>)  — rank-4 tensor C_ijkl (elastic stiffness)
53/// // TensorField(...)      — rank-2 tensor per node D_ij(x)
54/// // Tensors of rank > 4 are out of scope; extension via INV-4 for third-party frameworks.
55/// ```
56///
57/// # Examples
58///
59/// ```rust
60/// use oxiflow::context::value::ContextValue;
61/// use nalgebra::DVector;
62///
63/// let t   = ContextValue::Scalar(1.5);
64/// let flag = ContextValue::Boolean(false);
65/// let field = ContextValue::ScalarField(DVector::from_vec(vec![0.1, 0.2, 0.3]));
66///
67/// assert_eq!(t.as_scalar().unwrap(), 1.5);
68/// assert!(!field.as_scalar_field().unwrap().is_empty());
69/// ```
70// GPU-READY: bytemuck::Pod candidate (DA-026 INV-GPU-5) — pending v0.8.0 (J8),
71// for the `Vector`/`Matrix`/`ScalarField`/`VectorField` variants specifically;
72// `Scalar`/`Boolean` are trivially Pod-compatible, the enum discriminant is
73// the actual blocker for a direct `derive(Pod)` and will need to be resolved
74// at upload-boundary design time (DT-045), not before.
75#[non_exhaustive]
76#[derive(Debug, Clone, PartialEq)]
77#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
78pub enum ContextValue {
79    // ── Pointwise algebraic objects ───────────────────────────────────────────
80    /// Rank-0 scalar: time, step size, uniform coefficient.
81    Scalar(f64),
82
83    /// Logical flag: convergence condition, saturation state.
84    Boolean(bool),
85
86    /// Rank-1 vector: pointwise velocity, pointwise gradient.
87    Vector(DVector<f64>),
88
89    /// Rank-2 tensor: diffusion tensor D_ij, stress σ_ij, permeability K_ij.
90    ///
91    /// Covariant transformation is the caller's responsibility
92    /// via `DiscreteOperator` (INV-2, v0.14.0 (J14), [DD-057](https://github.com/biface/oxiflow/issues/159)):
93    ///
94    /// $$D'_{ij} = J_{ik}\,D_{kl}\,J_{jl}^{\top}$$
95    ///
96    /// This variant stores components in the current reference frame only.
97    ///
98    /// Memory layout: `DMatrix<f64>` is **column-major** (nalgebra default,
99    /// Fortran order) — [DA-026](https://github.com/biface/oxiflow/issues/73) INV-GPU-4. WGSL (the `wgpu` shader language)
100    /// is row-major; any transposition is resolved at the future GPU upload
101    /// boundary, never in this type.
102    Matrix(DMatrix<f64>),
103
104    // ── Nodal fields ──────────────────────────────────────────────────────────
105    /// One scalar value per mesh node: porosity(x), temperature field T(x).
106    ///
107    /// Length equals the number of degrees of freedom (`Mesh::n_dof()`).
108    ScalarField(DVector<f64>),
109
110    /// `d` values per mesh node, stored as an `n_nodes × d` matrix.
111    ///
112    /// Row `i` holds the values at node `i`. The `d` columns are either
113    /// the components of a vector quantity (`d` is then the spatial
114    /// dimension, as for a velocity field) or `d` unknowns carried by each
115    /// node, unrelated to the mesh dimension: two species on a 1D mesh give
116    /// `d = 2` ([#192](https://github.com/biface/oxiflow/issues/192)). Dimensions (`n_nodes`, `d`) are available via
117    /// `DMatrix::nrows()`/`ncols()` ([DA-026](https://github.com/biface/oxiflow/issues/73) INV-GPU-2).
118    ///
119    /// Memory layout: `DMatrix<f64>` is **column-major** (nalgebra default,
120    /// Fortran order) — [DA-026](https://github.com/biface/oxiflow/issues/73) INV-GPU-4. WGSL (the `wgpu` shader language)
121    /// is row-major; any transposition is resolved at the future GPU upload
122    /// boundary, never in this type.
123    VectorField(DMatrix<f64>),
124    // Reserved for v0.14.0 (J14) (DD-057) — requires DiscreteOperator (INV-2):
125    // Tensor4(ndarray::Array4<f64>)  — rank-4 tensor C_ijkl
126    // TensorField(...)               — rank-2 tensor per node D_ij(x)
127    // Rank > 4: out of scope; extension via INV-4 for third-party frameworks.
128}

133    /// Returns the variant name as a static string.
134    ///
135    /// Used to build `OxiflowError::TypeMismatch` messages.
136    pub fn variant_name(&self) -> &'static str {
137        match self {
138            Self::Scalar(_) => "Scalar",
139            Self::Boolean(_) => "Boolean",
140            Self::Vector(_) => "Vector",
141            Self::Matrix(_) => "Matrix",
142            Self::ScalarField(_) => "ScalarField",
143            Self::VectorField(_) => "VectorField",
144            // v0.14.0 variants (Tensor4, TensorField, DD-057) return their name when added
145            #[allow(unreachable_patterns)]
146            _ => "Unknown",
147        }
148    }

264    /// Extracts a single per-node scalar field to run stencil math against,
265    /// whether stored directly (`ScalarField`, `component` must be `None`)
266    /// or as one column of a per-node `VectorField` (`component` must be
267    /// `Some(i)`, one column per component -- see `VectorField`'s own
268    /// docs on row/column layout).
269    ///
270    /// This is the shared entry point for any calculator that operates
271    /// node-by-node on one field at a time: `FDGradientCalculator` and
272    /// `FDLaplacianCalculator` both go through it, so a future
273    /// per-component calculator does too, rather than re-deriving the same
274    /// match independently.
275    ///
276    /// Returns `Cow` rather than an owned `DVector`: the `ScalarField`
277    /// case borrows for free (this is the path every existing caller
278    /// before [#138](https://github.com/biface/oxiflow/issues/138) uses -- an unconditional clone here would have added a
279    /// full-length copy to every calculator call, on top of the stencil
280    /// computation itself. Measured directly: at n=1,000,000, cloning a
281    /// buffer this size costs ~650µs, comparable to the ~950µs the
282    /// Laplacian stencil itself took at that size in the v0.9.0
283    /// measurements ([#53](https://github.com/biface/oxiflow/issues/53)) -- close to doubling each of them). Only the
284    /// `VectorField` column-extraction case actually copies, since
285    /// `compute_from_dx` and friends take `&DVector<f64>` specifically,
286    /// not a generic matrix view, and a matrix column isn't stored the
287    /// same way a standalone vector is.
288    ///
289    /// # Errors
290    ///
291    /// - `OxiflowError::PreconditionFailed` if `component` is `Some(_)` for
292    ///   a `ScalarField` (mono-component; pass `None`), if `component` is
293    ///   `None` for a `VectorField` (a component index is required), or if
294    ///   the given component index is out of bounds for the `VectorField`.
295    /// - `OxiflowError::TypeMismatch` if the variant is neither
296    ///   `ScalarField` nor `VectorField`.
297    pub fn scalar_component(
298        &self,
299        component: Option<usize>,
300    ) -> Result<Cow<'_, DVector<f64>>, OxiflowError> {
301        match (self, component) {
302            (Self::ScalarField(v), None) => Ok(Cow::Borrowed(v)),
303            (Self::ScalarField(_), Some(c)) => Err(OxiflowError::PreconditionFailed {
304                context: "ContextValue::scalar_component",
305                message: format!(
306                    "component index {c} was given but the field is a plain ScalarField \
307                     (mono-component) -- pass None"
308                ),
309            }),
310            (Self::VectorField(m), Some(c)) => {
311                if c >= m.ncols() {
312                    return Err(OxiflowError::PreconditionFailed {
313                        context: "ContextValue::scalar_component",
314                        message: format!(
315                            "component index {c} out of bounds for VectorField with {} columns",
316                            m.ncols()
317                        ),
318                    });
319                }
320                Ok(Cow::Owned(m.column(c).into_owned()))
321            }
322            (Self::VectorField(_), None) => Err(OxiflowError::PreconditionFailed {
323                context: "ContextValue::scalar_component",
324                message: "state is a multi-component VectorField -- a component index is \
325                          required (pass Some(i)), None only fits ScalarField"
326                    .to_string(),
327            }),
328            (other, _) => Err(OxiflowError::TypeMismatch {
329                expected: "ScalarField or VectorField",
330                actual: other.variant_name(),
331            }),
332        }
333    }

335    /// Computes `self + dt * other`, handling `ScalarField` and
336    /// `VectorField` uniformly.
337    ///
338    /// The `u + dt·du/dt` update of `ForwardEulerSolver`. Generalizes the
339    /// same way [`Self::scalar_component`] did for calculator input
340    /// extraction, but for the solver's own state-update arithmetic. It is
341    /// the one-term case of the crate's internal linear combination of
342    /// states, which the multi-stage integrators use ([#192](https://github.com/biface/oxiflow/issues/192)).
343    ///
344    /// # Errors
345    ///
346    /// - `OxiflowError::TypeMismatch` if `self` and `other` are different
347    ///   variants, or neither is `ScalarField`/`VectorField`.
348    /// - `OxiflowError::InvalidDomain` if their shapes don't match (length
349    ///   for `ScalarField`, `(rows, cols)` for `VectorField`).
350    pub fn add_scaled(&self, dt: f64, other: &ContextValue) -> Result<ContextValue, OxiflowError> {
351        self.linear_combination(&[(dt, other)])
352    }
353
354    /// Computes `self + Σ cᵢ · termᵢ` for `ScalarField` and `VectorField`
355    /// states — the state algebra of every time integrator (Runge–Kutta
356    /// stages, θ-method and BDF2 predictors), kept in one place instead of
357    /// repeated per method ([#192](https://github.com/biface/oxiflow/issues/192)).
358    ///
359    /// Terms are accumulated in order, each as `acc ← acc + cᵢ·termᵢ`, so
360    /// `u.linear_combination(&[(dt, du)])` is bit-identical to `u + du * dt`.
361    /// An empty `terms` returns a copy of `self`.
362    ///
363    /// Crate-internal for now: it may become public once the state algebra
364    /// is settled by the finite-element work ([#150](https://github.com/biface/oxiflow/issues/150)).
365    ///
366    /// # Errors
367    ///
368    /// - `OxiflowError::TypeMismatch` if a term is not the same variant as
369    ///   `self`, or `self` is neither `ScalarField` nor `VectorField`.
370    /// - `OxiflowError::InvalidDomain` if a term's shape differs from
371    ///   `self` (length for `ScalarField`, `(rows, cols)` for `VectorField`).
372    pub(crate) fn linear_combination(
373        &self,
374        terms: &[(f64, &ContextValue)],
375    ) -> Result<ContextValue, OxiflowError> {
376        match self {
377            Self::ScalarField(_) | Self::VectorField(_) => self.clone().accumulate(terms, 0),
378            other => Err(OxiflowError::TypeMismatch {
379                expected: "ScalarField or VectorField",
380                actual: other.variant_name(),
381            }),
382        }
383    }
384
385    /// Computes `Σ cᵢ · termᵢ` for `ScalarField` and `VectorField` terms,
386    /// without a base state: the error estimate of an embedded Runge–Kutta
387    /// pair is the typical use ([#192](https://github.com/biface/oxiflow/issues/192)).
388    ///
389    /// The first term is scaled, then the others are accumulated in order
390    /// as in [`Self::linear_combination`], so the result is bit-identical
391    /// to `k₁ * c₁ + k₂ * c₂ + …` written with `DVector` arithmetic.
392    ///
393    /// # Errors
394    ///
395    /// - `OxiflowError::PreconditionFailed` if `terms` is empty.
396    /// - `OxiflowError::TypeMismatch` / `InvalidDomain` as for
397    ///   [`Self::linear_combination`], against the first term.
398    pub(crate) fn weighted_sum(
399        terms: &[(f64, &ContextValue)],
400    ) -> Result<ContextValue, OxiflowError> {
401        let Some((&(coefficient, first), rest)) = terms.split_first() else {
402            return Err(OxiflowError::PreconditionFailed {
403                context: "ContextValue::weighted_sum",
404                message: "at least one term is required".to_string(),
405            });
406        };
407        let scaled = match first {
408            Self::ScalarField(v) => Self::ScalarField(v * coefficient),
409            Self::VectorField(m) => Self::VectorField(m * coefficient),
410            other => {
411                return Err(OxiflowError::TypeMismatch {
412                    expected: "ScalarField or VectorField",
413                    actual: other.variant_name(),
414                })
415            }
416        };
417        scaled.accumulate(rest, 1)
418    }
419
420    /// Adds `Σ cᵢ · termᵢ` to `self` in place, in order. `first_index` is
421    /// the position of `terms[0]` in the caller's list, for error messages.
422    fn accumulate(
423        mut self,
424        terms: &[(f64, &ContextValue)],
425        first_index: usize,
426    ) -> Result<ContextValue, OxiflowError> {
427        for (offset, &(coefficient, term)) in terms.iter().enumerate() {
428            let index = first_index + offset;
429            match (&mut self, term) {
430                (Self::ScalarField(acc), Self::ScalarField(v)) => {
431                    if v.len() != acc.len() {
432                        return Err(OxiflowError::InvalidDomain(format!(
433                            "state length {} != length {} of term {index}",
434                            acc.len(),
435                            v.len()
436                        )));
437                    }
438                    acc.zip_apply(v, |a, b| *a += coefficient * b);
439                }
440                (Self::VectorField(acc), Self::VectorField(v)) => {
441                    if v.shape() != acc.shape() {
442                        return Err(OxiflowError::InvalidDomain(format!(
443                            "state shape {:?} != shape {:?} of term {index}",
444                            acc.shape(),
445                            v.shape()
446                        )));
447                    }
448                    acc.zip_apply(v, |a, b| *a += coefficient * b);
449                }
450                (state, term) => return Err(Self::term_mismatch(state, term)),
451            }
452        }
453        Ok(self)
454    }
455
456    /// `TypeMismatch` for a linear-combination term whose variant differs
457    /// from the state's.
458    fn term_mismatch(state: &ContextValue, term: &ContextValue) -> OxiflowError {
459        OxiflowError::TypeMismatch {
460            expected: state.variant_name(),
461            actual: term.variant_name(),
462        }
463    }

844    #[test]
845    fn weighted_sum_is_bit_identical_to_direct_arithmetic() {
846        let k1 = DVector::from_vec(vec![0.1, 1.0 / 3.0, -2.5e-9]);
847        let k2 = DVector::from_vec(vec![0.7, -2.0 / 3.0, 1.0e16]);
848        let k3 = DVector::from_vec(vec![-0.3, 0.2, 3.3]);
849        let (c1, c2, c3) = (0.013 * 71.0 / 57600.0, -0.013 * 71.0 / 16695.0, 0.017);
850        let direct = &k1 * c1 + &k2 * c2 + &k3 * c3;
851        let field = |v: &DVector<f64>| ContextValue::ScalarField(v.clone());
852        let sum =
853            ContextValue::weighted_sum(&[(c1, &field(&k1)), (c2, &field(&k2)), (c3, &field(&k3))])
854                .unwrap();
855        assert_eq!(sum.as_scalar_field().unwrap(), &direct);
856    }
