12// ── Interpolation ─────────────────────────────────────────────────────────────
13
14/// Interpolation strategy for [`ExternalTabulated`].
15///
16/// # Variants
17///
18/// - `Linear` — piecewise linear (1st-order accurate). Exact for linear f(t).
19///
20/// # Reserved
21///
22/// `PiecewiseCubic` (natural cubic spline, 4th-order accurate) is reserved, with
23/// no milestone assigned; it would require a `spline` feature flag.
24#[non_exhaustive]
25#[derive(Debug, Clone, PartialEq, Eq)]
26#[cfg_attr(feature = "serde", derive(serde::Deserialize))]
27pub enum Interpolation {
28    /// Piecewise linear interpolation: `f(t) = f_i + (f_{i+1} − f_i) · (t − t_i) / (t_{i+1} − t_i)`.
29    Linear,
30    // PiecewiseCubic — RESERVED, no milestone assigned (would require feature `spline`).
31}

33// ── TabulatedSeries ───────────────────────────────────────────────────────────
34
35/// `(t, value)` pairs with their interpolation: the data part of
36/// [`ExternalTabulated`], shared with every other consumer of a tabulated
37/// function of time, so that validation and interpolation exist once.
38#[derive(Debug, Clone)]
39pub(crate) struct TabulatedSeries {
40    /// (t, value) pairs, sorted by t ascending.
41    data: Vec<(f64, f64)>,
42    interpolation: Interpolation,
43}
44
45impl TabulatedSeries {
46    /// Checks `data` (at least 2 points, strictly ascending `t`) and keeps
47    /// it with `interpolation`. `owner` names the consumer in the error
48    /// messages.
49    ///
50    /// # Errors
51    ///
52    /// Returns `Err(OxiflowError::ExternalData)` if `data` has fewer than 2
53    /// points or is not sorted by strictly ascending `t`.
54    pub(crate) fn new(
55        owner: &str,
56        data: Vec<(f64, f64)>,
57        interpolation: Interpolation,
58    ) -> Result<Self, OxiflowError> {
59        if data.len() < 2 {
60            return Err(OxiflowError::ExternalData(format!(
61                "{owner} requires at least 2 data points, got {}",
62                data.len()
63            )));
64        }
65
66        // Verify ascending t order.
67        for w in data.windows(2) {
68            if w[0].0 >= w[1].0 {
69                return Err(OxiflowError::ExternalData(format!(
70                    "{owner} data must be sorted by ascending t: \
71                     t[i]={} >= t[i+1]={}",
72                    w[0].0, w[1].0
73                )));
74            }
75        }
76
77        Ok(Self {
78            data,
79            interpolation,
80        })
81    }
82
83    /// Interpolates the data at time `t`.
84    ///
85    /// Clamps to endpoint values outside the data range.
86    pub(crate) fn value_at(&self, t: f64) -> f64 {
87        let (t_min, v_min) = self.data[0];
88        let (t_max, v_max) = *self.data.last().unwrap();
89
90        // Clamp outside range.
91        if t <= t_min {
92            return v_min;
93        }
94        if t >= t_max {
95            return v_max;
96        }
97
98        // Binary search for the bracketing interval.
99        let idx = self
100            .data
101            .partition_point(|(ti, _)| *ti <= t)
102            .saturating_sub(1);
103
104        let (t0, v0) = self.data[idx];
105        let (t1, v1) = self.data[idx + 1];
106
107        match self.interpolation {
108            Interpolation::Linear => v0 + (v1 - v0) * (t - t0) / (t1 - t0),
109            // PiecewiseCubic would be added here (reserved, no milestone).
110            #[allow(unreachable_patterns)]
111            _ => v0, // unreachable while `Linear` is the only variant
112        }
113    }
114}

116// ── ExternalTabulated ─────────────────────────────────────────────────────────
117
118/// Provides a time-dependent scalar from tabulated (t, value) data.
119///
120/// Data points must be sorted by ascending `t`. The calculator interpolates at
121/// `ctx.time()` using the chosen [`Interpolation`] strategy.
122///
123/// Outside the data range the calculator clamps to the nearest endpoint value
124/// rather than extrapolating, and emits an `ExternalData` error if `data` is
125/// empty or has only one point (interpolation is undefined).
126///
127/// # Examples
128///
129/// ```rust
130/// use std::borrow::Cow;
131/// use oxiflow::context::calculator::ContextCalculator;
132/// use oxiflow::context::calculators::{ExternalTabulated, Interpolation};
133/// use oxiflow::context::compute::ComputeContext;
134/// use oxiflow::context::value::ContextValue;
135/// use oxiflow::context::variable::ContextVariable;
136///
137/// let var = ContextVariable::External { name: Cow::Borrowed("feed_conc") };
138/// let data = vec![(0.0, 1.0), (1.0, 2.0), (2.0, 1.5)];
139/// let calc = ExternalTabulated::new(var, data, Interpolation::Linear).unwrap();
140///
141/// // Interpolate at t = 0.5  →  1.0 + (2.0 − 1.0) × 0.5 = 1.5
142/// let ctx = ComputeContext::new(0.5, 0.01);
143/// let val = calc.compute(&ContextValue::Scalar(0.0), &ctx).unwrap();
144/// assert!((val.as_scalar().unwrap() - 1.5).abs() < 1e-10);
145/// ```
146#[derive(Debug)]
147pub struct ExternalTabulated {
148    variable: ContextVariable,
149    series: TabulatedSeries,
150}
151
152impl ExternalTabulated {
153    /// Loads tabulated data from an HDF5 file ([DT-027](https://github.com/biface/oxiflow/issues/75), issue [#105](https://github.com/biface/oxiflow/issues/105)).
154    ///
155    /// `variable` must be [`ContextVariable::External`] — its `name` is used
156    /// as the HDF5 group to read from. The group must contain two 1D `f64`
157    /// datasets, `t` and `value`, of equal length. Delegates to [`Self::new`]
158    /// for validation (≥ 2 points, strictly ascending `t`) — no duplicated
159    /// logic.
160    ///
161    /// Requires the `hdf5` feature.
162    ///
163    /// # Errors
164    ///
165    /// Returns `Err(OxiflowError::ExternalData)` if `variable` is not
166    /// `ContextVariable::External`. Returns `Err(OxiflowError::Persistence)`
167    /// if the file/group/dataset cannot be opened or read, or if `t` and
168    /// `value` have different lengths. Returns `Err(OxiflowError::ExternalData)`
169    /// (via [`Self::new`]) if the loaded data has fewer than 2 points or is
170    /// not sorted by ascending `t`.
171    #[cfg(feature = "hdf5")]
172    pub fn from_hdf5(
173        path: &std::path::Path,
174        variable: ContextVariable,
175        interpolation: Interpolation,
176    ) -> Result<Self, OxiflowError> {
177        let group_name = match &variable {
178            ContextVariable::External { name } => name.as_ref(),
179            _ => {
180                return Err(OxiflowError::ExternalData(
181                    "ExternalTabulated::from_hdf5 requires a ContextVariable::External \
182                     (its `name` is used as the HDF5 group to read)"
183                        .to_string(),
184                ));
185            }
186        };
187
188        let file = hdf5::File::open(path)
189            .map_err(|e| OxiflowError::Persistence(format!("cannot open {path:?}: {e}")))?;
190        let group = file.group(group_name).map_err(|e| {
191            OxiflowError::Persistence(format!("no group {group_name:?} in {path:?}: {e}"))
192        })?;
193
194        let read_column = |name: &str| -> Result<Vec<f64>, OxiflowError> {
195            let ds = group.dataset(name).map_err(|e| {
196                OxiflowError::Persistence(format!(
197                    "no dataset {name:?} in group {group_name:?} ({path:?}): {e}"
198                ))
199            })?;
200            let arr = ds.read_1d::<f64>().map_err(|e| {
201                OxiflowError::Persistence(format!(
202                    "cannot read dataset {name:?} in group {group_name:?} ({path:?}): {e}"
203                ))
204            })?;
205            Ok(arr.iter().copied().collect())
206        };
207
208        let t = read_column("t")?;
209        let value = read_column("value")?;
210
211        if t.len() != value.len() {
212            return Err(OxiflowError::Persistence(format!(
213                "group {group_name:?} ({path:?}): 't' has {} points, 'value' has {} — \
214                 must match",
215                t.len(),
216                value.len()
217            )));
218        }
219
220        let data: Vec<(f64, f64)> = t.into_iter().zip(value).collect();
221        Self::new(variable, data, interpolation)
222    }
223
224    /// Creates a new tabulated external calculator.
225    ///
226    /// # Arguments
227    ///
228    /// - `variable` — the `ContextVariable` this calculator provides.
229    /// - `data` — `(t, value)` pairs; must be sorted by ascending `t` and
230    ///   contain at least 2 points.
231    /// - `interpolation` — interpolation strategy.
232    ///
233    /// # Errors
234    ///
235    /// Returns `Err(OxiflowError::ExternalData)` if `data` has fewer than 2 points
236    /// or is not sorted by ascending `t`.
237    pub fn new(
238        variable: ContextVariable,
239        data: Vec<(f64, f64)>,
240        interpolation: Interpolation,
241    ) -> Result<Self, OxiflowError> {
242        Ok(Self {
243            variable,
244            series: TabulatedSeries::new("ExternalTabulated", data, interpolation)?,
245        })
246    }
247}
248
249impl RequiresContext for ExternalTabulated {
250    fn required_variables(&self) -> Vec<ContextVariable> {
251        vec![]
252    }
253
254    // External data runs before derived quantities (priority 100) but after
255    // time built-ins (priority 0).
256    fn priority(&self) -> u32 {
257        50
258    }
259}
260
261impl ContextCalculator for ExternalTabulated {
262    fn provides(&self) -> ContextVariable {
263        self.variable.clone()
264    }
265
266    fn compute(
267        &self,
268        _state: &ContextValue,
269        ctx: &ComputeContext,
270    ) -> Result<ContextValue, OxiflowError> {
271        let value = self.series.value_at(ctx.time());
272        Ok(ContextValue::Scalar(value))
273    }
274
275    fn name(&self) -> &str {
276        "external_tabulated (built-in)"
277    }
278}

424    #[test]
425    fn from_hdf5_round_trip() {
426        let path = std::env::temp_dir().join("oxiflow_test_from_hdf5_round_trip.h5");
427
428        // Write a fixture matching the schema from_hdf5 expects: one group
429        // named after the variable, two 1D f64 datasets "t"/"value".
430        {
431            let file = hdf5::File::create(&path).unwrap();
432            let group = file.create_group("feed_conc").unwrap();
433            group
434                .new_dataset::<f64>()
435                .shape(3)
436                .create("t")
437                .unwrap()
438                .write_raw(&[0.0, 1.0, 2.0])
439                .unwrap();
440            group
441                .new_dataset::<f64>()
442                .shape(3)
443                .create("value")
444                .unwrap()
445                .write_raw(&[1.0, 2.0, 1.5])
446                .unwrap();
447        }
448
449        let variable = ContextVariable::External {
450            name: "feed_conc".into(),
451        };
452        let calc = ExternalTabulated::from_hdf5(&path, variable, Interpolation::Linear).unwrap();
453
454        // Same fixture as the module-level rustdoc example (t=0.5 -> 1.5).
455        let val = calc.compute(&ContextValue::Scalar(0.0), &ctx(0.5)).unwrap();
456        assert!((val.as_scalar().unwrap() - 1.5).abs() < 1e-10);
457
458        std::fs::remove_file(&path).ok();
459    }
