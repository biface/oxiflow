1//! # Module `context::calculators::spatial`
2//!
3//! Finite-difference spatial calculators: gradient and Laplacian.
4//!
5//! Both calculators hold an `Arc<dyn Mesh>` internally (INV-1, DD-007) and
6//! delegate their stencil math to `operators::fd`'s `compute_from_dx`
7//! functions (#47, FD delegation refactor) — see that module's documentation
8//! for why the delegation bypasses `DiscreteOperator::apply()` itself.
9//!
10//! ## `parallel_threshold` via `solver::parallel::ParallelThreshold` (DD-014, DD-048)
11//!
12//! Both calculators hold a
13//! [`ParallelThreshold`](crate::solver::parallel::ParallelThreshold) field
14//! rather than reimplementing the atomic/sentinel/guard mechanism
15//! themselves — extracted into that shared type after this module wrote
16//! it twice, verbatim, on `FDGradientCalculator` and `FDLaplacianCalculator`.
17//! See that module's documentation for the full rationale (per-instance vs.
18//! chrom-rs's process-wide `static`, the `0`-as-sentinel design, why no
19//! trait is imposed on `ContextCalculator` for this).
20

35/// Translates `operators::fd`'s `InvalidDomain` (#47) into this calculator's
36/// `PreconditionFailed`, preserved from before the delegation refactor.
37///
38/// `operators::fd` is new code with no external consumer yet, so it is free
39/// to use `InvalidDomain` as `#47` specifies. `FDGradientCalculator`/
40/// `FDLaplacianCalculator` are a compatibility boundary already shipped in
41/// v0.2.0's public API — 21 existing tests assert `PreconditionFailed` for
42/// exactly this "field too short for the stencil" condition, so changing the
43/// variant here would be a breaking change unrelated to `#47`'s scope. Any
44/// other error variant (e.g. `TypeMismatch` from `as_scalar_field()`) passes
45/// through unchanged.
46fn translate_domain_error(err: OxiflowError, context: &'static str) -> OxiflowError {
47    match err {
48        OxiflowError::InvalidDomain(message) => {
49            OxiflowError::PreconditionFailed { context, message }
50        }
51        other => other,
52    }
53}

55// ── FDGradientCalculator ──────────────────────────────────────────────────────

57/// Computes the finite-difference spatial gradient of the primary field.
58///
59/// Provides `ContextVariable::SpatialGradient { dimension, component }` as a
60/// `ContextValue::ScalarField` — one gradient value per mesh node.
61///
62/// The mesh is held as `Arc<dyn Mesh>` internally (INV-1). Stencil math is
63/// delegated to `operators::fd::{UpwindGradient, CenteredGradient}` (#47).
64///
65/// # Boundary treatment
66///
67/// | Scheme | Interior | Left boundary (i = 0) | Right boundary (i = n−1) |
68/// |---|---|---|---|
69/// | `Forward` | `(u[i+1] − u[i]) / dx` | same | `(u[n−1] − u[n−2]) / dx` |
70/// | `Backward` | `(u[i] − u[i−1]) / dx` | `(u[1] − u[0]) / dx` | same |
71/// | `Central` | `(u[i+1] − u[i−1]) / 2dx` | `(u[1] − u[0]) / dx` | `(u[n−1] − u[n−2]) / dx` |
72///
73/// # Examples
74///
75/// ```rust
76/// use std::sync::Arc;
77/// use oxiflow::context::calculator::ContextCalculator;
78/// use oxiflow::context::calculators::{FDGradientCalculator, FDScheme};
79/// use oxiflow::context::compute::ComputeContext;
80/// use oxiflow::context::value::ContextValue;
81/// use oxiflow::context::variable::ContextVariable;
82/// use oxiflow::mesh::{Mesh, UniformGrid1D};
83/// use nalgebra::DVector;
84///
85/// let mesh = Arc::new(UniformGrid1D::new(5, 0.0, 1.0).unwrap());
86/// let calc = FDGradientCalculator::new(mesh, 0, None, FDScheme::Central);
87///
88/// // Linear field u = x  →  ∂u/∂x = 1 everywhere
89/// let u = DVector::from_vec(vec![0.0, 0.25, 0.5, 0.75, 1.0]);
90/// let ctx = ComputeContext::new(0.0, 0.01);
91/// let grad = calc.compute(&ContextValue::ScalarField(u), &ctx).unwrap();
92/// let field = grad.as_scalar_field().unwrap();
93/// for g in field.iter() {
94///     assert!((g - 1.0).abs() < 1e-10);
95/// }
96/// ```
97pub struct FDGradientCalculator {
98    mesh: Arc<dyn Mesh>,
99    dimension: usize,
100    component: Option<usize>,
101    scheme: FDScheme,
102    /// Rayon dispatch threshold (DD-014, DD-048) — see
103    /// [`ParallelThreshold`]'s own docs for the sentinel/gating rationale.
104    #[cfg(feature = "parallel")]
105    parallel_threshold: ParallelThreshold,
106}

109    /// Creates a new FD gradient calculator.
110    ///
111    /// # Arguments
112    ///
113    /// - `mesh` — shared mesh reference (INV-1 compliant).
114    /// - `dimension` — spatial dimension (0 → ∂u/∂x, 1 → ∂u/∂y, …).
115    /// - `component` — field component (`None` for mono-component, J1/J2 default).
116    /// - `scheme` — finite-difference stencil.
117    ///
118    /// `parallel_threshold` has no override by default —
119    /// [`Self::parallel_threshold`] reports
120    /// [`operators::fd::default_parallel_threshold`](crate::operators::fd::default_parallel_threshold)
121    /// (DD-014, #51) until one is set, at construction with
122    /// [`Self::with_parallel_threshold`] or at runtime with
123    /// [`Self::set_parallel_threshold`]; both available only when the crate
124    /// is built with the `parallel` feature.
125    pub fn new(
126        mesh: Arc<dyn Mesh>,
127        dimension: usize,
128        component: Option<usize>,
129        scheme: FDScheme,
130    ) -> Self {
131        Self {
132            mesh,
133            dimension,
134            component,
135            scheme,
136            #[cfg(feature = "parallel")]
137            parallel_threshold: ParallelThreshold::unset(),
138        }
139    }

141    /// Overrides the Rayon dispatch threshold at construction time (DD-014,
142    /// #51). Only exists when the crate is built with the `parallel`
143    /// feature — mirrors `BackwardEulerSolver::with_sparse_threshold`'s
144    /// gating (DD-043).
145    ///
146    /// # Panics
147    ///
148    /// Panics when `threshold == 0` — see [`ParallelThreshold::set`].
149    #[cfg(feature = "parallel")]
150    pub fn with_parallel_threshold(self, threshold: usize) -> Self {
151        self.parallel_threshold.set(threshold);
152        self
153    }
154
155    /// Reconfigures the Rayon dispatch threshold at runtime, through `&self`
156    /// — no rebuild required, unlike a consuming builder. Scoped to this
157    /// instance (see [`ParallelThreshold`]'s docs for why, vs. chrom-rs's
158    /// process-wide `set_parallel_threshold()`). Only exists when the crate
159    /// is built with the `parallel` feature.
160    ///
161    /// # Panics
162    ///
163    /// Panics when `threshold == 0` (see [`ParallelThreshold::set`]).
164    #[cfg(feature = "parallel")]
165    pub fn set_parallel_threshold(&self, threshold: usize) {
166        self.parallel_threshold.set(threshold);
167    }
168
169    /// Returns the effective Rayon dispatch threshold — the configured
170    /// override if one was set, otherwise
171    /// [`operators::fd::default_parallel_threshold`](crate::operators::fd::default_parallel_threshold)
172    /// resolved on demand. Only exists when the crate is built with the
173    /// `parallel` feature.
174    #[cfg(feature = "parallel")]
175    pub fn parallel_threshold(&self) -> usize {
176        self.parallel_threshold
177            .get(crate::operators::fd::default_parallel_threshold())
178    }

204impl ContextCalculator for FDGradientCalculator {
205    fn provides(&self) -> ContextVariable {
206        ContextVariable::SpatialGradient {
207            dimension: self.dimension,
208            component: self.component,
209        }
210    }
211
212    fn compute(
213        &self,
214        state: &ContextValue,
215        _ctx: &ComputeContext,
216    ) -> Result<ContextValue, OxiflowError> {
217        let u = state.scalar_component(self.component)?;
218        let dx = self.mesh.characteristic_length();
219
220        // Delegates stencil math to `operators::fd` (#47, FD delegation
221        // refactor) — `Arc<dyn Mesh>` here cannot produce the concrete
222        // `&UniformGrid1D` that `DiscreteOperator::apply()` requires, so the
223        // mesh-free `compute_from_dx` entry point is used directly instead
224        // (see `operators::fd`'s module documentation for why).
225        //
226        // The match itself is duplicated per cfg, not just its threshold
227        // input (DD-014, #51): `compute_from_dx`'s arity differs by feature
228        // (see `operators::fd`), so a single match sharing one `threshold`
229        // binding would either not compile without `parallel`, or compute a
230        // threshold nobody reads — the earlier version of this method did
231        // exactly that.
232        #[cfg(feature = "parallel")]
233        let grad = {
234            let threshold = self.parallel_threshold();
235            match self.scheme {
236                FDScheme::Forward => {
237                    UpwindGradient::compute_from_dx(&u, dx, Direction::Forward, threshold)
238                }
239                FDScheme::Backward => {
240                    UpwindGradient::compute_from_dx(&u, dx, Direction::Backward, threshold)
241                }
242                FDScheme::Central => CenteredGradient::compute_from_dx(&u, dx, threshold),
243                // J5+: higher-order stencils will be added here.
244                #[allow(unreachable_patterns)]
245                _ => {
246                    return Err(OxiflowError::PreconditionFailed {
247                        context: "FDGradientCalculator",
248                        message: "unsupported FDScheme variant".to_string(),
249                    })
250                }
251            }
252        };
253        #[cfg(not(feature = "parallel"))]
254        let grad = match self.scheme {
255            FDScheme::Forward => UpwindGradient::compute_from_dx(&u, dx, Direction::Forward),
256            FDScheme::Backward => UpwindGradient::compute_from_dx(&u, dx, Direction::Backward),
257            FDScheme::Central => CenteredGradient::compute_from_dx(&u, dx),
258            #[allow(unreachable_patterns)]
259            _ => {
260                return Err(OxiflowError::PreconditionFailed {
261                    context: "FDGradientCalculator",
262                    message: "unsupported FDScheme variant".to_string(),
263                })
264            }
265        };
266
267        let grad = grad.map_err(|e| translate_domain_error(e, "FDGradientCalculator"))?;
268
269        Ok(ContextValue::ScalarField(grad))
270    }
271
272    fn name(&self) -> &str {
273        "fd_gradient (built-in)"
274    }
275}

277// ── FDLaplacianCalculator ─────────────────────────────────────────────────────
278
279/// Computes the finite-difference Laplacian ∇²u of the primary field.
280///
281/// Provides a user-chosen `ContextVariable::External { name }` as a
282/// `ContextValue::ScalarField`. The Laplacian is a scalar nodal field of the
283/// same length as the primary field — no new `ContextVariable` variant is
284/// needed (DD-026 deferred).
285///
286/// # Stencil
287///
288/// Standard 3-point central: `(u[i−1] − 2u[i] + u[i+1]) / dx²`
289///
290/// Boundary treatment (1st-order one-sided):
291/// - `i = 0` → `(u[0] − 2u[1] + u[2]) / dx²`
292/// - `i = n−1` → `(u[n−3] − 2u[n−2] + u[n−1]) / dx²`
293///
294/// # Examples
295///
296/// ```rust
297/// use std::sync::Arc;
298/// use std::borrow::Cow;
299/// use oxiflow::context::calculator::ContextCalculator;
300/// use oxiflow::context::calculators::FDLaplacianCalculator;
301/// use oxiflow::context::compute::ComputeContext;
302/// use oxiflow::context::value::ContextValue;
303/// use oxiflow::context::variable::ContextVariable;
304/// use oxiflow::mesh::{Mesh, UniformGrid1D};
305/// use nalgebra::DVector;
306///
307/// let mesh = Arc::new(UniformGrid1D::new(5, 0.0, 1.0).unwrap());
308/// let var  = ContextVariable::External { name: Cow::Borrowed("laplacian") };
309/// let calc = FDLaplacianCalculator::new(mesh, var);
310///
311/// // Quadratic field u = x²  →  ∇²u = 2 everywhere (interior)
312/// let u = DVector::from_vec(vec![0.0, 0.0625, 0.25, 0.5625, 1.0]);
313/// let ctx = ComputeContext::new(0.0, 0.01);
314/// let lap = calc.compute(&ContextValue::ScalarField(u), &ctx).unwrap();
315/// let field = lap.as_scalar_field().unwrap();
316/// // Interior nodes: ∇²u ≈ 2.0
317/// assert!((field[2] - 2.0).abs() < 1e-6);
318/// ```
319pub struct FDLaplacianCalculator {
320    mesh: Arc<dyn Mesh>,
321    variable: ContextVariable,
322    /// Which column of a `VectorField` state to read, or `None` for a
323    /// plain mono-component `ScalarField` — see
324    /// [`ContextValue::scalar_component`]. Defaults to `None`; set via
325    /// [`Self::with_component`] for a multi-component state (one
326    /// calculator instance per component, each with its own `variable`
327    /// tag to keep them distinct in the context store).
328    component: Option<usize>,
329    /// Rayon dispatch threshold (DD-014, DD-048) — see
330    /// [`ParallelThreshold`]'s own docs for the sentinel/gating rationale.
331    #[cfg(feature = "parallel")]
332    parallel_threshold: ParallelThreshold,
333}

437impl ContextCalculator for FDLaplacianCalculator {
438    fn provides(&self) -> ContextVariable {
439        self.variable.clone()
440    }
441
442    fn compute(
443        &self,
444        state: &ContextValue,
445        _ctx: &ComputeContext,
446    ) -> Result<ContextValue, OxiflowError> {
447        let u = state.scalar_component(self.component)?;
448        let dx = self.mesh.characteristic_length();
449
450        // Delegates stencil math to `operators::fd` (#47, FD delegation
451        // refactor) — see `FDGradientCalculator::compute()` above for why
452        // `compute_from_dx` is used directly rather than `apply()`, and for
453        // why the call itself (not just its threshold argument) is split by
454        // cfg: `compute_from_dx`'s arity differs by feature.
455        #[cfg(feature = "parallel")]
456        let lap = CenteredLaplacian::compute_from_dx(&u, dx, self.parallel_threshold());
457        #[cfg(not(feature = "parallel"))]
458        let lap = CenteredLaplacian::compute_from_dx(&u, dx);
459
460        let lap = lap.map_err(|e| translate_domain_error(e, "FDLaplacianCalculator"))?;
461
462        Ok(ContextValue::ScalarField(lap))
463    }

466        "fd_laplacian (built-in)"
467    }

699    //
700    // Overriding the threshold must not change the result on fields far
701    // below and far above it — only which `operators::fd` code path runs.
702    // Gated: the builder itself only exists with the `parallel` feature
703    // (mirrors `BackwardEulerSolver::with_sparse_threshold`'s gating).
704
705    #[cfg(feature = "parallel")]
706    #[test]
707    fn gradient_with_parallel_threshold_does_not_change_result() {
708        let n = 5;
709        let mesh = grid(n);
710        let dx = mesh.characteristic_length();
711        let u: Vec<f64> = (0..n).map(|i| i as f64 * dx).collect();
712        let field = ContextValue::ScalarField(DVector::from_vec(u));
713
714        let default_calc = FDGradientCalculator::new(grid(n), 0, None, FDScheme::Central);
715        let forced_parallel = FDGradientCalculator::new(grid(n), 0, None, FDScheme::Central)
716            .with_parallel_threshold(1);
717
718        let default_grad = default_calc.compute(&field, &ctx()).unwrap();
719        let forced_grad = forced_parallel.compute(&field, &ctx()).unwrap();
720        assert_eq!(
721            default_grad.as_scalar_field().unwrap(),
722            forced_grad.as_scalar_field().unwrap()
723        );
724    }
