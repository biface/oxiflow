1//! # Module `operators::spec`
2//!
3//! External, serialisable description of a flux scheme — the scheme part of
4//! the WHAT-axis configuration ([DT-027](https://github.com/biface/oxiflow/issues/75)
5//! amendment 1, [#108](https://github.com/biface/oxiflow/issues/108)).
6//!
7//! [`FluxSchemeSpec`] names one of the [`FluxDivergenceOperator`](super::FluxDivergenceOperator)
8//! schemes of this module and its constructor parameters; a variant drops
9//! the role suffix of its type (`FVUpwindFlux` is `FVUpwind`), as
10//! `IntegratorSpec` does for solvers (`sparse` feature). It is
11//! consumed by [`ModelSpec`](crate::model::ModelSpec), which wraps the scheme
12//! in a [`DiscretizedModel`](crate::model::DiscretizedModel).
13//!
14//! [`FluxBoundarySpec`] covers the stencil closures that hold no data.
15//! [`FluxBoundary::GhostCell`] is left out: it holds the boundary conditions
16//! of the domain by position, a form that DD-051
17//! ([#152](https://github.com/biface/oxiflow/issues/152)) is to revisit.
18
19use super::limiters::Limiter;
20use super::FluxBoundary;
21
22/// Closure of the stencil at the two ends of the domain; see
23/// [`FluxBoundary`] for each convention.
24#[non_exhaustive]
25#[derive(Debug, Clone, Copy, PartialEq, Eq)]
26#[cfg_attr(feature = "serde", derive(serde::Deserialize))]
27pub enum FluxBoundarySpec {
28    /// [`FluxBoundary::Periodic`].
29    Periodic,
30    /// [`FluxBoundary::Truncation`].
31    Truncation,
32}
33
34impl FluxBoundarySpec {
35    /// The closure this spec describes.
36    pub(crate) fn into_flux_boundary(self) -> FluxBoundary {
37        match self {
38            Self::Periodic => FluxBoundary::Periodic,
39            Self::Truncation => FluxBoundary::Truncation,
40        }
41    }
42}
43
44/// Which flux scheme to build, and with what parameters. Each variant takes
45/// the parameters of the scheme's constructor.
46///
47/// With the `serde` feature, a scheme is written as its variant name around
48/// its parameters:
49///
50/// ```text
51/// scheme:
52///   Limited: { velocity: 1.0, diffusion: 1.0e-3, limiter: VanLeer, boundary: Truncation }
53/// ```
54#[non_exhaustive]
55#[derive(Debug, Clone, PartialEq)]
56#[cfg_attr(feature = "serde", derive(serde::Deserialize))]
57pub enum FluxSchemeSpec {
58    /// [`FVCenteredFlux`](super::fv::FVCenteredFlux).
59    FVCentered {
60        /// Advection velocity.
61        velocity: f64,
62        /// Diffusion coefficient.
63        diffusion: f64,
64        /// Closure of the stencil.
65        boundary: FluxBoundarySpec,
66    },
67    /// [`FVUpwindFlux`](super::fv::FVUpwindFlux).
68    FVUpwind {
69        /// Advection velocity.
70        velocity: f64,
71        /// Diffusion coefficient.
72        diffusion: f64,
73        /// Closure of the stencil.
74        boundary: FluxBoundarySpec,
75    },
76    /// [`WENO3`](super::weno::WENO3).
77    WENO3 {
78        /// Advection velocity.
79        velocity: f64,
80        /// Diffusion coefficient.
81        diffusion: f64,
82        /// Closure of the stencil.
83        boundary: FluxBoundarySpec,
84    },
85    /// [`WENO5`](super::weno::WENO5).
86    WENO5 {
87        /// Advection velocity.
88        velocity: f64,
89        /// Diffusion coefficient.
90        diffusion: f64,
91        /// Closure of the stencil.
92        boundary: FluxBoundarySpec,
93    },
94    /// [`LimitedFlux`](super::limiters::LimitedFlux).
95    Limited {
96        /// Advection velocity.
97        velocity: f64,
98        /// Diffusion coefficient.
99        diffusion: f64,
100        /// Flux limiter.
101        limiter: Limiter,
102        /// Closure of the stencil.
103        boundary: FluxBoundarySpec,
104    },
105    /// [`AdaptiveFlux`](super::adaptive::AdaptiveFlux).
106    Adaptive {
107        /// Advection velocity.
108        velocity: f64,
109        /// Diffusion coefficient.
110        diffusion: f64,
111        /// Flux limiter of the blended limited flux.
112        limiter: Limiter,
113        /// Mesh Péclet number above which the limited flux is blended in.
114        peclet_threshold: f64,
115        /// Closure of the stencil.
116        boundary: FluxBoundarySpec,
117    },
118}
119
120impl FluxSchemeSpec {
121    /// Short name of the scheme, the default name of a model built on it.
122    pub(crate) fn label(&self) -> &'static str {
123        match self {
124            Self::FVCentered { .. } => "fv_centered",
125            Self::FVUpwind { .. } => "fv_upwind",
126            Self::WENO3 { .. } => "weno3",
127            Self::WENO5 { .. } => "weno5",
128            Self::Limited { .. } => "limited",
129            Self::Adaptive { .. } => "adaptive",
130        }
131    }
132}

134#[cfg(test)]
135mod tests {
136    use super::*;
137
138    #[cfg(feature = "serde")]
139    fn read(json: &str) -> FluxSchemeSpec {
140        serde_json::from_str(json).unwrap()
141    }
142
143    #[cfg(feature = "serde")]
144    #[test]
145    fn every_scheme_reads_from_its_parameters() {
146        let common = r#""velocity": 1.0, "diffusion": 0.5, "boundary": "Truncation""#;
147        let cases = [
148            ("FVCentered", "fv_centered"),
149            ("FVUpwind", "fv_upwind"),
150            ("WENO3", "weno3"),
151            ("WENO5", "weno5"),
152        ];
153        for (variant, label) in cases {
154            let spec = read(&format!(r#"{{ "{variant}": {{ {common} }} }}"#));
155            assert_eq!(spec.label(), label);
156        }
157        let limited = read(&format!(
158            r#"{{ "Limited": {{ {common}, "limiter": "VanLeer" }} }}"#
159        ));
160        assert_eq!(
161            limited,
162            FluxSchemeSpec::Limited {
163                velocity: 1.0,
164                diffusion: 0.5,
165                limiter: Limiter::VanLeer,
166                boundary: FluxBoundarySpec::Truncation
167            }
168        );
169        let adaptive = read(&format!(
170            r#"{{ "Adaptive": {{ {common}, "limiter": "MinMod", "peclet_threshold": 2.0 }} }}"#
171        ));
172        assert_eq!(adaptive.label(), "adaptive");
173    }
174
175    #[cfg(feature = "serde")]
176    #[test]
177    fn a_closure_holding_boundary_conditions_is_refused() {
178        let json =
179            r#"{ "FVUpwind": { "velocity": 1.0, "diffusion": 0.0, "boundary": "GhostCell" } }"#;
180        assert!(serde_json::from_str::<FluxSchemeSpec>(json).is_err());
181    }
182
183    #[test]
184    fn closures_map_to_their_flux_boundary() {
185        assert!(matches!(
186            FluxBoundarySpec::Periodic.into_flux_boundary(),
187            FluxBoundary::Periodic
188        ));
189        assert!(matches!(
190            FluxBoundarySpec::Truncation.into_flux_boundary(),
191            FluxBoundary::Truncation
192        ));
193    }
