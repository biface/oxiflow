1//! # Module `operators`
2//!
3//! Spatial discretization operators — INV-2 invariant.
4//!
5//! ## Core principle (INV-2)
6//!
7//! Spatial discretization is decoupled from the rest of the engine through two
8//! sibling traits — never a single generic-over-everything trait, and never a
9//! scheme called directly from a consumer:
10//!
11//! - [`DiscreteOperator`] — raw differential quantities (`∂u/∂x`, `∂²u/∂x²`) that
12//!   never need physical parameters or context access; `dx` comes from the mesh
13//!   alone. Consumed transparently through [`crate::context::ContextCalculator`]
14//!   (e.g. `FDGradientCalculator`) — a model declares a required context
15//!   variable and never knows which scheme produced it (DD-012).
16//! - [`FluxDivergenceOperator`] — the complete `∇·F(u, ∇u)` term, which does need
17//!   physical parameters (advection velocity, diffusion coefficient) that may be
18//!   space/time-varying. Carries `ctx: &ComputeContext` and `RequiresContext`
19//!   directly on `apply()`. Consumed by `DiscretizedModel<Op>` (DD-038, DD-039).
20//!
21//! `FluxDivergenceOperator` is a **sibling** trait, not a sub-trait of
22//! `DiscreteOperator`: their `apply()` signatures diverge (with/without `ctx`),
23//! so a sub-trait relationship could not cleanly reuse the parent method.
24//! Forcing `ctx`/`RequiresContext` onto `DiscreteOperator` for every scheme was
25//! considered and rejected (DD-012, amendment 2) — it would have imposed that
26//! capability on FD, which never needs it.
27//!
28//! **Selection criterion for future schemes** (FEM, spectral methods, J20):
29//! does the scheme need per-call access to physical context, or does it reduce
30//! to a local differentiation from mesh data alone? The former implements
31//! `FluxDivergenceOperator`; the latter implements `DiscreteOperator`.
32//!
33//! ## Planned implementations
34//!
35//! | Type | Scheme | Trait | Milestone |
36//! |---|---|---|---|
37//! | [`fd::UpwindGradient`], [`fd::CenteredGradient`], [`fd::CenteredLaplacian`] | FD | `DiscreteOperator` | v0.5.0 (#47) ✅ |
38//! | [`fv::FVCenteredFlux`], [`fv::FVUpwindFlux`] | FV | `FluxDivergenceOperator` | v0.5.0 (#48) ✅ |
39//! | [`weno::WENO3`], [`weno::WENO5`] | WENO | `FluxDivergenceOperator` | v0.5.0 (#49) ✅ |
40//! | [`limiters::LimitedFlux`] (`MinMod`/`VanLeer`/`Superbee`) | MUSCL | `FluxDivergenceOperator` | v0.5.0 (#99) ✅ |
41//! | [`adaptive::AdaptiveFlux`] (WENO3/limiter blend, Péclet-gated) | — | `FluxDivergenceOperator` | v0.5.0 (#99) ✅ |
42//! | `FiniteElement` | FEM P1/P2 | TBD | J7 — v2.0 |

58/// Raw spatial differential operator (FD family) — INV-2.
59///
60/// Computes a single differential quantity (`∂u/∂x`, `∂²u/∂x²`, ...) from a
61/// field and a mesh, with no dependency on physical parameters or runtime
62/// context: `dx` comes entirely from `mesh`. Consumed transparently through
63/// [`crate::context::ContextCalculator`] (DD-012) — never called directly by a
64/// temporal integrator.
65///
66/// Uses an associated type `MeshType`, not a generic parameter, per DD-012
67/// amendment 1: each scheme implementation targets exactly one mesh family, so
68/// a generic `DiscreteOperator<M: Mesh>` would add flexibility no consumer
69/// needs, at the cost of object-safety (INV-4).
70///
71/// Distinct from [`FluxDivergenceOperator`] (DD-039), which computes the
72/// complete `∇·F(u, ∇u)` term and does need context access — see the module
73/// documentation above for the selection criterion between the two.
74pub trait DiscreteOperator: Send + Sync {
75    /// The mesh family this scheme is implemented for.
76    type MeshType: Mesh;
77
78    /// Applies the scheme to `field` on `mesh`, returning the differential
79    /// quantity the scheme computes.
80    fn apply(
81        &self,
82        field: &ContextValue,
83        mesh: &Self::MeshType,
84    ) -> Result<ContextValue, OxiflowError>;
85}

87/// Complete flux-divergence spatial operator (FV/WENO family) — DD-039.
88///
89/// Computes the full `∇·F(u, ∇u)` term of the canonical PDE form, which —
90/// unlike [`DiscreteOperator`] — requires knowledge of the physics of `F`
91/// (advection velocity, diffusion coefficient, ...), potentially varying in
92/// space or time. Carries [`RequiresContext`] and receives `ctx` directly on
93/// `apply()`.
94///
95/// A **sibling** trait to `DiscreteOperator`, not a sub-trait: the two
96/// `apply()` signatures diverge (with/without `ctx`), so a sub-trait
97/// relationship could not reuse the parent method cleanly (DD-039, option B
98/// rejected). Adding `ctx`/`RequiresContext` to `DiscreteOperator` itself was
99/// also rejected (DD-012, amendment 2; DD-039, option A) — it would force
100/// that capability onto FD schemes, which never need it.
101///
102/// Two consumption modes fall out of `RequiresContext` for free, with no new
103/// mechanism: a scheme with only constant physical parameters returns
104/// `required_variables() -> vec![]` and reads its own fields directly in
105/// `apply()`; a scheme with space/time-varying parameters declares them as
106/// [`crate::context::ContextVariable`]s in `required_variables()` and reads
107/// them from `ctx`, resolved and ordered beforehand by `chain.rs` (DD-009)
108/// exactly as for any `PhysicalModel`/`BoundaryCondition`.
109///
110/// Consumed by `DiscretizedModel<Op>` (DD-038), which binds `Op:
111/// FluxDivergenceOperator` — an FD scheme cannot be plugged in there; that
112/// becomes a compile error, not a silent runtime bug.
113pub trait FluxDivergenceOperator: RequiresContext + Send + Sync {
114    /// The mesh family this scheme is implemented for.
115    type MeshType: Mesh;
116
117    /// Applies the scheme to `field` on `mesh`, resolving any physical
118    /// parameter declared in [`RequiresContext::required_variables`] from
119    /// `ctx`, and returns `∇·F(u, ∇u)`.
120    fn apply(
121        &self,
122        field: &ContextValue,
123        mesh: &Self::MeshType,
124        ctx: &ComputeContext,
125    ) -> Result<ContextValue, OxiflowError>;
126
127    /// Half-width of the stencil this scheme reads on each side of a cell —
128    /// a property of the numerical scheme itself (HOW), never of the
129    /// physical model it backs (DD-043, amendment 1). No default: every
130    /// implementation must state its own value explicitly, the same
131    /// posture as [`crate::model::traits::RequiresContext::required_variables`].
132    ///
133    /// Read directly by the caller on the concrete `Op` — before it is ever
134    /// boxed into a `Box<dyn PhysicalModel>` — and handed explicitly to a
135    /// solver's own configuration (e.g.
136    /// `BackwardEulerSolver::with_jacobian_bandwidth(scheme.stencil_radius())`,
137    /// v0.6.0/#50). Never read back from a `PhysicalModel`: bandwidth is a
138    /// solver-side assertion (HOW), not a model declaration (WHAT) — see
139    /// DD-043, amendment 1, for the rejected alternative
140    /// (`PhysicalModel::jacobian_bandwidth`) and why it was rejected.
141    fn stencil_radius(&self) -> usize;
142}

146/// Boundary treatment shared by cell/face-based [`FluxDivergenceOperator`]
147/// families (`operators::fv`, `operators::weno`) — both face the same
148/// question: a finite `n`-node mesh only delimits `n−1` interior cells (FV)
149/// or lacks enough neighbors for a wide stencil at the edges (WENO) unless
150/// the domain wraps around.
151///
152/// Three families of treatment exist: periodic wrap (exact conservation by
153/// telescoping for FV; the only one implemented so far); ghost cells via the
154/// existing [`crate::boundary::BoundaryCondition`] system (a real
155/// integration, scheduled later this sprint, after `#49`); and FD-style
156/// one-sided truncation at the boundary (breaks FV's conservation property).
157/// Rather than picking one silently, `FluxBoundary` makes the choice an
158/// explicit constructor parameter on every operator that needs it.
159/// `#[non_exhaustive]` (DD-022) signals this is an extension point, not a
160/// closed set — only [`FluxBoundary::Periodic`] exists today.
161#[non_exhaustive]
162#[derive(Debug, Clone)]
163pub enum FluxBoundary {
164    /// The domain wraps around: the last cell/node's right neighbor is the
165    /// first cell/node. Gives an exact discrete conservation property for
166    /// FV — the sum of per-cell flux divergences telescopes to zero, since
167    /// every internal face flux is counted once as a `+` contribution (its
168    /// left cell) and once as a `−` contribution (its right cell).
169    Periodic,
170
171    /// One-sided (decentered) fallback at the two boundary cells — no wrap,
172    /// no invented value outside the domain.
173    ///
174    /// Each boundary cell is assigned the same stencil formula already
175    /// computed for the nearest cell that has a full interior stencil,
176    /// rather than reconstructing a face flux across the domain edge — the
177    /// same posture `operators::fd`'s `CenteredLaplacian`/`UpwindGradient`
178    /// already take at their own boundary nodes (DD-012): reuse an
179    /// available one-sided formula instead of inventing an exterior point.
180    /// Breaks FV's exact conservation property (`Periodic`'s telescoping
181    /// sum no longer holds, since the outermost face fluxes are each
182    /// counted only once, not twice) — a documented compromise (DD-042),
183    /// not an oversight.
184    ///
185    /// Requires enough cells for at least one cell with a full stencil to
186    /// exist: 3 for `operators::fv` (2-point stencil, one interior cell to
187    /// borrow from), 3 for `WENO3` (needs 1 cell of margin each side), 5
188    /// for `WENO5` (needs 2 cells of margin each side) — checked explicitly
189    /// by each operator, not silently producing a degenerate result.
190    Truncation,
191
192    /// Ghost-cell fallback (DD-042): the two boundary cells' missing
193    /// neighbors are supplied by the real [`crate::boundary::BoundaryCondition`]
194    /// already registered on the `Domain` for that edge, via
195    /// [`crate::boundary::BoundaryCondition::ghost_value`] — the one variant
196    /// of `FluxBoundary` that references actual boundary physics, unlike
197    /// `Periodic`/`Truncation`, which are purely numerical stencil-closure
198    /// conventions.
199    ///
200    /// `Arc` rather than `Box`: the same `BoundaryCondition` instance is
201    /// typically already owned by `Domain` (`Vec<Box<dyn BoundaryCondition>>`)
202    /// and shared here, not duplicated.
203    ///
204    /// If the referenced BC does not override `ghost_value()` (still
205    /// returns `None`), `apply()` fails explicitly via
206    /// [`OxiflowError::PreconditionFailed`] rather than silently
207    /// substituting an approximation — see DD-042 for why no generic
208    /// per-`BoundaryType` fallback is attempted here.
209    GhostCell(
210        std::sync::Arc<dyn crate::boundary::BoundaryCondition>,
211        std::sync::Arc<dyn crate::boundary::BoundaryCondition>,
212    ),
213}

217/// Checks the explicit advective CFL stability condition `|v|·dt/dx ≤ 1`.
218///
219/// Shared by `operators::fv` and `operators::weno` — both are explicit
220/// advective schemes subject to the same necessary stability condition
221/// (documented for Lax–Wendroff-type schemes). Covers the *advective*
222/// condition only; a diffusive term's own stability limit (Fourier number
223/// `D·dt/dx² ≤ 0.5`) is a separate concern, not checked here — conceptually
224/// the solver's step-size control's responsibility, not a spatial
225/// operator's, if ever enforced.
226pub(crate) fn check_cfl(
227    context: &'static str,
228    velocity: f64,
229    dt: f64,
230    dx: f64,
231) -> Result<(), OxiflowError> {
232    let cfl = velocity.abs() * dt / dx;
233    if cfl > 1.0 {
234        return Err(OxiflowError::PreconditionFailed {
235            context,
236            message: format!(
237                "CFL condition violated: |v|*dt/dx = {cfl:.6} > 1.0 \
238                 (v = {velocity}, dt = {dt}, dx = {dx}) — explicit advection \
239                 scheme is unstable at this step size"
240            ),
241        });
242    }
243    Ok(())
244}

255/// Periodic index `i + k` wrapped into `[0, n)`, `k` possibly negative.
256pub(crate) fn wrap(i: usize, k: isize, n: usize) -> usize {
257    (i as isize + k).rem_euclid(n as isize) as usize
258}

260/// Computes the divergence of a wide-stencil face flux for a periodic
261/// domain — analogous to `operators::fv`'s two-point `periodic_divergence`,
262/// generalized to a `face_flux` that reads an arbitrary window of `u`
263/// rather than just the two immediately adjacent values.
264///
265/// `face_flux(u, n, i)` evaluates the flux at the face between node `i` and
266/// node `(i+1) % n`.
267pub(crate) fn periodic_wide_divergence(
268    u: &DVector<f64>,
269    dx: f64,
270    min_nodes: usize,
271    context: &'static str,
272    face_flux: impl Fn(&DVector<f64>, usize, usize) -> f64,
273) -> Result<DVector<f64>, OxiflowError> {
274    let n = u.len();
275    if n < min_nodes {
276        return Err(OxiflowError::InvalidDomain(format!(
277            "{context} requires at least {min_nodes} nodes, got {n}"
278        )));
279    }
280
281    let mut div = DVector::zeros(n);
282    for i in 0..n {
283        let flux_right = face_flux(u, n, i);
284        let flux_left = face_flux(u, n, (i + n - 1) % n);
285        div[i] = (flux_right - flux_left) / dx;
286    }
287    Ok(div)
288}

290/// Computes the one-sided (decentered) divergence for a non-periodic domain
291/// — [`FluxBoundary::Truncation`], wide-stencil generalization of
292/// `operators::fv`'s `truncated_divergence`.
293///
294/// `face_flux(u, n, k)` reads offsets in `[-margin_left, margin_right]`
295/// relative to `k` (asymmetric for upwind-biased stencils — e.g. WENO3's
296/// left-biased reconstruction reads `{k−1, k, k+1}`, its right-biased one
297/// reads `{k, k+1, k+2}`; the caller computes `margin_left`/`margin_right`
298/// from the reconstruction actually selected, not a fixed value, so the
299/// boundary zone is no wider than the stencil in use actually requires). A
300/// cell `i`'s divergence reads `face_flux(i)` (right face) and
301/// `face_flux(i−1)` (left face); the safe (non-wrapping) range for both is
302/// `[margin_left+1, n−1−margin_right]`. Cells outside that range — lacking a
303/// full stencil on at least one side — are assigned the same value already
304/// computed for the nearest safe cell, the same posture as
305/// `operators::fv::truncated_divergence` and `operators::fd`'s boundary
306/// stencils. The two boundary zones are generally of different sizes
307/// (`margin_left+1` cells on the left, `margin_right` cells on the right) —
308/// an inherent consequence of `face(i)` and `face(i−1)` both needing to be
309/// safe for cell `i`, not an asymmetry bug.
310pub(crate) fn truncated_wide_divergence(
311    u: &DVector<f64>,
312    dx: f64,
313    margin_left: usize,
314    margin_right: usize,
315    context: &'static str,
316    face_flux: impl Fn(&DVector<f64>, usize, usize) -> f64,
317) -> Result<DVector<f64>, OxiflowError> {
318    let n = u.len();
319    let min_nodes = margin_left + margin_right + 2;
320    if n < min_nodes {
321        return Err(OxiflowError::InvalidDomain(format!(
322            "{context} with FluxBoundary::Truncation requires at least {min_nodes} nodes \
323             for this upwind direction (margin_left={margin_left}, margin_right={margin_right}), \
324             got {n}"
325        )));
326    }
327
328    let safe_start = margin_left + 1;
329    let safe_end = n - 1 - margin_right; // inclusive
330
331    let mut div = DVector::zeros(n);
332    for i in safe_start..=safe_end {
333        let flux_right = face_flux(u, n, i);
334        let flux_left = face_flux(u, n, i - 1);
335        div[i] = (flux_right - flux_left) / dx;
336    }
337    for i in 0..safe_start {
338        div[i] = div[safe_start];
339    }
340    for i in (safe_end + 1)..n {
341        div[i] = div[safe_end];
342    }
343    Ok(div)
344}

346/// Builds a ghost-padded copy of `u` for [`FluxBoundary::GhostCell`] —
347/// `margin_left` ghost values prepended, `u` in the middle, `margin_right`
348/// ghost values appended, so that every real cell's stencil (however wide)
349/// finds all its neighbors in-bounds with plain indexing — no wraparound.
350///
351/// Each ghost value comes from [`BoundaryCondition::ghost_value`] at the
352/// depth matching its distance from the domain edge, paired with the real
353/// interior value at the *symmetric* depth (method of images — DD-042,
354/// amendment 1): depth `k` pairs with interior index `k−1` from that edge.
355/// Every layer is derived directly from a real interior value, never from a
356/// previously computed ghost layer, so a Robin condition's exactness does
357/// not erode with depth.
358pub(crate) fn ghost_padded_field(
359    u: &DVector<f64>,
360    dx: f64,
361    margins: (usize, usize),
362    bcs: (&dyn BoundaryCondition, &dyn BoundaryCondition),
363    context: &'static str,
364) -> Result<Vec<f64>, OxiflowError> {
365    let (margin_left, margin_right) = margins;
366    let (left_bc, right_bc) = bcs;
367    let n = u.len();
368    let mut extended = Vec::with_capacity(n + margin_left + margin_right);
369
370    // Left ghosts, deepest first, so the array reads left-to-right in
371    // increasing physical position once `u` follows.
372    for depth in (1..=margin_left).rev() {
373        let interior_at_depth = u[depth - 1];
374        let g = left_bc
375            .ghost_value(depth, interior_at_depth, dx)
376            .ok_or_else(|| OxiflowError::PreconditionFailed {
377                context,
378                message: format!(
379                    "left boundary condition ({:?}) does not supply a ghost value at depth \
380                     {depth} — FluxBoundary::GhostCell requires an exact value, not a generic \
381                     fallback",
382                    left_bc.boundary_type()
383                ),
384            })?;
385        extended.push(g);
386    }
387
388    extended.extend(u.iter().copied());
389
390    for depth in 1..=margin_right {
391        let interior_at_depth = u[n - depth];
392        let g = right_bc
393            .ghost_value(depth, interior_at_depth, dx)
394            .ok_or_else(|| OxiflowError::PreconditionFailed {
395                context,
396                message: format!(
397                    "right boundary condition ({:?}) does not supply a ghost value at depth \
398                     {depth} — FluxBoundary::GhostCell requires an exact value, not a generic \
399                     fallback",
400                    right_bc.boundary_type()
401                ),
402            })?;
403        extended.push(g);
404    }
405
406    Ok(extended)
407}

409/// Computes the divergence for [`FluxBoundary::GhostCell`] by delegating to
410/// `face_flux` over a [`ghost_padded_field`] — every real cell's stencil,
411/// including the two boundary ones, reads only in-bounds data, so no
412/// separate boundary-cell code path is needed here (unlike
413/// [`truncated_wide_divergence`]): the padding does the work.
414pub(crate) fn ghost_cell_wide_divergence(
415    u: &DVector<f64>,
416    dx: f64,
417    margins: (usize, usize),
418    bcs: (&dyn BoundaryCondition, &dyn BoundaryCondition),
419    context: &'static str,
420    face_flux: impl Fn(&DVector<f64>, usize, usize) -> f64,
421) -> Result<DVector<f64>, OxiflowError> {
422    let n = u.len();
423    let margin_left = margins.0;
424    let extended = ghost_padded_field(u, dx, margins, bcs, context)?;
425    let extended = DVector::from_vec(extended);
426    let m = extended.len();
427
428    let mut div = DVector::zeros(n);
429    for i in 0..n {
430        let k = i + margin_left;
431        let flux_right = face_flux(&extended, m, k);
432        let flux_left = face_flux(&extended, m, k - 1);
433        div[i] = (flux_right - flux_left) / dx;
434    }
435    Ok(div)
